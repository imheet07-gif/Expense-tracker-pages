function cardByBank(bank) {
  const rupay = /rupay/.test(bank), key = bank.replace(/\s*rupay/, "")

// Finds your three Notion tables by name (they must be connected to your integration).
function findTables(token) {
  const props = PropertiesService.getScriptProperties()
  const find = name => {
    const saved = props.getProperty("DB_" + name)
    if (saved) return saved
    const res = notion(token, "post", "search", { query: name, filter: { property: "object", value: "database" }, page_size: 20 })
    const hit = (res.results || []).find(d => (d.title || []).map(t => t.plain_text).join("").trim().toLowerCase().startsWith(name.toLowerCase()))
    if (hit) props.setProperty("DB_" + name, hit.id.replace(/-/g, ""))
    return hit ? hit.id.replace(/-/g, "") : ""
  }
  ALERTS_DB = find("CC Transactions"); EXPENSES_DB = find("Expenses"); BILLS_DB = find("Card Bills"); LOANS_DB = find("Loans"); PEOPLE_DB = find("People"); DIARY_DB = find("Money Diary")
  if (!ALERTS_DB || !EXPENSES_DB) throw new Error("Couldn't find tables named \"Expenses\" and \"CC Transactions\" in your Notion. Duplicate the Expense Tracker template, then open its page › ••• › Connections and add your integration.")
}

function install() {
  const props = PropertiesService.getScriptProperties()
  if (NOTION_TOKEN) props.setProperty("NOTION_TOKEN", NOTION_TOKEN.trim())
  if (!props.getProperty("NOTION_TOKEN")) throw new Error("Paste your Notion token on the NOTION_TOKEN line first, then run install again.")
  const token = props.getProperty("NOTION_TOKEN")
  findTables(token)
  const found = detectAndSave(token)
  Logger.log(`Found ${found.cards.length} card(s): ${found.cards.map(c => c.name).join(", ") || "none yet (they're added as alerts arrive)"}`)
  uninstall()
  ScriptApp.newTrigger("syncAlerts").timeBased().everyMinutes(10).create()
  Logger.log("Installed: runs every 10 minutes. Now clear the token from the NOTION_TOKEN line (leave the quotes) and Save.")
  syncAlerts()
}

function uninstall() {
  ScriptApp.getProjectTriggers().filter(t => t.getHandlerFunction() === "syncAlerts").forEach(t => ScriptApp.deleteTrigger(t))
}

function syncAlerts() {

  const lock = LockService.getScriptLock()
  if (!lock.tryLock(5000)) return
  try {
    const token = PropertiesService.getScriptProperties().getProperty("NOTION_TOKEN")
    if (!token) throw new Error("Not set up yet. Run install() first.")
    findTables(token)
    loadCardRegistry(token)
    if (Date.now() - Number(SAVED.DETECTED_AT || 0) > 864e5) { try { detectAndSave(token) } catch (e) { Logger.log("card scan: " + e) } }
    const cutoff = Date.now() - 2 * 864e5 - 36e5

    }

    const recent = loadRecent(token)

    const skipNames = loadSkipNames(token)
    const counts = { added: 0, review: 0, dup: 0, ignored: 0, fxWait: 0 }
    for (const m of msgs) {
      const id = m.getId()
      if (seen.has(id)) continue
      seen.add(id)
      const r = parseAlert({ id, subject: m.getSubject(), from: m.getFrom(), body: m.getPlainBody(), date: m.getDate() })
      if (!r || r.skip) { counts.ignored++; continue }

    }

    } catch (e) {

// ---------- foreign currency ----------
// The alert has no rupee amount, so it's estimated from that day's exchange rate plus the card's forex
// markup and 18% GST on that markup. Markups are read from the Card Bills table (see below).
// ---------- forex markup per card ----------
// Kept on the Card Bills table's description in Notion, so you can change them from the dashboard's
// Settings (or in Notion) without touching any code. New foreign spends use the new number.
const MARKUP_DEFAULTS = Object.fromEntries(MY_CARDS.map(c => [c.name, /scapia/i.test(c.name) ? 0 : 3.5]))
const MARKUP_NAMES = Object.fromEntries(MY_CARDS.map(c => [c.name, c.name.replace(/\s*•+\s*\d+\s*$/, "")]))
const MARKUP_LABEL = "Forex markup (% added to foreign-currency spends, before 18% GST on it):"
function parseMarkups(desc) {
  const out = {}, s = String(desc || "")
  const at = s.indexOf("Forex markup")
  if (at < 0) return null
  const part = s.slice(at)
  for (const [card, name] of Object.entries(MARKUP_NAMES)) {
    const esc = name.replace(/[.*+?^${}()|[\]\\]/g, "\\$&") + (name === "Scapia" ? "(?!\\s*RuPay)" : "")
    const m = part.match(new RegExp(esc + "\\s*[:=]?\\s*(\\d+(?:\\.\\d+)?)\\s*%", "i"))
    out[card] = m ? Number(m[1]) : MARKUP_DEFAULTS[card]
  }
  return out
}
const formatMarkups = m => MARKUP_LABEL + " " + Object.entries(MARKUP_NAMES).map(([card, name]) => `${name} ${Number(m[card] != null ? m[card] : MARKUP_DEFAULTS[card])}%`).join(" · ")
// Replaces (or adds) the markup sentence in a description, keeping anything else written there.
function withMarkups(desc, m) {
  const s = String(desc || ""), at = s.indexOf("Forex markup")
  const before = (at < 0 ? s : s.slice(0, at)).trim()
  return (before ? before + "\n" : "") + formatMarkups(m)
}

let MARKUPS = null
function loadMarkups(token) {
  if (MARKUPS) return MARKUPS
  try {
    const db = notion(token, "get", `databases/${BILLS_DB}`)
    const desc = (db.description || []).map(t => t.plain_text).join("")
    MARKUPS = parseMarkups(desc)
    if (!MARKUPS) {   // first run: write the defaults so you can see and edit them
      MARKUPS = Object.assign({}, MARKUP_DEFAULTS)
      notion(token, "patch", `databases/${BILLS_DB}`, { description: [{ type: "text", text: { content: withMarkups(desc, MARKUPS) } }] })
    }
  } catch (e) {
    Logger.log("Couldn't read forex markups from Card Bills, using defaults: " + (e.message || e))
    MARKUPS = Object.assign({}, MARKUP_DEFAULTS)
  }
  return MARKUPS
}

// Sets r.amount to the estimated rupees you'll be billed, r.currency to e.g. "GBP 45.00". False if no rate yet.
function toRupees(r, token) {
  const day = istDay(r.time), rate = fxRate(r.fx.cur, day)
  if (!rate) return false
  const m = loadMarkups(token)[r.card], markup = (m != null ? m : 3.5) / 100
  r.amount = Math.round(r.fx.amt * rate * (1 + markup * 1.18) * 100) / 100
  r.currency = `${r.fx.cur} ${r.fx.amt.toFixed(2)}`
  r.raw = `Estimated: ${r.currency} × ${rate.toFixed(2)} (rate for ${day}) + ${(markup * 100).toFixed(1)}% forex markup + 18% GST on it`
  return true
}
// Rupees per unit of a currency on a day (ECB reference rate via frankfurter.app; open.er-api.com as a fallback).
function fxRate(cur, day) {
  const props = PropertiesService.getScriptProperties(), key = `fx|${cur}|${day}`
  const saved = props.getProperty(key)
  if (saved) return Number(saved)
  let rate = null
  const get = url => { try { const res = UrlFetchApp.fetch(url, { muteHttpExceptions: true }); return res.getResponseCode() === 200 ? JSON.parse(res.getContentText()) : null } catch (e) { return null } }
  const a = get(`https://api.frankfurter.app/${day}?from=${cur}&to=INR`)
  if (a && a.rates && a.rates.INR) rate = a.rates.INR
  if (!rate) { const b = get(`https://open.er-api.com/v6/latest/${cur}`); if (b && b.rates && b.rates.INR) rate = b.rates.INR }
  if (rate) props.setProperty(key, String(rate))
  return rate
}

function notion(token, method, path, body) {
  const res = UrlFetchApp.fetch("https://api.notion.com/v1/" + path, {
    method, muteHttpExceptions: true, contentType: "application/json",
    headers: { Authorization: "Bearer " + token, "Notion-Version": "2022-06-28" },
    payload: body ? JSON.stringify(body) : undefined,
  })
  const json = JSON.parse(res.getContentText() || "{}")
  if (json.object === "error") {
    if (json.code === "object_not_found") throw new Error("Your Notion integration can't see CC Transactions. In Notion open that table › ••• › Connections and add the integration.")
    throw new Error("Notion: " + json.message)
  }
  return json
}

const plain = p => ((p && (p.title || p.rich_text)) || []).map(t => t.plain_text).join("")

// Rows from the last few days, used to skip duplicates.
function loadRecent(token) {
  const since = new Date(Date.now() - 4 * 864e5).toISOString().slice(0, 10)
  const out = []
  let cursor
  do {
    const res = notion(token, "post", `databases/${ALERTS_DB}/query`, Object.assign(
      { page_size: 100, filter: { property: "Date", date: { on_or_after: since } } }, cursor ? { start_cursor: cursor } : {}))
    for (const pg of res.results || []) {
      const p = pg.properties
      out.push({
        pageId: pg.id, messageId: plain(p["Message ID"]), merchant: plain(p.Merchant), amount: p.Amount.number || 0,
        card: p.Card.select ? p.Card.select.name : "", type: p.Type.select ? p.Type.select.name : "",
        source: p.Source.select ? p.Source.select.name : "", time: p.Date.date ? Date.parse(p.Date.date.start) : 0,
      })
    }
    cursor = res.has_more ? res.next_cursor : null
  } while (cursor)
  return out
}

function createRow(token, r) {
  const sel = v => ({ select: { name: v } })
  const txt = v => ({ rich_text: [{ text: { content: String(v).slice(0, 1900) } }] })
  const props = {
    Merchant: { title: [{ text: { content: r.merchant.slice(0, 200) } }] },
    Amount: { number: r.amount },
    Date: { date: { start: r.iso } },
    Card: sel(r.card), Channel: sel(r.channel), Type: sel(r.type), Category: sel(r.category),
    Source: sel(r.source), Status: sel(r.status), "Message ID": txt(r.messageId), Currency: txt(r.currency || "INR"),
  }
  if (r.raw) props.Raw = txt(r.raw)
  if (r.ignored) props.Ignored = { checkbox: true }
  if (r.account) props.Account = sel(r.account)
  return notion(token, "post", "pages", { parent: { database_id: ALERTS_DB }, properties: props })
}

const CREDIT_CARDS = MY_CARDS.map(c => c.name)

const DAY_MS = 864e5

// Calendar day (yyyy-mm-dd) in India for a timestamp.
const istDay = ms => new Date(ms + 330 * 6e4).toISOString().slice(0, 10)
const dayNum = iso => Math.round(Date.parse(String(iso).slice(0, 10) + "T00:00:00Z") / DAY_MS)

// Which payment methods you might have picked for an alert's channel.
function methodFits(channel, method) {
  if (!method) return true
  switch (channel) {
    case "Card": return method === "Credit Card"
    case "UPI on credit card": return method === "Credit Card" || method === "UPI"
    case "UPI from bank a/c": return method === "UPI"
    case "Other bank debit": return method === "Debit Card" || method === "UPI"
    default: return method !== "Cash"   // auto-debits, charges, cheques, bill payments
  }
}

const WORD_SKIP = new Set(["the", "and", "pvt", "ltd", "india", "private", "limited", "services", "online", "order", "via", "upi", "payment", "store", "stores", "food", "co"])
const words = s => new Set(String(s || "").toLowerCase().replace(/[^a-z0-9 ]+/g, " ").split(/\s+/).filter(w => w.length >= 3 && !WORD_SKIP.has(w)))
const norm = s => String(s || "").toLowerCase().replace(/[^a-z0-9]+/g, "")
function sameName(a, b) {
  const A = words(a), B = words(b)
  for (const w of A) for (const v of B) if (w === v || (w.length >= 4 && v.length >= 4 && (w.startsWith(v) || v.startsWith(w)))) return true
  return false
}

// e: { id, date, amount, payment, category, merchant, notes, created, alertIds, card }
// a: { id, time, day, amount, card, channel, type, merchant, ignored, expenseIds, status }
// Returns { links: [{ expenseId, alertIds, card }], open: [alerts not logged], noCard: [card expenses with no alert] }
function matchAlerts(expenses, alerts, opts = {}) {
  const fromDay = opts.fromDay || null
  const nowMs = opts.now || Date.now()
  // Foreign-currency alerts carry an estimated rupee amount, so they match within 3% instead of 0.5%.
  const tol = (amt, fx) => Math.max(1, amt * (fx ? 0.03 : 0.005))
  const isDues = e => e.category === "Credit Card Dues"

  // A spend cancelled by a reversal/refund of the same amount on the same card isn't flagged.
  const cancelled = new Set()
  const backs = alerts.filter(a => a.type === "Reversal" || a.type === "Refund")
  for (const b of backs) {
    const s = alerts.find(a => a.type === "Spend" && !cancelled.has(a.id) && a.card === b.card && Math.abs(a.amount - b.amount) <= 0.01 && b.time - a.time >= -36e5 && b.time - a.time <= 14 * DAY_MS)
    if (s) { cancelled.add(s.id); cancelled.add(b.id) }
  }

  const linkedAlert = new Set(alerts.filter(a => a.expenseIds && a.expenseIds.length).map(a => a.id))
  for (const e of expenses) for (const id of e.alertIds || []) linkedAlert.add(id)
  const freeA = alerts.filter(a => (a.type === "Spend" || a.type === "Bill payment") && !a.ignored && a.status !== "Duplicate" && !linkedAlert.has(a.id) && !cancelled.has(a.id) && (!fromDay || a.day >= fromDay))
  const freeE = expenses.filter(e => !(e.alertIds && e.alertIds.length) && e.payment !== "Cash" && e.date && (!fromDay || e.date >= shiftDay(fromDay, -2)))

  const fits = (e, a) => (a.type === "Bill payment") === isDues(e)
  function score(e, list) {
    const amt = list.reduce((s, a) => s + a.amount, 0)
    const diff = Math.abs(e.amount - amt)
    if (diff > tol(Math.max(e.amount, amt), list.some(a => a.fx))) return null
    let worst = 0, pen = 0, named = false
    for (const a of list) {
      if (!fits(e, a)) return null
      const dd = Math.abs(dayNum(e.date) - dayNum(a.day))
      if (dd > 2) return null
      worst = Math.max(worst, dd)
      if (!methodFits(a.channel, e.payment)) pen += 1.2
      if (sameName(e.merchant + " " + e.notes, a.merchant)) named = true
      // Logged on the day it happened: the closer the logging time is to the alert, the likelier.
      if (e.created && istDay(Date.parse(e.created)) === e.date && a.day === e.date) pen += Math.min(1, Math.abs(Date.parse(e.created) - a.time) / (12 * 36e5)) * 0.6
    }
    const fx = list.some(a => a.fx)
    return worst + pen / list.length + (fx ? diff / Math.max(amt, 1) * 20 : diff * 0.2) - (named ? 0.7 : 0) + (list.length - 1) * 0.4
  }

  const usedA = new Set(), usedE = new Set(), links = []
  const take = (exps, als) => {
    exps.forEach(e => usedE.add(e.id)); als.forEach(a => usedA.add(a.id))
    const credit = als.map(a => a.card).find(c => CREDIT_CARDS.includes(c)) || null
    for (const e of exps) links.push({ expenseId: e.id, alertIds: als.map(a => a.id), card: credit })
  }

  // 1. one alert ↔ one expense, best pairs first
  const pairs = []
  for (const e of freeE) for (const a of freeA) { const s = score(e, [a]); if (s != null && s < 3) pairs.push([s, e, a]) }
  pairs.sort((x, y) => x[0] - y[0])
  for (const [, e, a] of pairs) if (!usedE.has(e.id) && !usedA.has(a.id)) take([e], [a])

  // 2. one payment you split into 2–3 expenses (₹2,545 at Blinkit logged as ₹2,362 + ₹183)
  for (const a of freeA) {
    if (usedA.has(a.id)) continue
    const c = freeE.filter(e => !usedE.has(e.id) && e.amount < a.amount && fits(e, a) && Math.abs(dayNum(e.date) - dayNum(a.day)) <= 1 && methodFits(a.channel, e.payment)).slice(0, 14)
    const hit = combo(c, a.amount, tol(a.amount, a.fx), x => x.amount)
    // only when they're clearly the same shop: all named alike, or named like the alert
    if (hit && (hit.every(e => sameName(e.merchant + " " + e.notes, a.merchant)) || hit.every(e => norm(e.merchant) === norm(hit[0].merchant) && norm(e.merchant)))) take(hit, [a])
  }
  // 3. one expense paid in 2–3 goes (₹80 metro logged once, paid ₹40 + ₹40)
  for (const e of freeE) {
    if (usedE.has(e.id)) continue
    const c = freeA.filter(a => !usedA.has(a.id) && a.amount < e.amount && fits(e, a) && Math.abs(dayNum(e.date) - dayNum(a.day)) <= 1 && methodFits(a.channel, e.payment)).slice(0, 14)
    const hit = combo(c, e.amount, tol(e.amount), x => x.amount)
    if (hit && hit.every(a => norm(a.merchant) === norm(hit[0].merchant))) take([e], hit)   // e.g. two metro rides
  }

  const open = freeA.filter(a => !usedA.has(a.id)).sort((x, y) => y.time - x.time)
  const noCard = expenses.filter(e => e.payment === "Credit Card" && !e.card && !usedE.has(e.id) && !(e.alertIds && e.alertIds.length) && !isDues(e) &&
    (!fromDay || e.date >= fromDay) && dayNum(istDay(nowMs)) - dayNum(e.date) >= 1)
  return { links, open, noCard }
}

// Smallest set of 2–3 items whose amounts add up to target (within tol).
function combo(items, target, tol, val) {
  const n = items.length
  for (let i = 0; i < n; i++) for (let j = i + 1; j < n; j++) {
    if (Math.abs(val(items[i]) + val(items[j]) - target) <= tol) return [items[i], items[j]]
  }
  for (let i = 0; i < n; i++) for (let j = i + 1; j < n; j++) for (let k = j + 1; k < n; k++) {
    if (Math.abs(val(items[i]) + val(items[j]) + val(items[k]) - target) <= tol) return [items[i], items[j], items[k]]
  }
  return null
}
function shiftDay(iso, n) { return new Date(Date.parse(iso + "T00:00:00Z") + n * DAY_MS).toISOString().slice(0, 10) }

const skipKey = name => String(name || "").toLowerCase().replace(/[^a-z0-9]+/g, " ").trim()

function queryAll(token, db, body) {
  const out = []
  let cursor = null, pages = 0
  do {
    const res = notion(token, "post", `databases/${db}/query`, Object.assign({ page_size: 100 }, body, cursor ? { start_cursor: cursor } : {}))
    out.push.apply(out, res.results || [])
    cursor = res.has_more ? res.next_cursor : null
  } while (cursor && ++pages < 30)
  return out
}

// The same window the dashboard uses: from your first logged expense, at most CHECK_DAYS back.
function checkFromDay(token) {
  const first = notion(token, "post", `databases/${EXPENSES_DB}/query`, { page_size: 1, sorts: [{ property: "Date", direction: "ascending" }], filter: { property: "Date", date: { is_not_empty: true } } })

function matchAndLink(token) {
  const fromDay = checkFromDay(token)
  if (!fromDay) return { linked: 0, open: 0 }
  const expenses = queryAll(token, EXPENSES_DB, { filter: { property: "Date", date: { on_or_after: shiftDay(fromDay, -2) } } }).map(pg => {
    const p = pg.properties
    return { id: pg.id, date: p.Date.date ? p.Date.date.start.slice(0, 10) : "", amount: p.Amount.number || 0, payment: sel(p["Payment Method"]),
      category: sel(p.Category) || "Other", merchant: plain(p.Merchant), notes: plain(p.Notes), created: pg.created_time,
      alertIds: rel(p.Alerts), card: sel(p.Card) }
  })
  const alerts = queryAll(token, ALERTS_DB, { filter: { property: "Date", date: { on_or_after: shiftDay(fromDay, -3) } } }).map(pg => {
    const p = pg.properties, start = p.Date.date ? p.Date.date.start : pg.created_time
    const time = Date.parse(start.length === 10 ? start + "T12:00:00+05:30" : start)
    const card = sel(p.Card) || "Unknown"
    return { id: pg.id, time, day: istDay(time), amount: p.Amount.number || 0, card, merchant: plain(p.Merchant),
      channel: sel(p.Channel) || (card === "Bank account" ? "Other bank debit" : "Card"), type: sel(p.Type) || "Spend",
      ignored: !!(p.Ignored && p.Ignored.checkbox), expenseIds: rel(p.Expenses), fx: isForeign(plain(p.Currency)) }
  })
  const res = matchAlerts(expenses, alerts, { fromDay })
  const byId = {}
  expenses.forEach(e => byId[e.id] = e)
  let linked = 0
  for (const l of res.links) {
    if (linked >= MAX_LINKS) break
    const e = byId[l.expenseId]
    const ids = Array.from(new Set(e.alertIds.concat(l.alertIds)))
    const props = { Alerts: { relation: ids.map(id => ({ id })) } }
    if (l.card && !e.card) props.Card = { select: { name: l.card } }
    notion(token, "patch", `pages/${l.expenseId}`, { properties: props })
    linked++
  }
  return { linked, open: res.open.length }
}

// ---------- category learning: what you did before decides what an auto-logged expense becomes ----------
// history: [{ merchant: bank's text, name: what you called it, category, amount }] from expenses linked to bank alerts.
// No model: a vote among your own past choices, so every guess can say why ("Food & Dining: 6 of 6 times for Swiggy").
const LEARN_STOP = new Set(["pvt", "ltd", "private", "limited", "india", "ind", "systems", "system", "online", "order", "payments", "payment", "services", "service", "the", "and", "via", "upi", "pos", "co", "llp", "technologies", "tech", "retail", "marketplace", "priv", "pri", "noida", "mumbai", "bangalore", "bengaluru", "delhi", "boi", "cred"])
const learnKey = s => String(s || "").toLowerCase().replace(/\([^)]*\)/g, " ").replace(/[^a-z ]+/g, " ").split(/\s+/).filter(w => w.length >= 2 && !LEARN_STOP.has(w)).join(" ")
const learnTokens = s => learnKey(s).split(" ").filter(w => w.length >= 3)
// a payee that reads like a person's name (a rickshaw driver, a street stall) rather than a business
const LEARN_BIZ = /\b(enterprises?|traders?|trading|store|stores|mart|foods?|cafe|coffee|restaurant|kitchen|snacks|hub|labs?|medical|pharma|hospital|clinic|motors|auto|services|ltd|pvt|co|kiosk|canteen|shop|bakery|sweets)\b/i
const learnPersonish = s => { const t = String(s || "").trim(); return /^[A-Za-z][A-Za-z .'-]+$/.test(t) && t.split(/\s+/).length >= 2 && t.split(/\s+/).length <= 4 && !LEARN_BIZ.test(t) }
function voteOf(list) {
  const by = {}
  for (const h of list) by[h.category] = (by[h.category] || 0) + 1
  const top = Object.entries(by).sort((a, b) => b[1] - a[1])[0]
  return top ? { category: top[0], n: top[1], of: list.length } : null
}
const topName = list => { const by = {}; for (const h of list) if (h.name) by[h.name] = (by[h.name] || 0) + 1; const t = Object.entries(by).sort((a, b) => b[1] - a[1])[0]; return t ? t[0] : "" }
// guess: the category the alert parser's rules would give (fallback when you've never paid this merchant)
function learnCategory(history, alert, guess) {
  const key = learnKey(alert.merchant), toks = learnTokens(alert.merchant)
  const say = (v, how, conf) => ({ category: v.category, confidence: conf, reason: `${how}: ${v.n} of ${v.of} time${v.of === 1 ? "" : "s"}`, name: "" })
  // 1. the same merchant as before
  const same = key ? history.filter(h => learnKey(h.merchant) === key) : []
  if (same.length) {
    const v = voteOf(same), share = v.n / v.of
    const conf = share >= 0.75 && v.n >= 2 ? "high" : share >= 0.6 && (v.n >= 2 || v.category === guess) ? "medium" : "low"
    const out = say(v, `You chose ${v.category} for this merchant`, conf)
    out.name = topName(same)
    return out
  }
  // 2. a close variant: shares a distinctive word ("SWIGGY INSTAMART" after "SWIGGY")
  if (toks.length) {
    const near = history.filter(h => { const t = learnTokens(h.merchant); return t.some(w => w.length >= 4 && toks.includes(w)) })
    if (near.length) { const v = voteOf(near); const out = say(v, `You chose ${v.category} for similar merchants`, v.n / v.of >= 0.75 && v.n >= 2 ? "medium" : "low"); out.name = ""; return out }
  }
  // 3. the built-in rules, when they recognise the merchant (Uber → Travel, Netflix → Subscriptions)
  if (guess && guess !== "Other" && guess !== "Personal") return { category: guess, confidence: "medium", reason: "From the merchant's name", name: "" }
  // 4. a person's name you've never paid: what your other payments to individuals of about this amount were
  if ((guess === "Personal" || learnPersonish(alert.merchant)) && alert.amount > 0) {
    const ppl = history.filter(h => (h.guess === "Personal" || learnPersonish(h.merchant)) && h.amount > 0)
      .map(h => ({ h, d: Math.abs(Math.log(h.amount / alert.amount)) })).sort((a, b) => a.d - b.d).slice(0, 5).filter(x => x.d < 1.2).map(x => x.h)
    if (ppl.length >= 2) { const v = voteOf(ppl); return say(v, `Payments of about this size to individuals were ${v.category}`, "low") }
  }
  // 5. nothing to go on
  return { category: guess === "Personal" ? "Other" : guess || "Other", confidence: "low", reason: "New merchant", name: "" }
}

// ---------- auto-logging: bank alerts become expenses by themselves ----------
// Turned on from the dashboard (gear › Logging); the choice is kept on the Expenses table's description as
// "Auto-log: fill · preview · since 2026-10-07T04:30:00Z". Modes:
//   fill  every alert you haven't logged yourself within an hour becomes an expense
//   auto  every alert becomes an expense straight away
// "preview" writes nothing (the dashboard shows what would have been added); "sms" means your iPhone does it from bank SMS,
// so this script stays out of the way. Only alerts from after "since" are ever logged, so turning it on doesn't add your past.
const AUTO_WAIT_MIN = 60
const AUTO_MAX = 30   // per run
function readAutoLog(token) {
  const db = notion(token, "get", `databases/${EXPENSES_DB}`)
  const desc = (db.description || []).map(t => t.plain_text).join("")
  const m = desc.match(/Auto-log:\s*(off|fill|auto)([^\n]*)/i)
  if (!m) return { mode: "off" }
  const since = (m[2].match(/since\s+(\S+)/i) || [])[1] || ""
  return { mode: m[1].toLowerCase(), preview: /preview/i.test(m[2]), sms: /\bsms\b/i.test(m[2]), since: Date.parse(since) || Date.now(), props: db.properties || {} }
}
// What you chose before: each expense linked to a bank alert pairs the bank's merchant text with your category and name.
function autoHistory(token) {
  const cache = CacheService.getScriptCache(), hit = cache.get("autoHistory")
  if (hit) try { return JSON.parse(hit) } catch (e) {}
  const since = shiftDay(istDay(Date.now()), -150)
  const exp = {}
  for (const pg of queryAll(token, EXPENSES_DB, { filter: { property: "Date", date: { on_or_after: since } } })) {
    const p = pg.properties
    exp[pg.id.replace(/-/g, "")] = { category: sel(p.Category) || "Other", name: plain(p.Merchant), amount: (p.Amount && p.Amount.number) || 0 }
  }
  const out = [], used = new Set()
  for (const pg of queryAll(token, ALERTS_DB, { filter: { and: [{ property: "Date", date: { on_or_after: since } }, { property: "Expenses", relation: { is_not_empty: true } }] } })) {
    const p = pg.properties
    for (const id of rel(p.Expenses)) { const e = exp[id.replace(/-/g, "")]; if (!e) continue; used.add(id.replace(/-/g, "")); out.push({ merchant: plain(p.Merchant), name: e.name, category: e.category, amount: (p.Amount && p.Amount.number) || e.amount, guess: guessCategory(plain(p.Merchant)) }) }
  }
  // expenses you logged without an alert still teach the names you use ("Swiggy" → Food & Dining)
  for (const [id, e] of Object.entries(exp)) if (!used.has(id) && e.name) out.push({ merchant: e.name, name: e.name, category: e.category, amount: e.amount, guess: guessCategory(e.name) })
  try { cache.put("autoHistory", JSON.stringify(out.slice(-1500)), 1800) } catch (e) {}
  return out
}
function ensureLoggedBy(token, props) {
  if (props && props["Logged by"]) return
  notion(token, "patch", `databases/${EXPENSES_DB}`, { properties: { "Logged by": { select: { options: [{ name: "Auto", color: "blue" }, { name: "Auto · check", color: "orange" }, { name: "SMS", color: "green" }] } } } })
}
function autoLog(token) {
  if (!EXPENSES_DB || !ALERTS_DB) return { added: 0 }
  const s = readAutoLog(token)
  if (s.mode === "off" || s.preview || s.sms) return { added: 0, mode: s.mode }
  const now = Date.now(), from = shiftDay(istDay(now), -3)
  const alerts = queryAll(token, ALERTS_DB, { filter: { property: "Date", date: { on_or_after: from } } }).map(pg => {
    const p = pg.properties, start = p.Date.date ? p.Date.date.start : pg.created_time
    return { id: pg.id, time: Date.parse(start.length === 10 ? start + "T12:00:00+05:30" : start), amount: (p.Amount && p.Amount.number) || 0, card: sel(p.Card) || "Unknown",
      merchant: plain(p.Merchant), channel: sel(p.Channel), type: sel(p.Type) || "Spend", category: sel(p.Category), status: sel(p.Status), ignored: !!(p.Ignored && p.Ignored.checkbox),
      expenseIds: rel(p.Expenses), currency: plain(p.Currency) }
  })
  // a spend cancelled by a refund or reversal of the same amount on the same card isn't logged
  const backs = alerts.filter(a => a.type === "Refund" || a.type === "Reversal")
  const cancelled = a => backs.some(b => b.card === a.card && Math.abs(b.amount - a.amount) <= 0.01 && b.time >= a.time - 36e5)
  const due = alerts.filter(a => a.type === "Spend" && !a.ignored && a.status !== "Duplicate" && !a.expenseIds.length && a.time >= s.since && !cancelled(a) &&
    !/^(Loan EMI)$/.test(a.category) && !/bill payment/i.test(a.channel || "") && (s.mode === "auto" || now - a.time >= AUTO_WAIT_MIN * 6e4))
  if (!due.length) return { added: 0, mode: s.mode }
  ensureLoggedBy(token, s.props)
  const history = autoHistory(token)
  let added = 0
  for (const a of due.slice(0, AUTO_MAX)) {
    const guess = ALERT_TO_EXPENSE_CAT[a.category] || guessCategory(a.merchant)
    const p = learnCategory(history, a, ALERT_TO_EXPENSE_CAT[guess] || guess)
    const name = p.name || autoName(a.merchant)
    const fx = a.currency && !/^\s*INR\b/i.test(a.currency) ? `${a.currency}, rupees estimated` : ""
    const props = {
      "Category Text": { title: [{ text: { content: p.category } }] }, Category: { select: { name: p.category } }, Amount: { number: a.amount },
      Date: { date: { start: istDay(a.time) } }, Merchant: { rich_text: [{ text: { content: name.slice(0, 200) } }] },
      "Payment Method": { select: { name: methodFor(a) } }, Alerts: { relation: [{ id: a.id }] },
      Notes: { rich_text: [{ text: { content: [fx, `Auto-logged · ${p.reason}`].filter(Boolean).join(" · ").slice(0, 400) } }] },
      "Logged by": { select: { name: p.confidence === "low" ? "Auto · check" : "Auto" } },
    }
    if (CREDIT_CARDS.includes(a.card)) props.Card = { select: { name: a.card } }
    notion(token, "post", "pages", { parent: { database_id: EXPENSES_DB }, properties: props })
    history.push({ merchant: a.merchant, name, category: p.category, amount: a.amount, guess })
    added++
  }
  return { added, mode: s.mode }
}
// "SWIGGY PVT LTD FOOD1" → "Swiggy Food"
function autoName(m) {
  let t = String(m || "").replace(/\b(pvt|private|ltd|limited|llp|india|ind|systems?|technologies|services|online|order)\b\.?/gi, " ").replace(/\b\w*\d\w*\b/g, " ").replace(/\s+/g, " ").trim() || String(m || "Unknown")
  if (t === t.toUpperCase()) t = t.toLowerCase().replace(/\b[a-z]/g, c => c.toUpperCase())
  return t
}

// ---------- card bills: read the bill, check it against your card alerts ----------
// A bill should equal: whatever was left unpaid from the last bill + spends − refunds in this billing period.
// Anything more is a charge you got no alert for: late fee, interest, annual fee + 18% GST, forex markup.
// Transactions in the last couple of days of a period often post on the next bill (and vice versa), so
// those are allowed to move between bills before anything is flagged.

const BILL_TOLERANCE = 2          // ₹ of rounding allowed before anything is flagged
const MONTHS = { jan: 0, feb: 1, mar: 2, apr: 3, may: 4, jun: 5, jul: 6, aug: 7, sep: 8, oct: 9, nov: 10, dec: 11 }
const isoOf = (y, m, d) => `${y}-${String(m + 1).padStart(2, "0")}-${String(d).padStart(2, "0")}`
const billNum = s => Number(String(s).replace(/,/g, ""))
const rupees = n => "₹" + (Math.round(n * 100) / 100).toLocaleString("en-IN", { minimumFractionDigits: n % 1 ? 2 : 0, maximumFractionDigits: 2 })
const shortDay = iso => { const [y, m, d] = iso.split("-").map(Number); return `${d} ${Object.keys(MONTHS)[m - 1].replace(/^./, c => c.toUpperCase())}` }

// "25 Aug 2026", "October 08, 2026", "Sep 17, 2026", "05-11-26", "05/11/2026" → yyyy-mm-dd
function billDay(s) {
  s = String(s || "").trim()
  let m = s.match(/^(\d{1,2})\s+([A-Za-z]{3})[A-Za-z]*,?\s+(\d{4})/)
  if (m && MONTHS[m[2].toLowerCase()] != null) return isoOf(+m[3], MONTHS[m[2].toLowerCase()], +m[1])
  m = s.match(/^([A-Za-z]{3})[A-Za-z]*\s+(\d{1,2}),?\s+(\d{4})/)
  if (m && MONTHS[m[1].toLowerCase()] != null) return isoOf(+m[3], MONTHS[m[1].toLowerCase()], +m[2])
  m = s.match(/^(\d{1,2})[-\/](\d{1,2})[-\/](\d{2,4})/)
  if (m) return isoOf(m[3].length === 2 ? 2000 + +m[3] : +m[3], +m[2] - 1, +m[1])
  return null
}
const DATE_RE = "(\\d{1,2}\\s+[A-Za-z]{3,9},?\\s+\\d{4}|[A-Za-z]{3,9}\\s+\\d{1,2},?\\s+\\d{4}|\\d{1,2}[-\\/]\\d{1,2}[-\\/]\\d{2,4})"
const AMT_RE = "(?:₹|Rs\\.?|INR)\\s*([\\d,]+(?:\\.\\d+)?)"

// Which card a bill is for. cardFor(text, digits) comes from the alert parser (cardOf).
// Returns { card, statementDay, cycleFrom?, cycleTo?, dueDay?, total, minDue?, source } or null.
function parseBill(msg, cardFor) {
  const subject = String(msg.subject || ""), from = String(msg.from || "")
  const isSms = /\[CC-SMS\]/i.test(subject)
  const text = String(msg.body || "").replace(/&amp;/g, "&").replace(/&#39;/g, "'").replace(/\|/g, " ").replace(/\s+/g, " ").trim()
  const sentDay = (() => { const d = msg.date instanceof Date ? msg.date : new Date(msg.date); return new Date(d.getTime() + 330 * 6e4).toISOString().slice(0, 10) })()
  const grab = re => { const m = text.match(re); return m ? m[1] : null }
  let m

  // Scapia's own statement email
  if (/scapia|federal/i.test(from) && /statement/i.test(subject)) {
    const total = grab(new RegExp("Total Amount Due\\s*" + AMT_RE, "i"))
    if (total == null) return null
    const cyc = text.match(new RegExp("(?:cycle|statement for)\\s+" + DATE_RE + "\\s*[-–]\\s*" + DATE_RE, "i"))
    return { card: /rupay/i.test(subject + " " + text.slice(0, 300)) ? cardByBank("scapia rupay") : cardByBank("scapia"), source: "Scapia",
      statementDay: billDay(grab(new RegExp("Statement Date:?\\s*" + DATE_RE, "i"))) || sentDay,
      cycleFrom: cyc ? billDay(cyc[1]) : null, cycleTo: cyc ? billDay(cyc[2]) : null,
      dueDay: billDay(grab(new RegExp("Due Date:?\\s*" + DATE_RE, "i"))),
      total: billNum(total), minDue: billNum(grab(new RegExp("Minimum Amount Due\\s*" + AMT_RE, "i")) || 0) || null }
  }

  // CRED's "your bill has been generated" email (works for every card you've added to CRED)
  if (/cred\.club/i.test(from) && (m = text.match(/bill (?:for \w+ )?has been generated\s+(.+?)\s*•+\s*(\d{4,5})\s+bill summary/i))) {
    const total = grab(new RegExp("total amount due\\s*" + AMT_RE, "i"))
    if (total == null) return null
    const card = cardFor(m[1], m[2])
    if (!card || card === "Unknown") return null
    return { card, source: "CRED",
      statementDay: billDay(grab(new RegExp("bill generated on\\s*" + DATE_RE, "i"))) || sentDay,
      dueDay: billDay(grab(new RegExp("due date\\s*" + DATE_RE, "i"))),
      total: billNum(total), minDue: billNum(grab(new RegExp("minimum due\\s*" + AMT_RE, "i")) || 0) || null }
  }

  // A forwarded bank SMS: "...statement... Total Amt Due: INR 1,234.00 ... Min Amt Due INR 100 ... due by 05-11-26"
  if (isSms && /statement|bill/i.test(text) && (m = text.match(new RegExp("(?:total|tot\\.?)\\s*(?:amount|amt\\.?)?\\s*due\\s*(?:is|of|:|-)?\\s*" + AMT_RE, "i")))) {
    const digits = (text.match(/(?:card|xx+|ending|no\.?)\s*(?:no\.?\s*)?[x*]*(\d{4,5})\b/i) || [])[1]
    const card = cardFor(text, digits)
    if (!card || card === "Unknown") return null
    return { card, source: "SMS", statementDay: sentDay,
      dueDay: billDay(grab(new RegExp("(?:due (?:date|by|on)|pay by)\\s*:?\\s*" + DATE_RE, "i"))),
      total: billNum(m[1]), minDue: billNum(grab(new RegExp("min(?:imum)?\\.?\\s*(?:amount|amt\\.?)?\\s*due\\s*(?:is|of|:|-)?\\s*" + AMT_RE, "i")) || 0) || null }
  }
  return null
}

const dayNumB = iso => Math.round(Date.parse(iso + "T00:00:00Z") / 864e5)
const addDays = (iso, n) => new Date(Date.parse(iso + "T00:00:00Z") + n * 864e5).toISOString().slice(0, 10)
function monthBefore(iso) {
  const [y, m, d] = iso.split("-").map(Number)
  const t = new Date(Date.UTC(y, m - 2, 1)), last = new Date(Date.UTC(y, m - 1, 0)).getUTCDate()
  return isoOf(t.getUTCFullYear(), t.getUTCMonth(), Math.min(d, last))
}

// bill: from parseBill. prev: the card's previous bill { statementDay, total } or null.
// alerts: [{ day, amount, type, merchant, card, fx }] for this card around the period; fx is e.g. "GBP 45.00"
// when the rupee amount was estimated from an exchange rate (then the check allows 1.5% of those spends).
// coverFrom: the first day you have alerts for at all (older periods can't be checked).
// Returns { cycleFrom, cycleTo, expected, difference, result, details, moved }
function checkBill(bill, prev, alerts, coverFrom) {
  const to = bill.cycleTo || addDays(bill.statementDay, -1)
  const from = bill.cycleFrom || (prev ? (prev.cycleTo ? addDays(prev.cycleTo, 1) : prev.statementDay) : monthBefore(bill.statementDay))
  const out = { cycleFrom: from, cycleTo: to }
  const mine = alerts.filter(a => a.card === bill.card)
  const sign = a => a.type === "Spend" ? 1 : (a.type === "Refund" || a.type === "Reversal") ? -1 : 0
  const inCycle = mine.filter(a => sign(a) && a.day >= from && a.day <= to)
  // What was left from the last bill: its total minus what you paid between the two statements.
  const paidFrom = prev ? prev.statementDay : from
  const payments = mine.filter(a => a.type === "Bill payment" && a.day >= paidFrom && a.day < bill.statementDay)
  const paid = payments.reduce((s, a) => s + a.amount, 0)
  const carry = prev ? Math.round((prev.total - paid) * 100) / 100 : 0

  if (coverFrom && from < coverFrom) {
    return Object.assign(out, { expected: null, difference: null, result: "Not enough data", moved: [],
      details: `Your card alerts start on ${shortDay(coverFrom)}, after this billing period began, so this bill can't be checked.` })
  }

  // Edge transactions that may have posted on the neighbouring bill.
  const early = mine.filter(a => sign(a) && a.day >= addDays(from, -3) && a.day < from)          // could be on this bill
  const late = inCycle.filter(a => a.day >= addDays(to, -1))                                      // could be on the next bill
  const after = mine.filter(a => sign(a) && a.day > to && a.day <= addDays(to, 1))                 // statement-day charges
  const edges = early.concat(late, after).slice(0, 10)
  const base = carry + inCycle.reduce((s, a) => s + sign(a) * a.amount, 0)
  const fxSpends = inCycle.filter(a => a.fx && sign(a) > 0)
  const fxTotal = fxSpends.reduce((s, a) => s + a.amount, 0)
  const tol = BILL_TOLERANCE + 0.015 * fxTotal
  let best = { expected: base, moved: [] }
  if (Math.abs(bill.total - base) > tol && edges.length) {
    let bestN = Infinity
    for (let mask = 1; mask < (1 << edges.length); mask++) {
      let e = base, n = 0, moved = []
      for (let i = 0; i < edges.length; i++) if (mask & (1 << i)) {
        const a = edges[i], inside = inCycle.includes(a)
        e += (inside ? -1 : 1) * sign(a) * a.amount; n++; moved.push({ a, out: inside })
      }
      if (Math.abs(bill.total - e) <= tol && n < bestN) { bestN = n; best = { expected: e, moved } }
    }
  }
  const expected = Math.round(best.expected * 100) / 100
  const diff = Math.round((bill.total - expected) * 100) / 100
  const spends = inCycle.filter(a => sign(a) > 0).length
  const parts = [`${spends ? spends : "no"} charge${spends === 1 ? "" : "s"} from ${shortDay(from)} to ${shortDay(to)}`]
  const gap = prev ? dayNumB(bill.statementDay) - dayNumB(prev.statementDay) : 0
  if (prev && gap > 40) parts.push(`counting from your ${shortDay(prev.statementDay)} bill of ${rupees(prev.total)} and the ${rupees(paid)} you've paid since`)
  else if (prev) parts.push(carry > BILL_TOLERANCE ? `${rupees(carry)} left from the last bill` : carry < -BILL_TOLERANCE ? `${rupees(-carry)} paid in advance` : "last bill paid in full")
  else parts.push("assuming the last bill was paid in full")
  if (fxSpends.length) {
    const byCur = {}
    for (const a of fxSpends) { const [cur, amt] = String(a.fx).split(/\s+/); byCur[cur] = (byCur[cur] || 0) + Number(amt || 0) }
    const list = Object.entries(byCur).map(([c, v]) => `${c} ${v.toFixed(2)}`).join(" + ")
    parts.push(`${fxSpends.length} foreign-currency spend${fxSpends.length === 1 ? "" : "s"} (${list}) estimated at about ${rupees(Math.round(fxTotal))} including forex markup`)
  }
  let movedNote = best.moved.map(x => `${x.a.merchant || "a charge"} ${rupees(x.a.amount)} (${shortDay(x.a.day)}) ${x.out ? "moves to the next bill" : "is on this bill"}`).join("; ")
  if (movedNote) movedNote = " " + movedNote.replace(/^./, c => c.toUpperCase()) + "."

  let result, details
  if (Math.abs(diff) <= tol) {
    result = "Matches"
    details = (fxSpends.length && Math.abs(diff) > BILL_TOLERANCE
      ? `Matches your alerts within the exchange-rate estimate (${diff > 0 ? "+" : "−"}${rupees(Math.abs(diff))}): `
      : "Matches your alerts: ") + `${parts.join(", ")}.` + movedNote
  } else if (diff > 0) {
    result = "Extra charges"
    const guesses = []
    const base18 = diff / 1.18
    if (Math.abs(base18 - Math.round(base18 / 50) * 50) < 1 && base18 >= 50) guesses.push(`${rupees(diff)} is ${rupees(Math.round(base18 / 50) * 50)} + 18% GST, which looks like a fee`)
    if (carry > BILL_TOLERANCE) guesses.push(`the last bill wasn't fully paid, so interest or a late fee is likely`)
    if (fxSpends.length) guesses.push(`that's beyond the 1.5% allowed for exchange-rate differences on your foreign spends`)
    details = `${rupees(diff)} more than your alerts add up to (${rupees(expected)}: ${parts.join(", ")}). ` +
      (guesses.length ? guesses.join("; ").replace(/^./, c => c.toUpperCase()) + ". " : "") +
      "Look in the statement for fees, interest, GST, forex markup or a charge you didn't get an alert for." + movedNote
  } else {
    result = "Lower than expected"
    details = `${rupees(-diff)} less than your alerts add up to (${rupees(expected)}: ${parts.join(", ")}). Usually a refund, cashback or payment that came without an alert.` + movedNote
  }
  return Object.assign(out, { expected, difference: diff, result, details, moved: best.moved })
}


// ---------- card bills ----------
// When a bill arrives (Scapia's statement email, CRED's "bill generated" email, or a forwarded bank SMS),
// it's saved to the Notion table "Card Bills" with what your alerts say it should be. If the bill is higher,
// you get an email so you can check the statement for fees, interest or a charge you had no alert for.

let BILLS_DB = ""      // Card Bills: found by name
const NOTIFY_EXTRA = true                              // email you when a bill has extra charges
const BILL_QUERY = days => `in:anywhere newer_than:${days}d (` + [
  "(from:scapiacards@federalbank.co.in subject:statement)",
  '(from:cred.club "bill summary")',
  '(subject:"[CC-SMS]" (statement OR "amt due" OR "amount due"))',
].join(" OR ") + ")"

function billRows(token) {
  return queryAll(token, BILLS_DB, { sorts: [{ property: "Statement date", direction: "ascending" }] }).map(pg => {
    const p = pg.properties, d = k => (p[k] && p[k].date ? p[k].date.start.slice(0, 10) : null)
    return { id: pg.id, card: sel(p.Card), statementDay: d("Statement date"), cycleFrom: d("Cycle from"), total: (p["Total due"] || {}).number || 0,
      messageId: plain(p["Message ID"]), result: sel(p.Result) }
  })
}

function syncBills(token) {
  let saved
  try { saved = billRows(token) } catch (e) {
    if (/can't see|object_not_found|Could not find/i.test(String(e.message))) throw new Error("connect the Card Bills table to your Notion integration")
    throw e
  }
  const seen = new Set(saved.map(b => b.messageId))
  const found = []
  for (const t of GmailApp.search(BILL_QUERY(saved.length ? 10 : 120), 0, 60)) for (const m of t.getMessages()) {
    if (seen.has(m.getId())) continue
    const b = parseBill({ subject: m.getSubject(), from: m.getFrom(), body: m.getPlainBody(), date: m.getDate() }, cardOf)
    if (b) { b.messageId = m.getId(); found.push(b) }
  }
  found.sort((a, b) => a.statementDay < b.statementDay ? -1 : 1)
  let added = 0, extra = 0
  for (const b of found) {
    // Same bill from two sources (Scapia's email and CRED's): keep the first.
    if (saved.some(s => s.card === b.card && Math.abs(dayNum(s.statementDay) - dayNum(b.statementDay)) <= 4)) continue
    const prev = saved.filter(s => s.card === b.card && s.statementDay < b.statementDay).pop() || null
    const from = shiftDay(b.cycleFrom || shiftDay(b.statementDay, -40), -4)
    const alerts = queryAll(token, ALERTS_DB, { filter: { and: [
      { property: "Card", select: { equals: b.card } },
      { property: "Date", date: { on_or_after: prev ? shiftDay(prev.statementDay, -1) : from } },
      { property: "Date", date: { on_or_before: shiftDay(b.statementDay, 2) } },
    ] } }).map(pg => {
      const p = pg.properties, start = p.Date.date ? p.Date.date.start : pg.created_time
      return { day: istDay(Date.parse(start.length === 10 ? start + "T12:00:00+05:30" : start)), amount: p.Amount.number || 0, type: sel(p.Type) || "Spend", merchant: plain(p.Merchant), card: sel(p.Card), fx: isForeign(plain(p.Currency)) ? plain(p.Currency) : null }
    })
    const r = checkBill(b, prev && { statementDay: prev.statementDay, cycleTo: null, total: prev.total }, alerts, alertsStartDay(token, b.card))
    const fresh = dayNum(istDay(Date.now())) - dayNum(b.statementDay) <= 10
    const notify = NOTIFY_EXTRA && fresh && r.result === "Extra charges"
    const short = CARD_SHORT[b.card] || b.card
    const dt = v => v ? { date: { start: v } } : { date: null }
    const page = notion(token, "post", "pages", { parent: { database_id: BILLS_DB }, properties: {
      Bill: { title: [{ text: { content: `${short} · ${b.statementDay}` } }] },
      Card: { select: { name: b.card } }, "Statement date": dt(b.statementDay), "Cycle from": dt(r.cycleFrom), "Due date": dt(b.dueDay),
      "Total due": { number: b.total }, "Minimum due": { number: b.minDue || null },
      Expected: { number: r.expected }, Difference: { number: r.difference }, Result: { select: { name: r.result } },
      Details: { rich_text: [{ text: { content: r.details.slice(0, 1900) } }] }, "Message ID": { rich_text: [{ text: { content: b.messageId } }] },
      Notified: { checkbox: notify },
    } })
    if (notify) {
      try {
        MailApp.sendEmail(Session.getEffectiveUser().getEmail(), `${short} bill: ${rupees(r.difference)} more than your card alerts`,
          `Your ${short} bill dated ${shortDay(b.statementDay)} is ${rupees(b.total)}${b.dueDay ? ", due " + shortDay(b.dueDay) : ""}.\n\n${r.details}\n\nOpen in Notion: ${page.url}\n\n(Sent by your card-alerts Google Apps Script.)`)
      } catch (e) { Logger.log("Couldn't send the bill email: " + e) }
    }
    saved.push({ card: b.card, statementDay: b.statementDay, total: b.total, messageId: b.messageId })
    added++; if (r.result === "Extra charges") extra++
  }
  return { added, extra }
}

// The first day you have an alert for this card (bills from before that can't be checked).
function alertsStartDay(token, card) {
  const res = notion(token, "post", `databases/${ALERTS_DB}/query`, { page_size: 1, sorts: [{ property: "Date", direction: "ascending" }],
    filter: { and: [{ property: "Card", select: { equals: card } }, { property: "Date", date: { is_not_empty: true } }] } })
  const d = res.results && res.results[0] && res.results[0].properties.Date.date
  return d ? d.start.slice(0, 10) : "9999-12-31"
}

// ---------- loans & EMIs ----------
// Matches EMI debits from your bank alerts to the loans in your Notion "Loans" table, works out whether this
// month's EMI is paid, due or late, and (once you add the rate and outstanding) the loan's progress:
// outstanding now, months left, interest vs principal, totals for the financial year (April–March),
// and what each prepayment saved. Interest is worked out monthly on the reducing balance, as Indian home
// loans usually are, so figures are close to your lender's statement but not to the rupee.

// Helpers are kept inside so their names can't clash with the rest of the script.
const { loanPayments, emiStatus, emiChange, loanProgress, suggestLoans, finYear, matchesLoan, learned, parseRates, formatRates } = (() => {
const DAY = 864e5
const dnum = iso => Math.round(Date.parse(String(iso).slice(0, 10) + "T00:00:00Z") / DAY)
const monthKeyOf = iso => String(iso).slice(0, 7)
const daysInMonth = (y, m) => new Date(Date.UTC(y, m, 0)).getUTCDate()   // m is 1–12
function dueDateIn(key, day) { const [y, m] = key.split("-").map(Number); return `${key}-${String(Math.min(day, daysInMonth(y, m))).padStart(2, "0")}` }
const shiftDays = (iso, n) => new Date(Date.parse(String(iso).slice(0, 10) + "T00:00:00Z") + n * DAY).toISOString().slice(0, 10)
function addMonths(key, n) { const [y, m] = key.split("-").map(Number); const t = new Date(Date.UTC(y, m - 1 + n, 1)); return t.toISOString().slice(0, 7) }
// Indian financial year containing a day: 2026-10-04 → { start: "2026-04-01", end: "2027-03-31", label: "FY 2026–27" }
function finYear(iso) {
  const [y, m] = String(iso).slice(0, 7).split("-").map(Number), s = m >= 4 ? y : y - 1
  return { start: `${s}-04-01`, end: `${s + 1}-03-31`, label: `FY ${s}–${String(s + 1).slice(2)}` }
}
const wordsOf = s => String(s || "").toLowerCase().replace(/[^a-z0-9 ]+/g, " ").split(/\s+/).filter(Boolean)
const matchesLoan = (loan, merchant) => { const w = wordsOf(loan.match || loan.name); const m = " " + wordsOf(merchant).join(" ") + " "; return w.length > 0 && w.every(x => m.includes(" " + x)) }

// alerts: [{ id, day, amount, type, merchant, channel }]. Returns { emis, prepayments, charges } (each [{ id, day, amount }]).
function loanPayments(loan, alerts, others = []) {
  const byAmount = a => loan.emi > 0 && loan.day && /auto-debit|emi/i.test(a.channel || "") && Math.abs(a.amount - loan.emi) <= loan.emi * 0.01 &&
    Math.abs(Number(String(a.day).slice(8, 10)) - loan.day) <= 4 && !others.some(o => o !== loan && matchesLoan(o, a.merchant))
  // two loans with the same lender (a home loan and its top-up): each payment goes to the loan whose EMI is nearest
  const nearest = a => {
    const same = [loan].concat(others.filter(o => o !== loan && matchesLoan(o, a.merchant) && o.emi > 0))
    if (same.length < 2 || !(loan.emi > 0)) return same.length < 2 || !same.some(o => o !== loan && o.emi > 0)
    const gap = o => Math.abs(Math.log(a.amount / o.emi))
    return same.every(o => gap(loan) <= gap(o))
  }
  // a lender receipt names the account ("… EMI · A/c …8406"): it belongs to that loan only
  const tagOf = a => { const m = String(a.merchant || "").match(/A\/c …(\w{4})\b/); return m ? m[1].toUpperCase() : "" }
  const acctIn = (l, a) => { const n = String(l.account || "").toUpperCase(); return n.length >= 6 && String(a.merchant || "").toUpperCase().includes(n) }
  const mineByAcct = a => { if (acctIn(loan, a)) return true; if (others.some(o => o !== loan && acctIn(o, a))) return false
    const t = tagOf(a); if (!t) return null; const own = String(loan.account || "").toUpperCase(); return own ? own.endsWith(t) : (others.some(o => o !== loan && String(o.account || "").toUpperCase().endsWith(t)) ? false : null) }
  // insurance, broking, cards and the like share a bank's name ("HDFC ERGO") but aren't loan payments
  const otherBusiness = a => /\bergo\b|insurance|securities|\bsec\b|mutual fund|\blife\b|credit card|card bill|forex|\bcredila\b/i.test(a.merchant || "")
  const beforeStart = a => loan.start && dnum(a.day) < dnum(loan.start) - 45
  const ours = a => (a.type || "Spend") === "Spend" && !beforeStart(a) && (mineByAcct(a) != null ? mineByAcct(a) : !otherBusiness(a) && ((matchesLoan(loan, a.merchant) && nearest(a)) || byAmount(a)))
  const mine = alerts.filter(a => ours(a) && (!loan.closed || a.day <= loan.closed)).sort((a, b) => a.day < b.day ? -1 : 1)
  // an EMI debited in the days after the loan closed (often refunded later)
  const afterClose = loan.closed ? alerts.filter(a => ours(a) && a.day > loan.closed && dnum(a.day) - dnum(loan.closed) <= 15).map(a => ({ id: a.id, day: a.day, amount: a.amount })) : []
  const sharedLender = loan.emi > 0 && others.some(o => o !== loan && o.emi > 0 && wordsOf(o.match || o.name).join(" ") === wordsOf(loan.match || loan.name).join(" "))
  // the same debit saved twice (an SMS and a late bank email): keep one
  const uniq = []
  for (const a of mine) if (!uniq.some(u => String(u.day).slice(0, 10) === String(a.day).slice(0, 10) && Math.abs(u.amount - a.amount) < 1)) uniq.push(a)
  mine.length = 0; mine.push(...uniq)
  const byMonth = {}
  for (const a of mine) (byMonth[monthKeyOf(a.day)] = byMonth[monthKeyOf(a.day)] || []).push(a)
  const emis = [], prepayments = [], charges = []
  // a new loan with no EMI yet learns it from its first auto-debit (₹1,000 or more); payments before that are fees
  const emiNow = () => (emis.length ? emis[emis.length - 1].amount : loan.emi) || 0
  for (const key of Object.keys(byMonth).sort()) {
    const list = byMonth[key], target = emiNow()
    // the EMI is the auto-debit closest to the EMI amount (within 15%), or any payment within 3% of it
    const near = (a, tol) => target && Math.abs(a.amount - target) <= target * tol
    const auto = a => /auto-debit|emi/i.test(a.channel || "")
    const pick = !target ? list.filter(a => auto(a) && a.amount >= 1000)[0]
      : list.filter(a => auto(a) && near(a, 0.15)).sort((x, y) => Math.abs(x.amount - target) - Math.abs(y.amount - target))[0]
        || list.filter(a => near(a, 0.03)).sort((x, y) => Math.abs(x.amount - target) - Math.abs(y.amount - target))[0]
        // a lender with two loans (home loan + top-up): the month's one auto-debit nearest this loan's EMI is its EMI, even after the EMI changed
        || (sharedLender && list.filter(a => auto(a) && a.amount < target * 5).length === 1 ? list.filter(a => auto(a) && a.amount < target * 5)[0] : null)
    for (const a of list) {
      const p = { id: a.id, day: a.day, amount: a.amount }
      if (a === pick) emis.push(p)
      else if (target && a.amount >= target * 0.5) prepayments.push(p)
      else charges.push(p)
    }
  }
  return { emis, prepayments, charges, afterClose }
}

// This month's EMI: paid, due soon, due today, expected (alert may be slow) or late.
function emiStatus(loan, pay, today) {
  if (loan.closed) return { state: "closed", closed: loan.closed, lastDay: (pay.emis[pay.emis.length - 1] || {}).day || "" }
  loan = learned(loan, pay)
  if (!loan.day || !loan.emi) return { state: "waiting", amount: loan.emi || 0 }
  const key = monthKeyOf(today), due = dueDateIn(key, loan.day || 1)
  const paid = pay.emis.find(e => monthKeyOf(e.day) === key)
  const nextDue = dueDateIn(addMonths(key, 1), loan.day || 1)
  const latest = pay.emis[pay.emis.length - 1]
  const amount = latest ? latest.amount : loan.emi
  if (paid) return { state: "paid", day: paid.day, amount: paid.amount, due, nextDue, daysTo: dnum(nextDue) - dnum(today) }
  const d = dnum(due) - dnum(today)
  if (d > 0) return { state: d <= 3 ? "soon" : "upcoming", due, amount, daysTo: d }
  if (d === 0) return { state: "today", due, amount, daysTo: 0 }
  if (d >= -1) return { state: "expected", due, amount, daysLate: -d }
  return { state: "late", due, amount, daysLate: -d }
}

// EMI and debit day from the first EMI seen, for a loan added before its first debit
function learned(loan, pay) {
  const first = pay.emis[0]
  return first && (!loan.emi || !loan.day) ? Object.assign({}, loan, { emi: loan.emi || first.amount, day: loan.day || Number(first.day.slice(8, 10)), start: loan.start || first.day, learnedFrom: first.day }) : loan
}
// The EMI amount changed (usually a rate change): { from, to, day } or null.
function emiChange(loan, pay) {
  if (loan.closed || !loan.emi) return null
  const last = pay.emis[pay.emis.length - 1]
  return last && loan.emi && Math.abs(last.amount - loan.emi) > 1 ? { from: loan.emi, to: last.amount, day: last.day } : null
}

// Rate history: "8.4% from 2026-04-01; 8.75% from 2026-11-01" ↔ [{ from, rate }], oldest first
function parseRates(text, current) {
  const out = []
  for (const m of String(text || "").matchAll(/(\d{1,2}(?:\.\d{1,3})?)\s*%?\s*(?:from|since)\s*(\d{4}-\d{2}-\d{2})/gi)) out.push({ rate: Number(m[1]), from: m[2] })
  out.sort((a, b) => a.from < b.from ? -1 : 1)
  if (!out.length && current > 0) out.push({ rate: current, from: "" })
  return out
}
const formatRates = list => list.filter(r => r.rate > 0).map(r => `${r.rate}% from ${r.from || "start"}`).join("; ").replace(/from start/g, "from 2000-01-01")
// Rate for an EMI on this day: the rate in force at the middle of the month it pays for
function rateOn(rates, fallback, day) {
  if (!rates || !rates.length) return fallback
  const mid = new Date(Date.parse(String(day).slice(0, 10) + "T00:00:00Z") - 15 * DAY).toISOString().slice(0, 10)
  let r = rates[0].rate
  for (const x of rates) if (!x.from || x.from <= mid) r = x.rate
  return r
}
// The monthly rate at which E repays O in n months (EMI formula, solved by halving)
function impliedRate(O, n, E) {
  if (!(O > 0) || !(n > 0) || !(E > O / n)) return null
  const emi = i => O * i * Math.pow(1 + i, n) / (Math.pow(1 + i, n) - 1)
  let lo = 1e-6, hi = 0.05
  for (let k = 0; k < 80; k++) { const mid = (lo + hi) / 2; if (emi(mid) > E) hi = mid; else lo = mid }
  return (lo + hi) / 2 * 1200
}
const monthsToRepay = (O, i, E) => { if (O <= 0) return 0; if (!(E > O * i)) return Infinity; return -Math.log(1 - O * i / E) / Math.log(1 + i) }

// Needs loan.rate (% a year), loan.outstanding and loan.asOf (outstanding right after the EMI on that day).
// EMIs from before your first EMI alert are assumed paid on the debit day.
function loanProgress(loan, pay, today) {
  loan = learned(loan, pay)
  const rates = loan.rates && loan.rates.length ? loan.rates : parseRates("", loan.rate)
  const current = rates.length ? rates[rates.length - 1].rate : loan.rate
  // no outstanding entered yet: start from the amount disbursed, just before the first EMI (a new loan)
  let fromDisbursed = false
  // (also when the outstanding typed in is just the amount disbursed: no EMI has cut it yet, whatever "As of" says)
  const sameAsDisbursed = loan.outstanding > 0 && loan.disbursed > 0 && Math.abs(loan.outstanding - loan.disbursed) < 1
  if ((!(loan.outstanding > 0) || sameAsDisbursed) && loan.disbursed > 0 && (loan.start || pay.emis.length)) {
    const first = (pay.emis[0] || {}).day || loan.start
    loan = Object.assign({}, loan, { outstanding: loan.disbursed, asOf: shiftDays(first < (loan.start || first) ? first : (loan.start || first), -1) }); fromDisbursed = true
  }
  if (!(current > 0) || !(loan.outstanding > 0) || !loan.asOf || !loan.emi) return null
  const closed = loan.closed ? String(loan.closed).slice(0, 10) : null
  if (closed && closed < today) today = closed   // a closed loan stops on its closing day
  const i = current / 1200, fy = finYear(today), asOf = String(loan.asOf).slice(0, 10)
  const iOn = day => rateOn(rates, current, day) / 1200
  const emiFor = key => (pay.emis.find(e => monthKeyOf(e.day) === key) || null)
  const firstKey = monthKeyOf(fy.start < asOf ? fy.start : asOf), lastKey = monthKeyOf(today)
  // months before your first EMI alert are assumed paid; after that, a month with no EMI alert counts as not paid yet
  const firstSeen = pay.emis.length ? monthKeyOf(pay.emis[0].day) : null
  // every EMI from the earlier of FY start / As of, up to today: actual alert, or assumed if its date has passed
  const events = []
  for (let k = firstKey; k <= lastKey; k = addMonths(k, 1)) {
    const e = emiFor(k), due = dueDateIn(k, loan.day || 1)
    if (closed && due > closed && !e) continue
    if (loan.start && (k < monthKeyOf(loan.start) || (due < loan.start && !e))) continue   // before this loan began (e.g. a balance transfer)
    if (fromDisbursed && !e && (!firstSeen || k < firstSeen)) continue                   // a new loan: nothing paid before its first EMI
    if (e) events.push({ kind: "emi", day: e.day, amount: e.amount })
    else if (due <= today && (!firstSeen || k < firstSeen)) events.push({ kind: "emi", day: due, amount: (pay.emis.filter(x => x.day <= due).pop() || {}).amount || loan.emi, assumed: true })
  }
  for (const p of pay.prepayments) if (p.day >= `${firstKey}-01` && p.day <= today) events.push({ kind: "prepay", day: p.day, amount: p.amount })
  events.sort((a, b) => a.day < b.day ? -1 : a.day > b.day ? 1 : a.kind === "emi" ? -1 : 1)
  // walk back from As of to the start, then forward to today
  let O = loan.outstanding
  const daily = loan.basis !== "monthly"   // most lenders charge interest on the actual days between EMIs
  const before = events.filter(e => e.day <= asOf).reverse(), after = events.filter(e => e.day > asOf)
  if (!daily) {
    for (const e of before) { e.after = O; O = e.kind === "emi" ? (O + e.amount) / (1 + iOn(e.day)) : O + e.amount; e.before = O }
    O = loan.outstanding
    for (const e of after) { e.before = O; O = e.kind === "emi" ? O * (1 + iOn(e.day)) - e.amount : Math.max(0, O - e.amount); e.after = O }
    for (const e of events) if (e.kind === "emi") { e.interest = e.before * iOn(e.day); e.principal = e.amount - e.interest }
  } else {
    // actual days: interest builds up daily on the balance (rate ÷ 365 a day) and is paid by the next EMI;
    // a prepayment cuts the balance from its own day
    const d = (a, b) => Math.max(0, dnum(b) - dnum(a)), r = day => iOn(day) * 12 / 365
    const emisSorted = events.filter(e => e.kind === "emi")
    const prevEmiDay = e => { const k = emisSorted.indexOf(e); return k > 0 ? emisSorted[k - 1].day : dueDateIn(addMonths(monthKeyOf(e.day), -1), Number(String(e.day).slice(8, 10)) || loan.day || 1) }
    // backwards: each EMI interval solved for the balance right after the previous EMI
    let k = 0
    while (k < before.length) {
      const e = before[k]
      if (e.kind !== "emi") { e.after = O; O = O + e.amount; e.before = O; k++; continue }   // a prepayment after the last EMI before As of
      const start = prevEmiDay(e), rate = r(e.day)
      const pre = []; let j = k + 1
      while (j < before.length && before[j].kind !== "emi" && before[j].day > start) pre.push(before[j++])
      const D = d(start, e.day), Psum = pre.reduce((t, p) => t + p.amount * (1 + rate * d(p.day, e.day)), 0)
      const B0 = (O + e.amount + Psum) / (1 + rate * D)
      e.after = O
      let b = B0, last = start, acc = 0
      for (const p of pre.slice().reverse()) { acc += b * rate * d(last, p.day); p.before = b; b -= p.amount; p.after = b; last = p.day }
      acc += b * rate * d(last, e.day)
      e.before = b; e.interest = acc; e.principal = e.amount - acc
      O = B0; k = j
    }
    // forwards from As of
    O = loan.outstanding
    let last = before.find(e => e.kind === "emi") ? before.find(e => e.kind === "emi").day : asOf, acc = 0
    if (last < asOf) last = asOf
    // a new loan: interest before the first EMI is paid separately (pre-EMI interest), so the first EMI covers one month
    if (fromDisbursed && after.length) { const f = after.find(e => e.kind === "emi"); if (f) { const m1 = dueDateIn(addMonths(monthKeyOf(f.day), -1), Number(f.day.slice(8, 10))); if (m1 > last) last = m1 } }
    for (const e of after) {
      acc += O * r(e.day) * d(last, e.day); last = e.day
      if (e.kind === "emi") { e.before = O; e.interest = acc; e.principal = e.amount - acc; O = O - e.principal; acc = 0; e.after = O }
      else { e.before = O; O = Math.max(0, O - e.amount); e.after = O }
    }
  }
  const outstanding = Math.max(0, O)
  const E = (pay.emis[pay.emis.length - 1] || {}).amount || loan.emi
  const n = monthsToRepay(outstanding, i, E)
  const inFy = events.filter(e => e.kind === "emi" && e.day >= fy.start && e.day <= today)
  const fyInterest = inFy.reduce((s, e) => s + e.interest, 0), fyPrincipal = inFy.reduce((s, e) => s + e.principal, 0)
  // rest of the financial year at the current EMI
  let restI = 0, restP = 0, o = outstanding
  const projected = []
  for (let k = addMonths(lastKey, 0); k <= monthKeyOf(fy.end); k = addMonths(k, 1)) {
    const due = dueDateIn(k, loan.day || 1)
    if ((due <= today && !(k === lastKey && !emiFor(k))) || o <= 0) continue   // still to pay, including a late one this month
    const days = Math.max(1, dnum(due) - dnum(dueDateIn(addMonths(k, -1), loan.day || 1)))
    const int = daily ? o * i * 12 / 365 * days : o * i, pr = Math.min(o, E - int); restI += int; restP += pr; o = o + int - E
    projected.push({ key: k, day: due, interest: Math.round(int), principal: Math.round(pr), est: true })
  }
  const fyPrepaid = events.filter(e => e.kind === "prepay" && e.day >= fy.start && e.day <= today).reduce((s, e) => s + e.amount, 0)
  // what each prepayment in the last 12 months saved
  const saved = events.filter(e => e.kind === "prepay" && dnum(today) - dnum(e.day) <= 366).map(e => {
    const without = monthsToRepay(e.before, i, E), withIt = monthsToRepay(e.after, i, E)
    const months = without - withIt
    return { day: e.day, amount: e.amount, months: Math.round(months), interest: Math.max(0, Math.round(months * E - e.amount)) }
  })
  const last = events.filter(e => e.kind === "emi").pop()
  // the next EMI still to come is this month's if its date hasn't passed, otherwise next month's
  const nextKey = emiFor(lastKey) || dueDateIn(lastKey, loan.day || 1) <= today ? addMonths(lastKey, 1) : lastKey
  const payoffKey = isFinite(n) && n > 0 ? addMonths(nextKey, Math.ceil(n) - 1) : null
  const fyCharges = (pay.charges || []).filter(c => c.day >= fy.start && c.day <= today).reduce((s, c) => s + c.amount, 0)
  // EMI changes with no rate change on record: if the lender kept the term and changed the EMI (as Bajaj does),
  // the new EMI tells us the new rate
  const rateHints = []
  const seen = events.filter(e => e.kind === "emi" && !e.assumed)
  for (let k = 1; k < seen.length; k++) {
    const a = seen[k - 1], b = seen[k]
    if (Math.abs(b.amount - a.amount) <= a.amount * 0.005) continue
    const near = d => rates.some(r => r.from && Math.abs(dnum(r.from) - dnum(d)) <= 45)
    if (near(b.day)) continue
    const prepaid = events.some(e => e.kind === "prepay" && e.day <= b.day && dnum(b.day) - dnum(e.day) <= 75)
    const iOld = iOn(a.day), n = monthsToRepay(a.after, iOld, a.amount)
    const r = prepaid || !isFinite(n) ? null : impliedRate(a.after, Math.round(n), b.amount)
    rateHints.push({ day: b.day, from: a.amount, to: b.amount, oldRate: Math.round(iOld * 120000) / 100, rate: r ? Math.round(r * 100) / 100 : null, prepaid })
  }
  const schedule = inFy.map(e => ({ key: monthKeyOf(e.day), day: e.day, interest: Math.round(e.interest), principal: Math.round(e.principal), assumed: !!e.assumed }))
  // tenure: the term you signed for (months, from the first EMI) against when the loan now ends
  let tenure = null
  if (loan.tenure > 0 && (loan.start || pay.emis.length)) {
    const firstKey2 = monthKeyOf((pay.emis[0] || {}).day && (pay.emis[0].day > (loan.start || "") ? pay.emis[0].day : loan.start) || loan.start)
    const plannedEnd = addMonths(firstKey2, loan.tenure - 1)
    const diff = payoffKey ? (Number(payoffKey.slice(0, 4)) - Number(plannedEnd.slice(0, 4))) * 12 + Number(payoffKey.slice(5)) - Number(plannedEnd.slice(5)) : null
    tenure = { months: loan.tenure, plannedEnd, diff }
  }
  const rateInfo = { current, history: rates.filter(r => r.rate > 0) }
  if (closed) return { closed, schedule, rateInfo, fromDisbursed, rateHints, payoff: outstanding > 1 ? { day: closed, principal: Math.round(outstanding) } : null, outstanding: Math.round(outstanding), lastEmi: last ? { day: last.day, interest: Math.round(last.interest), principal: Math.round(last.principal), assumed: !!last.assumed } : null,
    fy: { label: fy.label, interest: Math.round(fyInterest), principal: Math.round(fyPrincipal), prepaid: Math.round(fyPrepaid), interestYear: Math.round(fyInterest), principalYear: Math.round(fyPrincipal), assumedMonths: inFy.filter(e => e.assumed).length, charges: Math.round(fyCharges) }, saved }
  return {
    outstanding: Math.round(outstanding), monthsLeft: isFinite(n) ? Math.ceil(n) : null, payoff: payoffKey,
    interestLeft: isFinite(n) ? Math.max(0, Math.round(n * E - outstanding)) : null,
    lastEmi: last ? { day: last.day, interest: Math.round(last.interest), principal: Math.round(last.principal), assumed: !!last.assumed } : null,
    fy: { label: fy.label, interest: Math.round(fyInterest), principal: Math.round(fyPrincipal), prepaid: Math.round(fyPrepaid),
      interestYear: Math.round(fyInterest + restI), principalYear: Math.round(fyPrincipal + restP), assumedMonths: inFy.filter(e => e.assumed).length, charges: Math.round(fyCharges) },
    saved, schedule: schedule.concat(projected), rateInfo, tenure, fromDisbursed, rateHints,
  }
}

// Recurring loan-like debits that aren't tracked yet: [{ name, match, emi, day, months }]
// Lenders and words that mark a debit as a loan repayment
const LOAN_WORDS = /\bloan|housing|home fin|\bemi\b|\bnach\b|\bach\b|\becs\b|bajaj fin|bajaj housing|lic housing|pnb housing|tata capital|aditya birla|idfc first|hdfc ltd|hdfc limited|hdfc bank home|icici home|sbi home|axis finance|kotak mahindra prime|l ?& ?t finance|cholamandalam|shriram|muthoot|mahindra finance|home credit|piramal|aavas|indiabulls|poonawalla|hero fincorp|fullerton|smfg|navi|moneyview|kreditbee|cred ?flash|\bpaysense/i
// Regular debits that are not loans: subscriptions, investments, insurance, utilities
const NOT_LOAN = /\bsip\b|mutual|\bmf\b|zerodha|groww|kuvera|\bcoin\b|\bnps\b|\bppf\b|insurance|\blic\b|policy|apple|google|netflix|spotify|youtube|prime|hotstar|jio|airtel|vodafone|electricity|power|gas\b|broadband|\brent\b|society|maintenance|gym|cult/i

// Debits that look like an EMI but aren't tracked yet. Two kinds:
// - lender-like debits (loan words or a lender's name) seen in 2+ months at about the same amount;
// - any other auto-debit of ₹1,000+ seen in 3+ months at the same amount (within 1%) around the same day.
// ignored: payee keys you said aren't loans.
function suggestLoans(alerts, loans, ignored = []) {
  const groups = {}
  for (const a of alerts) {
    if ((a.type || "Spend") !== "Spend" || !(a.amount >= 100)) continue
    const lenderish = a.category === "Loan EMI" || LOAN_WORDS.test(a.merchant)
    const auto = /auto-debit|other bank debit/i.test(a.channel || "")
    if (!lenderish && !auto) continue
    if (NOT_LOAN.test(a.merchant) || /Subscriptions|LIC|Health insurance|Bills & Utilities/.test(a.category || "")) continue
    // a debit to a lender you already track, at an amount that isn't any of its loans' EMIs, may be a second loan (a top-up)
    const sameLender = loans.filter(l => matchesLoan(l, a.merchant))
    const nearAnEmi = l => l.emi > 0 && Math.abs(a.amount - l.emi) <= l.emi * 0.15
    if (sameLender.length && (sameLender.some(nearAnEmi) || !/auto-debit|other bank debit/i.test(a.channel || "") || a.amount >= Math.max(...sameLender.map(l => l.emi || 0)) * 0.5)) continue
    if (!sameLender.length && loans.some(l => l.emi > 0 && Math.abs(a.amount - l.emi) <= l.emi * 0.01 && (!l.day || Math.abs(Number(String(a.day).slice(8, 10)) - l.day) <= 4))) continue
    const second = sameLender.length > 0
    const w = wordsOf(a.merchant).map(x => x.replace(/\d+$/, "")).filter(x => x.length > 1 && !/^(emi|ltd|limited|pvt|private|billdesk|nach|ach|ecs|payment|dr|to|tp|si|auto|debit|\d+)$/.test(x)).slice(0, 2)
    if (!w.length) continue
    const key = w.join(" ")
    if (ignored.includes(key)) continue
    ;(groups[key] = groups[key] || { list: [], lenderish: false, second: false }).list.push(a)
    if (lenderish) groups[key].lenderish = true
    if (second) { groups[key].second = true; groups[key].lenderName = String(sameLender[0].name || "").split("·").slice(1).join("·").replace(/\(A\/c[^)]*\)/, "").trim() }
  }
  const out = []
  for (const [match, g] of Object.entries(groups)) {
    const byAmt = {}
    for (const a of g.list) { const k = Object.keys(byAmt).find(x => Math.abs(Number(x) - a.amount) <= a.amount * (g.lenderish ? 0.03 : 0.01)); (byAmt[k || a.amount] = byAmt[k || a.amount] || []).push(a) }
    for (const [amt, same] of Object.entries(byAmt)) {
      const months = new Set(same.map(a => monthKeyOf(a.day)))
      const days = same.map(a => Number(a.day.slice(8, 10))).sort((x, y) => x - y)
      const mid = days[Math.floor(days.length / 2)]
      if (g.lenderish ? months.size < 2 : (months.size < 3 || Number(amt) < 1000 || days.some(d => Math.abs(d - mid) > 4 && Math.abs(d - mid) < 25))) continue
      out.push({ name: (g.second && g.lenderName) || match.replace(/\b\w/g, c => c.toUpperCase()), match, emi: Math.round(Number(amt)), day: mid, months: months.size, sure: g.lenderish, second: g.second })
    }
  }
  return out.sort((a, b) => b.sure - a.sure || b.emi - a.emi)
}

return { loanPayments, emiStatus, emiChange, loanProgress, suggestLoans, finYear, matchesLoan, learned, parseRates, formatRates }
})()


// ---------- loans: reminder before the EMI and an email if it's missed ----------
let LOANS_DB = ""      // Loans: found by name
const REMIND_DAYS_BEFORE = 2   // email this many days before the debit date

function syncLoans(token) {
  const props = PropertiesService.getScriptProperties(), now = Date.now()
  const hourIst = new Date(now + 330 * 6e4).getUTCHours()
  if (hourIst < 8 || hourIst >= 21) return { sent: 0 }                                // no emails at night
  if (now - Number(props.getProperty("LOANS_CHECKED_AT") || 0) < 3 * 36e5) return { sent: 0 }   // every 3 hours is plenty
  props.setProperty("LOANS_CHECKED_AT", String(now))
  if (!LOANS_DB) return { sent: 0 }
  const today = istDay(now)
  const date = p => (p && p.date ? p.date.start.slice(0, 10) : "")
  const loans = queryAll(token, LOANS_DB, {}).map(pg => { const p = pg.properties
    return { id: pg.id, url: pg.url, name: plain(p.Loan) || "Loan", match: plain(p["Lender match"]) || plain(p.Loan), emi: (p.EMI || {}).number || 0,
      day: (p["Debit day"] || {}).number || 0, start: date(p.Start), active: !!(p.Active && p.Active.checkbox), reminded: date(p["Reminder sent"]), lateSent: date(p["Late alert sent"]) } }).filter(l => l.active)
  if (!loans.length) return { sent: 0 }
  const alerts = queryAll(token, ALERTS_DB, { filter: { and: [{ property: "Date", date: { on_or_after: shiftDay(today, -100) } },
    { property: "Status", select: { does_not_equal: "Duplicate" } },
    { or: [{ property: "Category", select: { equals: "Loan EMI" } }, { property: "Channel", select: { equals: "EMI / auto-debit" } }, { property: "Channel", select: { equals: "Other bank debit" } }] }] } }).map(pg => {
    const p = pg.properties, start = p.Date.date ? p.Date.date.start : pg.created_time
    return { id: pg.id, day: istDay(Date.parse(start.length === 10 ? start + "T12:00:00+05:30" : start)), amount: p.Amount.number || 0, merchant: plain(p.Merchant), channel: sel(p.Channel), category: sel(p.Category), type: sel(p.Type) || "Spend" }
  })
  const to = Session.getEffectiveUser().getEmail(), fmt = iso => { const [y, m, d] = iso.split("-").map(Number); return `${d} ${["Jan","Feb","Mar","Apr","May","Jun","Jul","Aug","Sep","Oct","Nov","Dec"][m - 1]}` }
  let sent = 0
  for (const l of loans) {
    const pay = loanPayments(l, alerts, loans), s = emiStatus(l, pay, today)
    // a loan's EMI or part payment that came in as a plain bank debit ("HDFC Bank Ltd/… with UMRN …", category Other): file it under Loan EMI
    for (const e of pay.emis.concat(pay.prepayments)) { const a = alerts.find(x => x.id === e.id)
      if (a && a.category !== "Loan EMI") { notion(token, "patch", `pages/${a.id}`, { properties: { Category: { select: { name: "Loan EMI" } } } }); a.category = "Loan EMI" } }
    // a new loan (e.g. after a balance transfer) learns its EMI, debit day and start from the first EMI debit
    if ((!l.emi || !l.day || !l.start) && pay.emis.length) {
      const lv = learned(l, pay)
      notion(token, "patch", `pages/${l.id}`, { properties: { EMI: { number: lv.emi }, "Debit day": { number: lv.day }, Start: { date: { start: lv.start || pay.emis[0].day } } } })
    }
    if (s.state === "waiting" || s.state === "closed") continue
    const label = l.name.split("·")[0].trim() || "Loan"
    // the next debit: this month's, or next month's once this month is paid
    const next = s.state === "paid" ? { due: s.nextDue, daysTo: s.daysTo, amount: s.amount } : s
    if (s.state !== "late" && s.state !== "expected" && next.daysTo != null && next.daysTo <= REMIND_DAYS_BEFORE && next.daysTo >= 0 && l.reminded.slice(0, 7) !== next.due.slice(0, 7)) {
      MailApp.sendEmail(to, `${label} EMI of ${rupees(next.amount)} debits ${next.daysTo === 0 ? "today" : "on " + fmt(next.due)}`,
        `Your ${l.name} EMI of ${rupees(next.amount)} is due ${next.daysTo === 0 ? "today" : `on ${fmt(next.due)} (in ${next.daysTo} day${next.daysTo === 1 ? "" : "s"})`}.\n\nKeep enough balance in the account it's debited from; a bounced EMI usually brings a charge from the bank and the lender.\n\n(Sent by your card-alerts Google Apps Script.)`)
      notion(token, "patch", `pages/${l.id}`, { properties: { "Reminder sent": { date: { start: next.due } } } }); sent++
    }
    if (s.state === "late" && l.lateSent.slice(0, 7) !== s.due.slice(0, 7)) {
      MailApp.sendEmail(to, `${label} EMI for ${fmt(s.due)} not seen yet`,
        `No debit alert for your ${l.name} EMI of ${rupees(s.amount)}, due on ${fmt(s.due)} (${s.daysLate} days ago).\n\nCheck your bank app. If it was debited, the alert may simply be missing and you can ignore this. If it bounced, pay it soon to keep late charges and your credit score in check.\n\n(Sent by your card-alerts Google Apps Script.)`)
      notion(token, "patch", `pages/${l.id}`, { properties: { "Late alert sent": { date: { start: s.due } } } }); sent++
    }
  }
  return { sent }
}

// Lenders email you about each loan: "EMI due", "EMI received", sanction and disbursement, foreclosure.
// parseLoanMail() reads one email and returns what it says about one loan account, or null.
// Promotional mail ("loans starting at 7.15%") is ignored: an email must name a loan account to count.

const LM_MON = { jan: 0, feb: 1, mar: 2, apr: 3, may: 4, jun: 5, jul: 6, aug: 7, sep: 8, oct: 9, nov: 10, dec: 11 }
const lmNum = s => Number(String(s).replace(/[,*]/g, ""))
const lmPad = n => String(n).padStart(2, "0")
// "02-Jul-26", "2-Jun-26", "02 May 2026", "31-Aug-26", "27/09/2026" → "2026-07-02"
function lmDay(s) {
  let m = String(s).match(/(\d{1,2})[-\s]([A-Za-z]{3,9})[-\s,]*(\d{2,4})/)
  if (m && LM_MON[m[2].slice(0, 3).toLowerCase()] != null) { const y = m[3].length === 2 ? 2000 + Number(m[3]) : Number(m[3]); return `${y}-${lmPad(LM_MON[m[2].slice(0, 3).toLowerCase()] + 1)}-${lmPad(m[1])}` }
  m = String(s).match(/(\d{1,2})[-\/.](\d{1,2})[-\/.](\d{2,4})/)
  if (m) { const y = m[3].length === 2 ? 2000 + Number(m[3]) : Number(m[3]); return `${y}-${lmPad(m[2])}-${lmPad(m[1])}` }
  return ""
}
function lenderOf(from, text) {
  const dom = String(from).replace(/^.*@/, "").replace(/>.*$/, "")
  for (const [re, name, match] of LENDERS) if (re.test(dom)) return { name, match }
  for (const [re, name, match] of LENDERS) if (re.test(text.slice(0, 600))) return { name, match }
  const d = dom.split(".")[0]
  return d ? { name: d.replace(/\b\w/g, c => c.toUpperCase()), match: d } : null
}
function loanTypeOf(text) {
  if (/top[- ]?up/i.test(text) && /home|housing|mortgage|property/i.test(text)) return "Home loan top-up"
  if (/home loan|housing loan|housing finance|mortgage|property/i.test(text)) return "Home loan"
  if (/car loan|vehicle loan|auto loan|two[- ]wheeler/i.test(text)) return "Car loan"
  if (/education loan/i.test(text)) return "Education loan"
  if (/personal loan/i.test(text)) return "Personal loan"
  if (/gold loan/i.test(text)) return "Gold loan"
  if (/business loan/i.test(text)) return "Business loan"
  return "Loan"
}

// msg: { from, subject, body, date }. Returns null, or
// { lender, match, account, type, emi, due, paidMonth, closed, sanctioned, disbursed, asOf, start, rate, day }
function parseLoanMail(msg) {
  const from = String(msg.from || ""), subject = String(msg.subject || "")
  const body = String(msg.body || "").replace(/&#39;/g, "'").replace(/&amp;/g, "&").replace(/[ \t ]+/g, " ")
  const text = (subject + " · " + body).replace(/\s+/g, " ")
  const mailDay = new Date(msg.date instanceof Date ? msg.date.getTime() : Date.parse(msg.date)).toISOString().slice(0, 10)
  if (/information@|mailers?\.|news@|newsletter|offers?@|marketing|promo/i.test(from)) return null
  if (/pre-?approved|offer|apply now|starting at|lowest rate|congratulations|upgrade your|dream (home|car)/i.test(subject)) return null
  if (!/loan|\bEMI\b|foreclos/i.test(text)) return null
  // the loan account: "Loan Account Number H001HHL1234567", "Loan A/c No. 701234567", "File Number: 701234567", "{H001HHL1234567}"
  const acct = text.match(/(?:loan\s*(?:account|a\/c)(?:\s*(?:number|no\.?|bearing))?|file\s*number|\bLAN\b)\s*[:_\-\s{]*\s*([A-Z]{0,6}\d[A-Z0-9]{5,19})/i)
    || (/loan/i.test(subject) && text.match(/account\s*(?:number|no\.?|bearing)\s*[:\s*]*\s*([A-Z]{0,6}\d[A-Z0-9]{5,19})/i))
  if (!acct || /^x/i.test(acct[1])) return null
  const lender = lenderOf(from, text)
  if (!lender) return null
  const r = { lender: lender.name, match: lender.match, account: acct[1].toUpperCase(), type: loanTypeOf(text + " " + lender.name), mailDay }
  let m
  // "EMI of Rs. 53783 towards Loan Account Number … is due on 02-Jul-26"
  if ((m = text.match(/EMI\s*(?:of|amount(?: of)?)?\s*(?:Rs\.?|INR|₹)\s*([\d,]+(?:\.\d+)?)[\s\S]{0,120}?\bdue\s*(?:on|date)?\s*:?\s*(\d{1,2}[-\s][A-Za-z]{3,9}[-\s,]*\d{2,4}|\d{1,2}[-\/.]\d{1,2}[-\/.]\d{2,4})/i))) {
    r.emi = lmNum(m[1]); r.due = lmDay(m[2]); if (r.due) r.day = Number(r.due.slice(8, 10))
  }
  // "receipt of EMI of ₹53783 for July'26", "EMI of Rs 12,000 has been received/debited"
  if ((m = text.match(/receipt of (?:your )?EMI (?:of )?(?:Rs\.?|INR|₹)\s*([\d,]+(?:\.\d+)?)(?:\s*for\s*([A-Za-z]{3,9})'?\s*(\d{2,4}))?/i))
    || (m = text.match(/EMI (?:of )?(?:Rs\.?|INR|₹)\s*([\d,]+(?:\.\d+)?) (?:has been|was|is) (?:received|paid|debited|collected)/i))) {
    r.emi = r.emi || lmNum(m[1])
    if (m[2] && LM_MON[m[2].slice(0, 3).toLowerCase()] != null) { const y = m[3].length === 2 ? 2000 + Number(m[3]) : Number(m[3]); r.paidMonth = `${y}-${lmPad(LM_MON[m[2].slice(0, 3).toLowerCase()] + 1)}` }
  }
  // foreclosure / closure
  if (/successfully (?:fore)?closed|has been (?:fully )?(?:fore)?closed|confirmation of (?:fore)?closure|loan (?:account )?(?:is|has been) closed|closure of (?:your )?loan|foreclosure of your loan/i.test(text) && !/request|quote|letter will|how to|would like to/i.test(subject)) r.closed = mailDay
  // sanction and disbursement
  if ((m = text.match(/(?:amount\s*)?sanctioned\s*(?:amount)?\s*[:\-]?\s*(?:Rs\.?|INR|₹)?\s*\*?([\d,]{5,})/i))) r.sanctioned = lmNum(m[1])
  if ((m = text.match(/for an amount of\s*(?:Rs\.?|INR|₹)\s*\*?([\d,]{5,})\*?\s*availed on\s*\*?(\d{1,2}[-\s][A-Za-z]{3}[-\s]\d{2,4})/i))) { r.sanctioned = r.sanctioned || lmNum(m[1]); r.start = lmDay(m[2]) }
  if ((m = text.match(/(?:amount\s*)?disbursed\s*(?:amount)?\s*[:\-]?\s*(?:Rs\.?|INR|₹)?\s*\*?([\d,]{5,})/i))) r.disbursed = lmNum(m[1])
  if ((m = text.match(/\bas on\s*:?\s*(\d{1,2}[-\s][A-Za-z]{3}[-\s]\d{2,4}|\d{1,2}[-\/.]\d{1,2}[-\/.]\d{2,4})/i))) r.asOf = lmDay(m[1])
  // the rate: "rate of interest 8.40% p.a.", "ROI revised to 8.5%"
  if ((m = text.match(/(?:rate of interest|interest rate|\bROI\b)([^%]{0,100}%(?:\s*p\.?\s*a\.?)?(?:\s*to\s*[\d.]+\s*%)?)/i))) {
    const pcts = [...m[1].matchAll(/(\d{1,2}(?:\.\d{1,2})?)\s*%/g)].map(x => Number(x[1]))
    const v = /\bto\s*[\d.]+\s*%/i.test(m[1]) ? pcts[pcts.length - 1] : pcts[0]
    if (v >= 4 && v <= 30) r.rate = v
  }
  if (r.rate && (m = text.match(/(?:with effect from|w\.?e\.?f\.?|effective (?:from|date)\s*:?)\s*(\d{1,2}(?:st|nd|rd|th)?[-\s][A-Za-z]{3,9}[-\s,]*\d{2,4}|\d{1,2}[-\/.]\d{1,2}[-\/.]\d{2,4})/i))) r.rateFrom = lmDay(m[1].replace(/(\d)(st|nd|rd|th)/, "$1"))
  if (r.rate && !r.rateFrom) r.rateFrom = mailDay
  // the term: "tenure of 240 months", "tenure: 20 years", "240 months tenure"
  if ((m = text.match(/(?:balance |remaining |revised )?tenure\s*(?:of|:|is|-)?\s*(\d{2,3})\s*months/i) || text.match(/(\d{2,3})\s*months?\s*tenure/i))) r.tenure = Number(m[1])
  else if ((m = text.match(/tenure\s*(?:of|:|is|-)?\s*(\d{1,2})\s*years/i))) r.tenure = Number(m[1]) * 12
  if (r.tenure && /balance|remaining|residual/i.test(text.slice(Math.max(0, text.search(/tenure/i) - 30), text.search(/tenure/i) + 10))) { r.tenureLeft = r.tenure; delete r.tenure }
  // part payment / prepayment acknowledgement: "Your payment of Rs. 60125 towards Loan account no … has been processed"
  if (/part[- ]?payment|pre[- ]?payment|partial payment/i.test(subject + " " + text.slice(0, 400)) && (m = text.match(/payment of\s*(?:Rs\.?|INR|₹)\s*([\d,]+(?:\.\d+)?)/i))) r.partPayment = lmNum(m[1])
  if (!(r.emi || r.closed || r.sanctioned || r.disbursed || r.rate || r.tenure || r.partPayment)) return null
  if (/pre[- ]?emi|paying interest on the amount disbursed|not started paying back the main loan/i.test(text)) r.preEmi = true
  return r
}

// Combine what all the emails say, per account (latest wins).
function loanMailSummary(found) {
  const by = {}
  for (const f of found.sort((a, b) => a.mailDay < b.mailDay ? -1 : 1)) {
    const s = by[f.account] = by[f.account] || { account: f.account, lender: f.lender, match: f.match, type: f.type, lastMail: "" }
    for (const k of ["emi", "due", "day", "paidMonth", "closed", "sanctioned", "disbursed", "asOf", "start", "rate", "rateFrom", "preEmi", "tenure", "tenureLeft"]) if (f[k] != null && f[k] !== "") s[k] = f[k]
    if (f.rate) (s.rates = s.rates || []).push({ rate: f.rate, from: f.rateFrom || f.mailDay })
    if (f.type !== "Loan") s.type = f.type
    s.lastMail = f.mailDay
    if (f.emi && !f.closed) s.emiDay = f.mailDay
  }
  // a later EMI notice means the loan is still running
  for (const s of Object.values(by)) if (s.closed && s.emiDay && s.emiDay > s.closed) delete s.closed
  return Object.values(by)
}


// ---------- loans: fill the Loans table from your lender's emails, and look back for past EMIs ----------
// Once a day it reads lender emails (EMI due, EMI received, sanction and disbursement, foreclosure) and:
// - fills in a tracked loan's account number, EMI, debit day, rate, loan amount and amount disbursed when they're empty,
//   and closes it when the lender confirms the foreclosure;
// - adds a loan it hasn't seen before as a suggestion (Active off), which you confirm on the dashboard.
// The first run reads a year of lender emails, and six months of bank alerts for EMI debits (no new expenses).

function ensureLoanColumns(token) {
  const props = PropertiesService.getScriptProperties()
  if (props.getProperty("LOAN_COLS_OK") === "4") return
  notion(token, "patch", `databases/${LOANS_DB}`, { properties: {
    Account: { rich_text: {} }, "Loan amount": { number: { format: "rupee" } }, Disbursed: { number: { format: "rupee" } },
    Suggested: { checkbox: {} }, "From email": { rich_text: {} }, "Rate history": { rich_text: {} }, Tenure: { number: {} }, "Doc figures": { rich_text: {} } } })
  props.setProperty("LOAN_COLS_OK", "4")
}

function syncLoanMails(token) {
  if (!LOANS_DB) return { updated: 0, added: 0 }
  const props = PropertiesService.getScriptProperties(), now = Date.now()
  const receiptsDone = props.getProperty("LOAN_RECEIPTS_V2") === "1"
  if (receiptsDone && now - Number(props.getProperty("LOAN_MAIL_AT") || 0) < 20 * 36e5) return { updated: 0, added: 0 }
  ensureLoanColumns(token)
  const first = !props.getProperty("LOAN_MAIL_AT") || !receiptsDone   // a year back, once (again after the update that reads receipts)
  const q = `(loan OR EMI OR foreclosure OR foreclosed) -category:promotions -category:social newer_than:${first ? 400 : 4}d`
  const found = []
  for (const t of GmailApp.search(q, 0, first ? 300 : 60))
    for (const m of t.getMessages()) {
      const r = parseLoanMail({ from: m.getFrom(), subject: m.getSubject(), body: m.getPlainBody(), date: m.getDate() })
      if (r) found.push(r)
    }
  const summary = loanMailSummary(found.slice())
  const res = applyLoanMails(token, summary, istDay(now))
  try { res.paid = saveLenderReceipts(token, found, summary); props.setProperty("LOAN_RECEIPTS_V2", "1") } catch (e) { Logger.log("Lender receipts failed: " + (e.stack || e)) }
  props.setProperty("LOAN_MAIL_AT", String(now))
  return res
}

function applyLoanMails(token, summary, today) {
  const date = p => (p && p.date ? p.date.start.slice(0, 10) : "")
  const rows = queryAll(token, LOANS_DB, {}).map(pg => { const p = pg.properties
    return { id: pg.id, name: plain(p.Loan), match: plain(p["Lender match"]) || plain(p.Loan), account: plain(p.Account).toUpperCase(), emi: (p.EMI || {}).number || 0,
      day: (p["Debit day"] || {}).number || 0, rate: (p.Rate || {}).number || 0, amount: (p["Loan amount"] || {}).number || 0, disbursed: (p.Disbursed || {}).number || 0,
      start: date(p.Start), closed: date(p.Closed), active: !!(p.Active && p.Active.checkbox), rateHistory: plain(p["Rate history"]), tenure: (p.Tenure || {}).number || 0 } })
  const to = (() => { try { return Session.getEffectiveUser().getEmail() } catch (e) { return "" } })()
  const fyStart = finYear(shiftDay(today, -366)).start
  const lenderHit = (row, s) => { const w = wordsOfLoan(row.match); const t = " " + wordsOfLoan(s.lender + " " + s.match).join(" ") + " "; return w.length > 0 && w.every(x => t.includes(" " + x)) }
  const near = (a, b) => !a || !b || Math.abs(a - b) <= Math.max(a, b) * 0.03
  let updated = 0, added = 0
  // accounts with an EMI first, biggest first, so a home loan claims its row before its small top-up
  for (const s of summary.sort((a, b) => (b.emi || 0) - (a.emi || 0))) {
    let row = rows.find(r => r.account && r.account === s.account)
      || rows.find(r => !r.account && lenderHit(r, s) && near(r.emi, s.emi) && (r.active || !r.closed || !s.closed || s.closed <= r.closed))
    const fmt = iso => { const [y, m, d] = iso.split("-").map(Number); return `${d} ${["Jan","Feb","Mar","Apr","May","Jun","Jul","Aug","Sep","Oct","Nov","Dec"][m - 1]} ${y}` }
    const note = [`Loan A/c ${s.account}`, s.emi ? `EMI ${rupees(s.emi)}${s.due ? ", due " + fmt(s.due) : ""}` : "",
      s.sanctioned ? `loan ${rupees(s.sanctioned)}` : "", s.disbursed ? `disbursed ${rupees(s.disbursed)}${s.asOf ? " as on " + fmt(s.asOf) : ""}` : "",
      s.rate ? `rate ${s.rate}%` : "", s.preEmi ? "paying pre-EMI interest (principal not started)" : "", s.closed ? `closed (email of ${fmt(s.closed)})` : "",
      `from ${s.lender}'s email of ${fmt(s.lastMail)}`].filter(Boolean).join(" · ")
    const P = {}
    const txt = v => ({ rich_text: [{ text: { content: String(v).slice(0, 1900) } }] })
    if (row) {
      if (!row.account) P.Account = txt(s.account)
      // EMI from the lender's own notice: fill it, or update it when a recent notice shows a new amount
      if (s.emi && s.emi >= 500 && !s.closed && (!row.emi || (s.emiDay && s.emiDay >= shiftDay(today, -45) && Math.abs(s.emi - row.emi) > row.emi * 0.01))) P.EMI = { number: s.emi }
      if (s.day && (!row.day || P.EMI)) P["Debit day"] = { number: s.day }
      // rates: keep a history, so each month's interest uses the rate that applied then
      if (s.rate) {
        const hist = parseRates(row.rateHistory, row.rate), cur = hist.length ? hist[hist.length - 1].rate : 0
        for (const x of (s.rates || [{ rate: s.rate, from: s.rateFrom }])) {
          const last = hist.length ? hist[hist.length - 1] : null
          if (last && Math.abs(last.rate - x.rate) < 0.005) continue
          if (hist.some(h => Math.abs(h.rate - x.rate) < 0.005 && h.from === x.from)) continue
          hist.push({ rate: x.rate, from: hist.length ? x.from : "2000-01-01" }); hist.sort((a, b) => a.from < b.from ? -1 : 1)
        }
        const now = hist.length ? hist[hist.length - 1].rate : 0
        if (formatRates(hist) !== formatRates(parseRates(row.rateHistory, row.rate))) {
          P["Rate history"] = txt(formatRates(hist)); P.Rate = { number: now }
          if (cur && Math.abs(now - cur) >= 0.005 && to && row.active) {
            const from = hist[hist.length - 1].from
            MailApp.sendEmail(to, `${row.name || s.lender}: rate ${now > cur ? "up" : "down"} to ${now}%`,
              `${s.lender} changed your loan rate from ${cur}% to ${now}%${from ? ` from ${fmt(from)}` : ""}.\n\n` +
              `${P.EMI ? `Your EMI changes to ${rupees(s.emi)}.` : "If your EMI stays the same, the loan simply runs " + (now > cur ? "longer" : "shorter") + "."} The dashboard's Loans tab now uses the new rate from that date, and shows the new end date.\n\n(Sent by your card-alerts Google Apps Script.)`)
          }
        }
      }
      if (s.tenure && !row.tenure) P.Tenure = { number: s.tenure }
      if (s.sanctioned && s.sanctioned !== row.amount) P["Loan amount"] = { number: s.sanctioned }
      if (s.disbursed && s.disbursed !== row.disbursed) P.Disbursed = { number: s.disbursed }
      if (s.start && !row.start) P.Start = { date: { start: s.start } }
      if (s.closed && row.active && !row.closed) { P.Closed = { date: { start: s.closed } }; P.Active = { checkbox: false } }
      if (Object.keys(P).length) { P["From email"] = txt(note); notion(token, "patch", `pages/${row.id}`, { properties: P }); updated++ }
      row.account = s.account
      continue
    }
    // a loan we haven't seen: suggest it (or keep it as closed, for this year's totals)
    if (s.closed && s.closed < fyStart) continue
    if (!s.emi && !s.sanctioned) continue
    const twin = rows.some(r => lenderHit(r, s))
    const name = `${s.type} · ${s.lender}${twin ? ` (A/c …${s.account.slice(-4)})` : ""}`
    const props = { Loan: { title: [{ text: { content: name } }] }, Type: { select: { name: s.type === "Loan" ? "Other" : s.type === "Home loan top-up" ? "Home loan" : s.type } }, "Lender match": txt(s.match), Account: txt(s.account),
      Active: { checkbox: false }, Suggested: { checkbox: !s.closed }, "From email": txt(note), Notes: txt(`Found in ${s.lender}'s emails.`) }
    if (s.emi && s.emi >= 100) props.EMI = { number: s.emi }
    if (s.day) props["Debit day"] = { number: s.day }
    if (s.rate) { props.Rate = { number: s.rate }; props["Rate history"] = txt(formatRates((s.rates || [{ rate: s.rate, from: s.rateFrom }]).filter((x, k, l) => !k || Math.abs(l[k - 1].rate - x.rate) >= 0.005).map((x, k) => ({ rate: x.rate, from: k ? x.from : "2000-01-01" })))) }
    if (s.tenure) props.Tenure = { number: s.tenure }
    if (s.sanctioned) props["Loan amount"] = { number: s.sanctioned }
    if (s.disbursed) props.Disbursed = { number: s.disbursed }
    if (s.start) props.Start = { date: { start: s.start } }
    if (s.closed) props.Closed = { date: { start: s.closed } }
    notion(token, "post", "pages", { parent: { database_id: LOANS_DB }, properties: props })
    rows.push({ id: "", name, match: s.match, account: s.account, emi: s.emi || 0, closed: s.closed || "", active: false })
    added++
  }
  return { updated, added }
}
const LOAN_DEBIT = /loan|housing|home fin|bajaj fin|lic housing|pnb housing|tata capital|aditya birla|idfc first|hdfc ltd|hdfc limited|icici home|sbi home|axis finance|kotak mahindra prime|l ?& ?t fin|cholamandalam|shriram|muthoot|mahindra fin|home credit|piramal|aavas|indiabulls|poonawalla|hero fincorp/i
const wordsOfLoan = s => String(s || "").toLowerCase().replace(/[^a-z0-9 ]+/g, " ").split(/\s+/).filter(Boolean)

// One-time: read six months of bank alerts and save the EMI-like debits, so loans are recognised from day one.
// Runs in small steps (one per sync) and never adds expenses or To do items: these are all older than your logging.
function scanLoanHistory(token) {
  const props = PropertiesService.getScriptProperties()
  const step = props.getProperty("LOAN_HISTORY_AT")
  if (step === "done") return { saved: 0 }
  const start = Number(step || 0), PAGE = 100
  const threads = GmailApp.search(GMAIL_QUERY.replace(/newer_than:\S+/, "newer_than:185d") + " older_than:2d", start, PAGE)
  const since = shiftDay(istDay(Date.now()), -190)
  const have = queryAll(token, ALERTS_DB, { filter: { and: [{ property: "Date", date: { on_or_after: since } },
    { or: [{ property: "Category", select: { equals: "Loan EMI" } }, { property: "Channel", select: { equals: "EMI / auto-debit" } }, { property: "Channel", select: { equals: "Other bank debit" } }] }] } })
    .map(pg => { const p = pg.properties, s = p.Date.date ? p.Date.date.start : pg.created_time
      return { messageId: plain(p["Message ID"]), card: sel(p.Card), type: sel(p.Type) || "Spend", amount: p.Amount.number || 0, source: sel(p.Source), merchant: plain(p.Merchant), time: Date.parse(s) } })
  const seen = new Set(have.map(r => r.messageId))
  let saved = 0
  for (const t of threads) for (const m of t.getMessages()) {
    const id = m.getId()
    if (seen.has(id)) continue
    seen.add(id)
    const r = parseAlert({ id, subject: m.getSubject(), from: m.getFrom(), body: m.getPlainBody(), date: m.getDate() })
    if (!r || r.skip || r.fx || r.type !== "Spend") continue
    // only clear loan repayments from a bank account: a lender's name or loan words, ₹1,000 or more
    if (r.card !== "Bank account" || !(r.amount >= 1000) || r.category !== "Loan EMI") continue
    if (!LOAN_DEBIT.test(r.merchant) || /convert this spend|find a store|what.s more/i.test(r.merchant)) continue
    if (isDuplicate(r, have)) continue
    r.pageId = createRow(token, r).id
    have.push(r); saved++
  }
  props.setProperty("LOAN_HISTORY_AT", threads.length < PAGE || start + PAGE >= 500 ? "done" : String(start + PAGE))
  return { saved }
}

// "EMI received" emails from the lender are proof of payment. Some banks send no alert for a NACH debit
// (a small top-up EMI, say), so each receipt is saved as a payment unless a bank alert for it already exists.
function saveLenderReceipts(token, found, summary) {
  const paid = found.filter(f => (f.paidMonth && f.emi > 0 || f.partPayment > 0) && !f.closed)
  if (!paid.length) return 0
  const pad2 = n => String(n).padStart(2, "0")
  const dues = {}
  for (const f of found) if (f.due) dues[f.account + "|" + f.due.slice(0, 7)] = f.due
  const since = shiftDay(istDay(Date.now()), -420)
  const have = queryAll(token, ALERTS_DB, { filter: { and: [{ property: "Date", date: { on_or_after: since } },
    { or: [{ property: "Category", select: { equals: "Loan EMI" } }, { property: "Channel", select: { equals: "EMI / auto-debit" } }, { property: "Channel", select: { equals: "Other bank debit" } }] }] } })
    .map(pg => { const q = pg.properties, s = q.Date.date ? q.Date.date.start : pg.created_time
      return { pageId: pg.id, messageId: plain(q["Message ID"]), card: sel(q.Card), type: sel(q.Type) || "Spend", amount: q.Amount.number || 0, source: sel(q.Source), merchant: plain(q.Merchant), time: Date.parse(s) } })
  const seen = new Set(have.map(h => h.messageId))
  let saved = 0
  for (const f of paid) {
    const part = f.partPayment > 0 && !f.paidMonth
    const id = part ? `lender|${f.account}|part|${f.mailDay}|${f.partPayment}` : `lender|${f.account}|${f.paidMonth}`
    if (seen.has(id)) continue
    const s = summary.find(x => x.account === f.account) || {}
    const day = part ? f.mailDay : dues[f.account + "|" + f.paidMonth] || `${f.paidMonth}-${pad2(s.day || Number(f.mailDay.slice(8, 10)) || 1)}`
    const time = Date.parse(day + "T10:00:00+05:30"), tag = ` · A/c …${f.account.slice(-4)}`
    const r = { messageId: id, source: "Statement", status: "Parsed", amount: part ? f.partPayment : f.emi, card: "Bank account", type: "Spend",
      channel: part ? "Other bank debit" : "EMI / auto-debit", category: "Loan EMI", merchant: `${f.lender} ${part ? "part payment" : "EMI"}${tag}`, time, iso: istIso(time), ignored: true }
    // already seen as a bank alert (same amount, within a few days)? Then tag that alert with the account, so it goes to the right loan
    const bank = have.find(h => h.type === "Spend" && Math.abs(h.amount - r.amount) < 1 && (part ? h.time <= time + 864e5 && time - h.time <= 4 * 864e5 : Math.abs(h.time - time) <= 4 * 864e5))
    if (bank) {
      if (bank.pageId && !/A\/c …\w{4}/.test(bank.merchant)) { notion(token, "patch", `pages/${bank.pageId}`, { properties: { Merchant: { title: [{ text: { content: (bank.merchant + tag).slice(0, 200) } }] } } }); bank.merchant += tag }
      seen.add(id); continue
    }
    createRow(token, r); have.push(r); seen.add(id); saved++
  }
  return saved
}

// ---------- loans: read your lender's PDFs (statement of account, repayment schedule, sanction letter, interest certificate) ----------
// parseLoanDoc() takes the text of one PDF and returns the loan figures it states, or null.
// Layouts differ by lender, so it looks for labelled values ("Rate of Interest 8.05%", "Principal Outstanding ₹ 68,12,345")
// rather than table positions. Nothing is written to a loan until you tap "Use these" on the dashboard.

// ---------- reading labelled values out of a PDF's text (shared by the loan and policy readers) ----------
// The same figure comes out of a PDF in different shapes: "Label: value" on one line, a label with its value on the
// next line, a row of column headings with the values on the row below, or (from Google's converter) a block of
// labels followed by a block of values. docKV(text).get(labels, kind) tries each shape, label by label in the order
// given, and keeps the first candidate that is the right kind of value (an amount, a date, a percentage…).
function docKV(text) {
  const MONTHS = { jan: 1, feb: 2, mar: 3, apr: 4, may: 5, jun: 6, jul: 7, aug: 8, sep: 9, oct: 10, nov: 11, dec: 12 }
  const MON_RE = "(jan(?:uary)?|feb(?:ruary)?|mar(?:ch)?|apr(?:il)?|may|june?|july?|aug(?:ust)?|sept?(?:ember)?|oct(?:ober)?|nov(?:ember)?|dec(?:ember)?)"
  const pad = n => String(n).padStart(2, "0")
  const src = String(text || "").replace(/[\u00a0\u2007\u202f]/g, " ").replace(/\r/g, "").replace(/[–—]/g, "-")
    .replace(/([a-z)])\d(?=[\s:|])/g, "$1")            // footnote marks glued to a label: "Interest Rate1 (FRR)", "EMI Amount4"
  const lines = src.split("\n").map(l => l.replace(/[ ]+$/, ""))
  const cellsOf = l => l.split(/\t+| {2,}| \| /).map(c => c.trim()).filter(Boolean)

  // ----- values -----
  const dateRes = [
    [new RegExp("\\b(\\d{1,2})(?:st|nd|rd|th)?[-\\/.\\s]*" + MON_RE + "\\b\\.?[-\\/.\\s,']*(\\d{4}|\\d{2})\\b", "i"), m => [m[3], MONTHS[m[2].slice(0, 3).toLowerCase()], m[1]]],
    [new RegExp("\\b" + MON_RE + "\\b\\.?\\s+(\\d{1,2})(?:st|nd|rd|th)?,?\\s+(\\d{4})\\b", "i"), m => [m[3], MONTHS[m[1].slice(0, 3).toLowerCase()], m[2]]],
    [/\b(\d{4})-(\d{1,2})-(\d{1,2})\b/, m => [m[1], m[2], m[3]]],
    [/(?<![\d.\/-])(\d{1,2})[-\/.](\d{1,2})[-\/.](\d{4}|\d{2})(?![\d.\/-]*\d)/, m => [m[3], m[2], m[1]]]
  ]
  // the first date in s, as { iso, index, length }
  function dateIn(s) {
    s = String(s || ""); let best = null
    for (const [re, pick] of dateRes) {
      const m = s.match(re); if (!m) continue
      let [y, mo, d] = pick(m).map(Number)
      if (String(pick(m)[0]).length === 2) y += 2000
      if (!(mo >= 1 && mo <= 12 && d >= 1 && d <= 31 && y >= 1990 && y <= 2100)) continue
      if (!best || m.index < best.index) best = { iso: `${y}-${pad(mo)}-${pad(d)}`, index: m.index, length: m[0].length }
    }
    return best
  }
  const blank = (s, from, len) => s.slice(0, from) + " ".repeat(len) + s.slice(from + len)
  // dates, percentages and "12 months" taken out so their digits aren't read as amounts
  function scrub(s) {
    s = String(s || ""); let d
    for (let i = 0; i < 6 && (d = dateIn(s)); i++) s = blank(s, d.index, d.length)
    return s.replace(/\d+(?:\.\d+)?\s*%/g, m => " ".repeat(m.length)).replace(/\b\d{1,3}\s*(?:months?|years?|yrs?|days?|emis?)\b/gi, m => " ".repeat(m.length))
  }
  const AMT = /(₹|\brs\b\.?|\binr\b)?\s*(?<![A-Za-z0-9.,])(\d{1,3}(?:,\d{2,3})+(?:\.\d{1,2})?|\d+(?:\.\d{1,2})?)(?![\d,]*[A-Za-z0-9])(?:\s*\/-)?(?:\s*(crores?|cr|lakhs?|lacs?|l)\b)?/gi
  // the first amount in s: { value, index }
  function moneyIn(s, min) {
    const t = scrub(s); AMT.lastIndex = 0; let m
    while ((m = AMT.exec(t))) {
      const digits = m[2].replace(/[,.]/g, "")
      if (!m[1] && !m[3] && !/,/.test(m[2]) && digits.length >= 9) continue          // a phone or account number
      let n = Number(m[2].replace(/,/g, "")); const u = (m[3] || "").toLowerCase()
      if (/^cr/.test(u)) n *= 1e7; else if (u) n *= 1e5
      if (n >= (min || 1)) return { value: Math.round(n * 100) / 100, index: m.index }
    }
    return null
  }
  function pctIn(s) {
    const m = String(s || "").match(/(\d{1,2}(?:\.\d{1,3})?)\s*%/) || String(s || "").match(/\b(\d{1,2}\.\d{1,3})\s*(?:p\.?\s*a\.?|per annum)/i)
    return m ? { value: Number(m[1]), index: m.index } : null
  }
  function intIn(s) {
    const t = scrub(String(s || "").replace(/\d+\s*(?:months?|emis?)/gi, m => m.replace(/\D+$/, "") + "#"))   // keep "289 months"
    const m = t.match(/(?<![\d.,\/-])(\d{1,3})(?![\d.,\/-]*\d)/)
    return m ? { value: Number(m[1]), index: m.index } : null
  }
  function codeIn(s) {
    const t = scrub(s), m = t.match(/(?<![A-Za-z0-9])([A-Z]{0,6}[\/-]?\d[\dA-Z\/-]{4,30})(?![A-Za-z0-9])/i)
    return m && m[1].replace(/\D/g, "").length >= 5 ? { value: m[1].toUpperCase(), index: m.index } : null
  }
  const valueIn = (s, kind, o) => kind === "money" ? moneyIn(s, o.min) : kind === "date" ? (d => d && { value: d.iso, index: d.index })(dateIn(s))
    : kind === "pct" ? pctIn(s) : kind === "int" ? intIn(s) : kind === "code" ? codeIn(s)
    : (v => v ? { value: v, index: 0 } : null)(String(s || "").replace(/^[\s:\-|.=]+/, "").split(/\t| {2,}/)[0].trim())
  const ok = (v, o) => v && (o.max == null || v.value <= o.max) && (o.min == null || typeof v.value !== "number" || v.value >= o.min) && (!o.test || o.test(v.value))
  const hasValue = (s, kind, o) => ok(valueIn(s, kind, o), o)
  const VALUE_WORD = /^(?:rs\.?|inr|lakhs?|lacs?|crores?|cr|months?|years?|yrs?|days?|p\.?a\.?|monthly|quarterly|half[- ]?yearly|yearly|annual(?:ly)?|single|yly|hly|qly|mly|floating|fixed|yes|no|nil|not|applicable|na|n\/a|self|spouse|floater|individual|active|closed)$/i
  // a line that is only a value: amounts, dates, codes, and words like "Monthly" or "Lakh"
  const isValueLine = l => { l = String(l || "").trim(); if (!l || /\t| {2,}/.test(l)) return false
    let t = l, d; for (let k = 0; k < 3 && (d = dateIn(t)); k++) t = blank(t, d.index, d.length)
    const words = t.split(/[\s,:()\/₹%-]+/).filter(Boolean)
    if (!words.length) return t !== l                 // only a date
    return words.length <= 4 && words.every(w => /\d/.test(w) || VALUE_WORD.test(w)) && (/\d/.test(l) || words.length <= 2) }
  const looksLabel = l => /[A-Za-z]{3}/.test(l) && !/\d/.test(l) && l.length <= 60 && !isValueLine(l)

  // ----- labels -----
  const reOf = l => l instanceof RegExp ? new RegExp(l.source, "i") : new RegExp("(?<![A-Za-z])" + l + "(?![A-Za-z])", "i")
  function candidates(i, m, kind, o) {
    const line = lines[i], out = [], after = line.slice(m.index + m[0].length)
    // 1. same line, after the label (a few filler words allowed: "Sanctioned", "(in Rs.)", "till date")
    const v1 = valueIn(kind === "text" ? after.replace(/^[\s:\-|.=]+/, "") : after, kind, o)
    // (not across a column break into another heading: "Sum Insured <tab> Premium 12,345")
    if (v1 && (kind === "text" || (v1.index <= (o.gap || 40) && !/(\t| {2,}) *[A-Za-z]{3}/.test(after.slice(0, v1.index))))) out.push(v1)
    // 2. a row of headings with the values on the row below, same column
    const cells = cellsOf(line), at = cells.findIndex(c => reOf(o.label).test(c))
    if (cells.length >= 2 && at >= 0 && !hasValue(cells[at].replace(reOf(o.label), ""), kind, o)) {
      for (let j = i + 1, seen = 0; j < lines.length && seen < 3; j++) {
        if (!lines[j].trim()) continue; seen++
        const row = cellsOf(lines[j])
        if (row.length === cells.length || (row.length >= 2 && row.length === cells.length - 1 && at < row.length)) { const v = valueIn(row[at], kind, o); if (v && (kind === "text" || v.index <= 6)) { out.push(v); break } }
      }
    }
    // 3. the value on the next line (a label on its own line)
    if (!after.trim() || !hasValue(after, kind, o)) {
      for (let j = i + 1, seen = 0; j < lines.length && seen < 2; j++) {
        if (!lines[j].trim()) continue; seen++
        const v = valueIn(lines[j], kind, o)
        if (v && (kind === "text" ? !/:\s*$/.test(lines[j]) && !/\b(number|no\.?|date|name|amount|premium|frequency|mode|term|insured|assured|period|type|plan)\s*$/i.test(lines[j]) : v.index <= 6)) { out.push(v); break }
        if (!looksLabel(lines[j].trim())) break
      }
    }
    // 4. a block of labels, then a block of values in the same order
    if (looksLabel(line.trim())) {
      let s = i, e = i
      while (s > 0 && looksLabel(lines[s - 1].trim())) s--
      while (e + 1 < lines.length && looksLabel(lines[e + 1].trim())) e++
      // values: the lines after the block that hold only a value; they pair with the last labels of the block
      // (a title or two often sits above the labels)
      const vals = []; for (let j = e + 1; j < lines.length && vals.length <= e - s && isValueLine(lines[j]); j++) vals.push(lines[j])
      const first = e - vals.length + 1
      if (vals.length >= 2 && i >= first) { const v = valueIn(vals[i - first], kind, o); if (v && v.index <= 6) out.push(v) }
    }
    return out
  }
  // labels: strings (regex source, matched as whole words) or RegExps, best first.
  // kind: money | date | pct | int | code | text. o: { min, max, not (regex for the words just before the label), test, gap }
  function get(labels, kind, o) {
    o = o || {}
    for (const l of [].concat(labels)) {
      const re = reOf(l)
      for (let i = 0; i < lines.length; i++) {
        const g = new RegExp(re.source, "gi"); let m
        while ((m = g.exec(lines[i]))) {
          if (o.not && o.not.test(lines[i].slice(Math.max(0, m.index - 25), m.index))) continue
          for (const v of candidates(i, m, kind, Object.assign({}, o, { label: l }))) if (ok(v, o)) return v.value
          if (!m[0]) break
        }
      }
    }
    return null
  }
  return { get, lines, text: src, dateIn: s => (d => d ? d.iso : "")(dateIn(s)), moneyIn: (s, min) => (v => v ? v.value : 0)(moneyIn(s, min)), pctIn: s => (v => v ? v.value : null)(pctIn(s)) }
}

const LD_MON = { jan: 0, feb: 1, mar: 2, apr: 3, may: 4, jun: 5, jul: 6, aug: 7, sep: 8, oct: 9, nov: 10, dec: 11 }
const ldPad = n => String(n).padStart(2, "0")
const ldNum = s => Number(String(s).replace(/[,\s]/g, ""))
function ldDay(s) {
  s = String(s || "").replace(/(\d)(st|nd|rd|th)\b/i, "$1")
  let m = s.match(/(\d{1,2})[-\s\/.]([A-Za-z]{3,9})[-\s\/.,]*(\d{2,4})/)
  if (m && LD_MON[m[2].slice(0, 3).toLowerCase()] != null) { const y = m[3].length === 2 ? 2000 + Number(m[3]) : Number(m[3]); return `${y}-${ldPad(LD_MON[m[2].slice(0, 3).toLowerCase()] + 1)}-${ldPad(m[1])}` }
  m = s.match(/([A-Za-z]{3,9})\s+(\d{1,2}),?\s+(\d{4})/)
  if (m && LD_MON[m[1].slice(0, 3).toLowerCase()] != null) return `${m[3]}-${ldPad(LD_MON[m[1].slice(0, 3).toLowerCase()] + 1)}-${ldPad(m[2])}`
  m = s.match(/(\d{1,2})[-\/.](\d{1,2})[-\/.](\d{2,4})/)
  if (m) { const y = m[3].length === 2 ? 2000 + Number(m[3]) : Number(m[3]); return `${y}-${ldPad(m[2])}-${ldPad(m[1])}` }
  m = s.match(/(\d{4})-(\d{2})-(\d{2})/)
  return m ? m[0] : ""
}
const LD_DATE_RE = "(\\d{1,2}(?:st|nd|rd|th)?[-\\s\\/.][A-Za-z]{3,9}[-\\s\\/.,]*\\d{2,4}|\\d{1,2}[-\\/.]\\d{1,2}[-\\/.]\\d{2,4}|[A-Za-z]{3,9}\\s+\\d{1,2},?\\s+\\d{4})"
const LD_AMT_RE = "(?:Rs\\.?|INR|₹)?\\s*(\\d[\\d,]{3,}(?:\\.\\d{1,2})?|\\d{1,3}\\.\\d{2}\\b)"

// The one-line patterns the reader started with; parseLoanDoc() runs these first, then docKV() over them.
function ldRegexFigures(text, meta = {}) {
  const t = String(text || "").replace(/[ \t ]+/g, " ").replace(/\r/g, "")
  const lines = String(text || "").replace(/\r/g, "").split("\n")
  const flat = t.replace(/\s*\n\s*/g, " ")
  const name = String(meta.filename || "") + " " + String(meta.subject || "")
  if (!/loan|emi|repayment|amortization|amortisation|foreclosure|sanction|interest certificate/i.test(flat + " " + name)) return null
  if (/credit card statement|card account summary|minimum amount due/i.test(flat.slice(0, 3000))) return null
  const r = { kind: "statement", mailDay: meta.mailDay || "" }
  const find = (re, i = 1) => { const m = flat.match(re); return m ? m[i] : null }
  // what kind of document
  if (/interest certificate|provisional certificate|certificate of interest/i.test(flat + name)) r.kind = "certificate"
  else if (/foreclosure (statement|letter|amount)|foreclos/i.test(name) || /foreclosure amount|amount payable for foreclosure/i.test(flat)) r.kind = "foreclosure"
  else if (/sanction letter|letter of sanction|sanctioned/i.test(flat.slice(0, 2500) + name) && !/statement of account/i.test(flat + name)) r.kind = "sanction"
  else if (/repayment schedule|amortization|amortisation/i.test(flat + name)) r.kind = "schedule"
  const acct = find(/(?:loan\s*(?:account|a\/c)(?:\s*(?:number|no\.?))?|\bLAN\b|agreement\s*(?:no|number)\.?|file\s*number)\s*[:\-]?\s*([A-Z]{0,6}\d[A-Z0-9]{5,19})/i)
    || find(/statement\s*of\s*account\s*(?:for|of|no\.?|number)\s*[:\-]?\s*([A-Z]{1,6}\d[A-Z0-9]{5,19})/i)
  if (acct && !/^x/i.test(acct)) r.account = acct.toUpperCase()
  let v
  if ((v = find(/(?:current\s*|applicable\s*|revised\s*|annualized\s*|effective\s*)?(?:rate of interest|interest rate|\bROI\b)\s*(?:\(%?\s*p\.?\s*a\.?\)?)?\s*[:\-]?\s*(\d{1,2}(?:\.\d{1,3})?)\s*%?/i)) && Number(v) >= 4 && Number(v) <= 30) r.rate = Number(v)
  if ((v = find(new RegExp("(?:principal\\s*outstanding|outstanding\\s*principal|principal\\s*o\\/s|\\bPOS\\b|balance\\s*principal|principal\\s*balance)(?:\\s*(?:amount|balance))?(?:\\s*\\(?(?:Rs\\.?|INR|₹)\\)?)?(?:\\s*as\\s*on\\s*" + LD_DATE_RE + ")?\\s*[:\\-]?\\s*" + LD_AMT_RE, "i"), 2))) r.outstanding = ldNum(v)
  if ((v = find(new RegExp("(?:principal\\s*outstanding|outstanding\\s*principal|principal\\s*o\\/s|\\bPOS\\b)[^.]{0,40}?as\\s*on\\s*" + LD_DATE_RE, "i")))) r.asOf = ldDay(v)
  if (!r.asOf && (v = find(new RegExp("(?:statement\\s*(?:date|as on)|as\\s*on|period\\s*(?:from)?[^\\n]{0,40}?to)\\s*[:\\-]?\\s*" + LD_DATE_RE, "i")))) r.asOf = ldDay(v)
  if ((v = find(new RegExp("(?:\\bEMI\\b|instal{1,2}ment)\\s*(?:amount)?\\s*(?:\\(?(?:Rs\\.?|INR|₹)\\)?)?\\s*[:\\-]?\\s*" + LD_AMT_RE, "i")))) { const n = ldNum(v); if (n >= 100) r.emi = n }
  if ((v = find(/(?:balance|remaining|residual|pending)\s*(?:tenure|tenor|term|emis?|instal{1,2}ments)\s*(?:\(?in months\)?)?\s*[:\-]?\s*(\d{1,3})/i))) r.tenureLeft = Number(v)
  if ((v = find(/(?:original\s*|total\s*|loan\s*)?(?:tenure|tenor|term)\s*(?:\(?in months\)?)?\s*[:\-]?\s*(\d{2,3})\s*(?:months)?/i)) && !/balance|remaining|residual/i.test(flat.slice(Math.max(0, flat.search(/tenure|tenor/i) - 15), flat.search(/tenure|tenor/i)))) r.tenure = Number(v)
  if ((v = find(new RegExp("(?:loan\\s*amount|amount\\s*financed|sanctioned\\s*amount|amount\\s*sanctioned|loan\\s*sanctioned)\\s*(?:\\(?(?:Rs\\.?|INR|₹)\\)?)?\\s*[:\\-]?\\s*" + LD_AMT_RE, "i")))) r.sanctioned = ldNum(v)
  if ((v = find(new RegExp("(?:amount\\s*disbursed|disbursed\\s*amount|total\\s*disbursement)\\s*(?:\\(?(?:Rs\\.?|INR|₹)\\)?)?\\s*[:\\-]?\\s*" + LD_AMT_RE, "i")))) r.disbursed = ldNum(v)
  if (r.kind === "foreclosure" && (v = find(new RegExp("(?:total\\s*)?(?:foreclosure\\s*amount|amount\\s*payable)[^\\d₹]{0,30}" + LD_AMT_RE, "i")))) r.foreclosure = ldNum(v)
  // interest certificate: the year and its interest and principal
  if (r.kind === "certificate") {
    const fy = flat.match(/(?:financial\s*year|F\.?Y\.?)\s*[:\-]?\s*(20\d{2})\s*[-–\/]\s*(\d{2,4})/i) || flat.match(/01[-\/.]0?4[-\/.](20\d{2})\s*(?:to|-)\s*31[-\/.]0?3[-\/.](20\d{2})/i)
    if (fy) r.fy = `FY ${fy[1]}–${String(fy[2]).slice(-2)}`
    if ((v = find(new RegExp("(?:interest\\s*(?:component|amount|paid|repaid|recovered)?)\\s*(?:\\(?(?:Rs\\.?|INR|₹)\\)?)?\\s*[:\\-]?\\s*" + LD_AMT_RE, "i")))) r.fyInterest = ldNum(v)
    if ((v = find(new RegExp("(?:principal\\s*(?:component|amount|paid|repaid|recovered))\\s*(?:\\(?(?:Rs\\.?|INR|₹)\\)?)?\\s*[:\\-]?\\s*" + LD_AMT_RE, "i")))) r.fyPrincipal = ldNum(v)
  }
  // payments listed in a statement (EMIs received, part payments): one per table row that starts with a date
  return r
}

// text: the PDF's text. meta: { filename, subject, from, mailDay }
// Labelled values are read with docKV() first (it copes with tables and labels on their own line);
// the older one-line patterns below fill whatever it didn't find.
function parseLoanDoc(text, meta = {}) {
  const r = ldRegexFigures(text, meta)
  if (!r) return null
  const kv = docKV(text), g = (labels, kind, o) => kv.get(labels, kind, o)
  const before = /(first|1st|pre[- ]?emi|overdue|bounce|penal)\s*$/i
  const acct = g(["loan\\s*(?:account|a\\/c)\\s*(?:number|no\\.?)?", "\\bLAN\\b", "agreement\\s*(?:no\\.?|number)", "statement\\s*of\\s*account\\s*(?:for|of)", "loan\\s*(?:no\\.?|number|reference(?:\\s*number)?)"], "code",
    { test: v => !/^X/i.test(v) && /[0-9]{4}/.test(v) })
  if (acct) r.account = acct
  const rateL = ["effective\\s*rate\\s*of\\s*interest", "current\\s*rate\\s*of\\s*interest", "applicable\\s*rate\\s*of\\s*interest", "revised\\s*rate\\s*of\\s*interest", "rate\\s*of\\s*interest", "interest\\s*rate", "\\bROI\\b"]
  const rate = g(rateL, "pct", { min: 4, max: 30 }) || g(rateL, "money", { min: 4, max: 30, gap: 25 })
  if (rate) r.rate = rate
  const outstanding = g(["principal\\s*outstanding(?:\\s*amount)?", "outstanding\\s*principal(?:\\s*amount)?", "principal\\s*o\\/s", "\\bPOS\\b", "balance\\s*principal", "principal\\s*balance", "loan\\s*outstanding", "outstanding\\s*(?:loan\\s*)?amount", "(?:principal\\s*)?outstanding\\s*balance", "balance\\s*outstanding", "total\\s*outstanding", "closing\\s*balance"], "money", { min: 1000 })
  if (outstanding) r.outstanding = outstanding
  const sanctioned = g(["loan\\s*amount\\s*sanctioned", "sanctioned\\s*(?:loan\\s*)?amount", "amount\\s*sanctioned", "amount\\s*financed", "loan\\s*sanctioned", "loan\\s*amount"], "money", { min: 10000 })
  if (sanctioned) r.sanctioned = sanctioned
  const disbursed = g(["loan\\s*amount\\s*disbursed", "amount\\s*disbursed(?:\\s*till\\s*date)?", "disbursed\\s*amount(?:\\s*till\\s*date)?", "total\\s*disburse?ment(?:\\s*amount)?", "disbursement\\s*amount"], "money", { min: 10000 })
  if (disbursed) r.disbursed = disbursed
  const cap = r.sanctioned || r.disbursed || r.outstanding
  const emi = g(["regular\\s*emi(?:\\s*amount)?", "current\\s*emi(?:\\s*amount)?", "revised\\s*emi(?:\\s*amount)?", "emi\\s*amount", "monthly\\s*emi", "emi", "instal{1,2}ment\\s*amount", "monthly\\s*instal{1,2}ment"], "money",
    { min: 100, not: before, test: v => !cap || v <= cap / 6 })
  if (emi) r.emi = emi
  const left = g(["balance\\s*(?:tenure|tenor|term)", "remaining\\s*(?:tenure|tenor|term)", "residual\\s*(?:tenure|tenor|term)", "pending\\s*(?:tenure|tenor)", "(?:balance|remaining|pending|outstanding)\\s*(?:emis?|instal{1,2}ments)", "(?:emis?|instal{1,2}ments)\\s*(?:remaining|pending|left)"], "int", { min: 1, max: 480 })
  if (left) r.tenureLeft = left
  const tenL = ["original\\s*(?:tenure|tenor|term)", "loan\\s*(?:tenure|tenor|term)", "total\\s*(?:tenure|tenor)", "tenure", "tenor", "repayment\\s*period"], notLeft = /(balance|remaining|residual|pending)\s*$/i
  let tenure = g(tenL, "int", { min: 6, max: 480, not: notLeft })
  if (!tenure) { const y = g(tenL, "text", { not: notLeft }); const m = String(y || "").match(/^(\d{1,2})\s*(?:years?|yrs?)/i); if (m) tenure = Number(m[1]) * 12 }
  if (tenure && tenure > 0) r.tenure = tenure
  const asOf = g([/statement\s*(?:date|as\s*on)/, /\bas\s*(?:on|of)\b/, /\bfrom\b\s*:?\s*\S+\s+to\b/, /\bperiod\b[^\n]{0,30}?\bto\b/], "date")
  const posOn = String(text || "").match(/(?:principal\s*outstanding|outstanding\s*principal|principal\s*o\/s|\bPOS\b)[^.\n]{0,40}?as\s*(?:on|of)\s*([^\n]{6,30})/i)
  if (posOn && kv.dateIn(posOn[1])) r.asOf = kv.dateIn(posOn[1]); else if (asOf) r.asOf = asOf
  if (r.kind === "foreclosure") { const f = g(["total\\s*foreclosure\\s*amount", "foreclosure\\s*amount", "net\\s*(?:amount\\s*)?payable", "total\\s*amount\\s*payable", "amount\\s*payable"], "money", { min: 1000 }); if (f) r.foreclosure = f }
  if (r.kind === "certificate") {
    const i = g(["total\\s*interest(?:\\s*(?:paid|amount|component))?", "interest\\s*(?:paid|component|amount|repaid|recovered)", "interest"], "money", { min: 100, not: /(rate\s*of|pre[- ]?emi|broken\s*period)\s*$/i })
    if (i) r.fyInterest = i
    const p = g(["total\\s*principal(?:\\s*(?:paid|amount|component))?", "principal\\s*(?:paid|component|amount|repaid|recovered)", "principal(?!\\s*(?:outstanding|o\\/s|balance))"], "money", { min: 100 })
    if (p) r.fyPrincipal = p
  }
  // payments listed in a statement (EMIs received, part payments)
  if (r.kind === "statement" || r.kind === "foreclosure") { const paid = docPayments(kv.lines, r.emi); if (paid.length) r.paid = paid }
  const useful = ["paid", "rate", "outstanding", "emi", "tenure", "tenureLeft", "sanctioned", "disbursed", "fyInterest", "foreclosure"].filter(k => r[k] != null)
  return useful.length ? r : null
}

// Table rows like "02-Jul-2026  Receipt - NACH Presentation  106.00  0.00" or "05/10/2026  EMI Recd - ECS  43,051.00  2,226.00  40,825.00".
// The date can sit anywhere in the row: some statements (and Google's converter) put the amount or description first.
// Dues, charges, bounces, disbursals and the foreclosure itself are left out: only money you paid towards EMIs or part payments.
function docPayments(lines, emi) {
  const out = [], anyDate = new RegExp(LD_DATE_RE)
  for (const raw of lines) {
    const line = String(raw).replace(/\s+/g, " ").trim(), m = line.match(anyDate)
    if (!m) continue
    const rest = (line.slice(0, m.index) + " " + line.slice(m.index + m[0].length)).trim()
    // headings and summary lines ("Loan Start Date 21/03/2025 … EMI Amount 53,783.00"), not transactions
    if (/\bdate\b|\bfrom\b.*\bto\b|statement|tenure|tenor|maturity|rate of interest/i.test(rest)) continue
    if ((line.match(new RegExp(LD_DATE_RE, "g")) || []).length > 2) continue
    if (/\b(due|demand|raised|bounce|bounced|return(ed)?|dishono(u)?r|charges?|fee|penal|gst|disburs\w*|reversal|reversed|cancel\w*|refund|foreclos\w*|closure|insurance|interest (applied|debited|charged|accrued))\b/i.test(rest)) continue
    // "Part Payment applied" and "Payment Adjustment" only book a payment already listed as received
    if (/\bapplied\b|\badjust(ment|ed)?\b/i.test(rest)) continue
    const prepayWords = /part[\s-]*(pre)?pay|pre[\s-]?payment|prepaid/i.test(rest)
    if (!prepayWords && !/\b(emi|instal{1,2}ment|repayment|receipt|received|recd|rcvd|paid|payment|e?nach|ach|ecs|si|auto[\s-]?debit|mandate|cheque|chq|neft|imps|rtgs|upi|collection)\b/i.test(rest)) continue
    const amts = (rest.match(/\b\d{1,3}(?:,\d{2,3})+(?:\.\d{1,2})?\b|\b\d+\.\d{2}\b/g) || []).map(ldNum).filter(n => n > 0)
    if (!amts.length) continue
    let amount = emi && !prepayWords ? amts.find(n => Math.abs(n - emi) <= emi * 0.01) : null
    if (amount == null) amount = amts[0]
    if (!(amount >= 100)) continue
    const prepay = prepayWords || (emi > 0 && amount > emi * 3)
    const p = { day: ldDay(m[1]), amount, kind: prepay ? "prepay" : "emi" }
    const k = amts.indexOf(amount)
    if (amts.length >= k + 3 && Math.abs(amts[k + 1] + amts[k + 2] - amount) <= 2) { p.principal = amts[k + 1]; p.interest = amts[k + 2] }
    const dd = d => Date.parse(d + "T00:00:00Z") / 864e5
    if (p.day && !out.some(o => Math.abs(o.amount - p.amount) < 1 && Math.abs(dd(o.day) - dd(p.day)) <= 3)) out.push(p)
  }
  // no EMI printed (a top-up after a part payment, say): a payment far above the usual one is a part payment
  if (!emi && out.length >= 3) { const mid = out.map(o => o.amount).sort((a, b) => a - b)[Math.floor(out.length / 2)]; for (const o of out) if (o.amount > mid * 3) o.kind = "prepay" }
  return out.slice(-36)
}

// The statement's payments that aren't in CC Transactions yet (no debit of that amount within 4 days), as rows to add.
// have: [{ day, amount, type, messageId }]. The rows are marked Ignored, like lender receipts: loan payments, not expenses to log.

function docPaymentRows(paid, have, account, lenderName) {
  const tag = account ? ` · A/c …${String(account).slice(-4)}` : "", rows = [], seen = new Set(have.map(h => h.messageId))
  const dn = d => Date.parse(String(d).slice(0, 10) + "T00:00:00Z") / 864e5
  for (const p of paid || []) {
    const id = `doc|${account || lenderName}|${p.day}|${p.amount}`
    if (seen.has(id)) continue
    if (have.some(h => (h.type || "Spend") === "Spend" && Math.abs(h.amount - p.amount) < 1 && Math.abs(dn(h.day) - dn(p.day)) <= 4)) continue
    rows.push({ messageId: id, day: p.day, amount: p.amount, merchant: `${lenderName || "Loan"} ${p.kind === "prepay" ? "part payment" : "EMI"}${tag}`,
      channel: p.kind === "prepay" ? "Other bank debit" : "EMI / auto-debit" })
    seen.add(id)
  }
  return rows
}


// ---------- loans: read lender PDFs (from their emails, or ones you send with the "Send loan doc" shortcut) ----------
// Each PDF is turned into text by Google Drive (scans too), read, and the copy is deleted straight away.
// Only the figures are saved, to the loan's "Doc figures" in Notion, for you to confirm on the dashboard.
// Password-protected PDFs can't be read by Drive; they're listed so you can send an unlocked copy.
const DOC_TAG = "[LOAN-DOC]"

// Run this once after pasting the code, so Google asks for the new Drive permission.
function authorizeDocs() { DriveApp.getRootFolder(); Logger.log("Drive access OK. Loan PDFs will be read from now on.") }

function pdfToText(blob) {
  const token = ScriptApp.getOAuthToken(), boundary = "loandoc" + Date.now()
  const meta = JSON.stringify({ name: "loan-doc-temp-" + Date.now(), mimeType: "application/vnd.google-apps.document" })
  const head = `--${boundary}\r\nContent-Type: application/json; charset=UTF-8\r\n\r\n${meta}\r\n--${boundary}\r\nContent-Type: application/pdf\r\n\r\n`
  const payload = Utilities.newBlob(head).getBytes().concat(blob.getBytes()).concat(Utilities.newBlob(`\r\n--${boundary}--`).getBytes())
  const up = UrlFetchApp.fetch("https://www.googleapis.com/upload/drive/v3/files?uploadType=multipart&ocrLanguage=en",
    { method: "post", contentType: `multipart/related; boundary=${boundary}`, payload, headers: { Authorization: "Bearer " + token }, muteHttpExceptions: true })
  if (up.getResponseCode() >= 300) {
    const msg = up.getContentText()
    return { locked: /encrypt|password|protected|cannot be converted|conversion/i.test(msg), error: msg.slice(0, 200) }
  }
  const id = JSON.parse(up.getContentText()).id
  try {
    const ex = UrlFetchApp.fetch(`https://www.googleapis.com/drive/v3/files/${id}/export?mimeType=text/plain`, { headers: { Authorization: "Bearer " + token }, muteHttpExceptions: true })
    const text = ex.getResponseCode() < 300 ? ex.getContentText() : ""
    return text.trim().length < 40 ? { locked: true, error: "no text" } : { text }
  } finally {
    UrlFetchApp.fetch(`https://www.googleapis.com/drive/v3/files/${id}`, { method: "delete", headers: { Authorization: "Bearer " + token }, muteHttpExceptions: true })
  }
}

function syncLoanDocs(token) {
  if (!LOANS_DB) return { read: 0, locked: 0 }
  const props = PropertiesService.getScriptProperties(), now = Date.now()
  if (props.getProperty("DOCS_OFF") === "1") return { read: 0, locked: 0 }
  const done = new Set(JSON.parse(props.getProperty("DOCS_SEEN") || "[]"))
  const first = !props.getProperty("DOCS_AT")
  const lenders = LENDERS.map(x => x[2].split(/\s+/)[0]).filter((x, i, a) => a.indexOf(x) === i).join(" OR ")
  const q = `has:attachment filename:pdf newer_than:${first ? 400 : 4}d (subject:"${DOC_TAG}" OR ((loan OR EMI OR repayment OR "interest certificate" OR foreclosure OR sanction) (${lenders}) -"credit card" -category:promotions))`
  const rows = loadDocRows(token)
  let read = 0, locked = 0, budget = 6   // PDFs per run; the rest wait for the next run
  for (const t of GmailApp.search(q, 0, first ? 60 : 20)) for (const m of t.getMessages()) {
    const from = m.getFrom(), subject = m.getSubject(), mailDay = istDay(m.getDate().getTime())
    for (const att of m.getAttachments({ includeInlineImages: false })) {
      if (!/\.pdf$/i.test(att.getName()) && att.getContentType() !== "application/pdf") continue
      const key = m.getId() + "|" + att.getName()
      if (done.has(key)) continue
      if (budget-- <= 0) break
      done.add(key)
      let out
      try { out = pdfToText(att.copyBlob()) } catch (e) { out = { error: String(e.message || e) } }
      const lender = lenderOf(from, subject + " " + (out.text || "").slice(0, 600))
      const doc = out.text ? parseLoanDoc(out.text, { filename: att.getName(), subject, from, mailDay }) : null
      const name = att.getName().replace(/^.*\//, "").slice(0, 60)
      let entry = null
      if (doc) { entry = Object.assign({ file: name, lender: lender && lender.name }, doc); read++ }
      else if (out.locked && (subject.includes(DOC_TAG) || /loan|\bemi\b|repayment|interest certificate|foreclos|sanction|statement of account|\bsoa\b/i.test(subject + " " + name))) { entry = { file: name, lender: lender && lender.name, locked: true, mailDay }; locked++ }
      if (!entry) continue
      attachDoc(token, rows, entry, name, lender)
    }
  }
  props.setProperty("DOCS_SEEN", JSON.stringify([...done].slice(-400)))
  if (budget > 0) props.setProperty("DOCS_AT", String(now))   // finished this batch; until then the year-long look-back continues
  return { read, locked }
}

function loadDocRows(token) {
  return queryAll(token, LOANS_DB, {}).map(pg => { const p = pg.properties
    return { id: pg.id, name: plain(p.Loan), match: plain(p["Lender match"]) || plain(p.Loan), account: plain(p.Account).toUpperCase(), emi: (p.EMI || {}).number || 0,
      live: !!(p.Active && p.Active.checkbox) || !!(p.Closed && p.Closed.date) || !!(p.Suggested && p.Suggested.checkbox),
      docs: (() => { try { return JSON.parse(plain(p["Doc figures"]) || "[]") } catch (e) { return [] } })() } })
}
// Saves a document's figures on its loan: by account number, else the only loan with this lender. Returns the loan's name, or "".
function attachDoc(token, rows, entry, fileName, lender) {
  const acct = entry.account || (String(fileName).match(/([A-Z]{0,6}\d[A-Z0-9]{5,19})/) || [])[1] || ""
  let row = acct && rows.find(r => r.live && r.account && (r.account === acct.toUpperCase() || r.account.endsWith(acct.toUpperCase().slice(-6))))
  if (!row && lender) { const mine = rows.filter(r => r.live && wordsOfLoan(r.match).every(w => (" " + wordsOfLoan(lender.name + " " + lender.match).join(" ") + " ").includes(" " + w))); if (mine.length === 1) row = mine[0] }
  if (!row) return ""
  // EMIs and part payments the statement lists but no alert caught: added to CC Transactions, so the dashboard counts them
  if (entry.paid && entry.paid.length) {
    try { entry.paidAdded = saveDocPayments(token, entry.paid, row.account || entry.account || acct, (lender && lender.name) || row.match) || undefined } catch (e) {}
    entry.paidCount = entry.paid.length; delete entry.paid
  }
  row.docs = row.docs.filter(d => !(d.file === entry.file && d.mailDay === entry.mailDay)).concat([entry]).sort((a, b) => (a.asOf || a.mailDay || "") < (b.asOf || b.mailDay || "") ? -1 : 1).slice(-4)
  notion(token, "patch", `pages/${row.id}`, { properties: { "Doc figures": { rich_text: [{ text: { content: JSON.stringify(row.docs).slice(0, 1990) } }] } } })
  return row.name
}

function saveDocPayments(token, paid, account, lenderName) {
  const since = shiftDay(istDay(Date.now()), -460)
  const have = queryAll(token, ALERTS_DB, { filter: { and: [{ property: "Date", date: { on_or_after: since } },
    { or: [{ property: "Category", select: { equals: "Loan EMI" } }, { property: "Channel", select: { equals: "EMI / auto-debit" } }, { property: "Channel", select: { equals: "Other bank debit" } }] }] } })
    .map(pg => { const q = pg.properties, s = q.Date.date ? q.Date.date.start : pg.created_time
      return { messageId: plain(q["Message ID"]), type: sel(q.Type) || "Spend", amount: q.Amount.number || 0, day: istDay(Date.parse(s.length === 10 ? s + "T12:00:00+05:30" : s)), status: sel(q.Status) } })
    .filter(h => h.status !== "Duplicate")
  const rows = docPaymentRows(paid, have, account, lenderName)
  for (const r of rows) {
    const time = Date.parse(r.day + "T10:00:00+05:30")
    createRow(token, { messageId: r.messageId, source: "Statement", status: "Parsed", amount: r.amount, card: "Bank account", type: "Spend", channel: r.channel, category: "Loan EMI",
      merchant: r.merchant, time, iso: istIso(time), ignored: true, raw: "From a loan statement PDF" })
  }
  return rows.length
}

// One-time setup (on a computer): Deploy › New deployment › Web app, Execute as: Me, Who has access: Anyone › Deploy.
// Then run uploadLink() and paste the link it prints into the dashboard (Loans tab › Add a loan document).
// The link carries a secret key; anyone with the full link could send PDFs to this script, so keep it private.
function uploadLink() {
  const props = PropertiesService.getScriptProperties()
  let key = props.getProperty("UPLOAD_KEY")
  if (!key) { key = Utilities.getUuid().replace(/-/g, ""); props.setProperty("UPLOAD_KEY", key) }
  const url = ScriptApp.getService().getUrl()
  if (!url) { Logger.log("Deploy this script as a web app first: Deploy › New deployment › Web app (Execute as: Me, Who has access: Anyone)."); return }
  const link = url + "?k=" + key
  Logger.log("Paste this link into the dashboard:\n" + link)
  // also email it to you, so it's on your phone to copy
  try { MailApp.sendEmail(Session.getEffectiveUser().getEmail(), "Your loan upload link", `Copy this link on your iPhone, then in the dashboard tap Loans › Add a loan document › Paste upload link:\n\n${link}\n\nKeep it private: anyone with it could send files to your script. (Sent by your card-alerts Google Apps Script.)`) } catch (e) {}
  return link
}
// Makes a new key, so an old link stops working. Run uploadLink() again afterwards.
function resetUploadKey() { PropertiesService.getScriptProperties().deleteProperty("UPLOAD_KEY"); Logger.log("Key reset. Run uploadLink() for a new link.") }

function doPost(e) {
  const out = o => ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON)
  const props = PropertiesService.getScriptProperties()
  const key = props.getProperty("UPLOAD_KEY")
  if (!key || !e || !e.parameter || e.parameter.k !== key) return out({ ok: false, error: "This upload link isn't valid. Run uploadLink() in the script and paste the new link." })
  let body
  try { body = JSON.parse(e.postData.contents) } catch (x) { return out({ ok: false, error: "Couldn't read the upload." }) }
  const token = props.getProperty("NOTION_TOKEN")
  // insurance scans from the dashboard: read here, saved on the policy by the dashboard
  if (body.kind === "policy") {
    const results = []
    for (const f of (body.files || []).slice(0, 5)) {
      const name = String(f.name || "document.pdf").slice(0, 80)
      try {
        const t = pdfToText(Utilities.newBlob(Utilities.base64Decode(f.b64), "application/pdf", name))
        if (!t.text) { results.push({ name, locked: !!t.locked, error: t.locked ? "" : "Google couldn't read this PDF." }); continue }
        const figures = parsePolicyDoc(t.text, { filename: name })
        results.push(figures ? { name, figures } : { name, error: "No policy details found in it." })
      } catch (x) { results.push({ name, error: String(x.message || x).slice(0, 120) }) }
    }
    return out({ ok: true, results })
  }
  if (!token || !LOANS_DB) return out({ ok: false, error: "The sync isn't set up yet." })
  const rows = loadDocRows(token), today = istDay(Date.now()), results = []
  for (const f of (body.files || []).slice(0, 5)) {
    const name = String(f.name || "document.pdf").slice(0, 80)
    try {
      const blob = Utilities.newBlob(Utilities.base64Decode(f.b64), "application/pdf", name)
      const t = pdfToText(blob)
      if (!t.text) { results.push({ name, locked: !!t.locked, error: t.locked ? "" : "Google couldn't read this PDF." }); continue }
      const lender = lenderOf("", t.text.slice(0, 1500))
      const doc = parseLoanDoc(t.text, { filename: name, subject: "", from: "", mailDay: today })
      if (!doc) { results.push({ name, error: "No loan figures found in it." }); continue }
      const entry = Object.assign({ file: name, lender: lender && lender.name }, doc)
      const loan = attachDoc(token, rows, entry, name, lender)
      results.push({ name, kind: doc.kind, lender: lender && lender.name, loan, figures: Object.assign({}, doc, { paid: undefined, paidCount: (doc.paid || []).length || undefined, paidAdded: entry.paidAdded }), entry: loan ? null : Object.assign({}, entry, { paid: undefined }), unmatched: !loan })
    } catch (x) { results.push({ name, error: String(x.message || x).slice(0, 120) }) }
  }
  return out({ ok: true, results })
}

// ---------- duplicates ----------
// The same payment often arrives twice: the bank's email and the forwarded SMS, or the bank debit
// plus the card company's "payment received". Keep one.
function isDuplicate(r, recent) {
  const MIN = 6e4
  return recent.find(x => {
    if (x.messageId && x.messageId === r.messageId) return true
    const dt = Math.abs(x.time - r.time)
    if (r.type === "Bill payment") {
      if (x.type !== "Bill payment") return false
      if (Math.abs(x.amount - r.amount) <= 1 && dt <= 10 * MIN) return true                       // SMS sent twice
      return Math.abs(x.amount - r.amount) <= 50 && dt <= 3 * 1440 * MIN && x.merchant !== r.merchant &&
        (x.card === r.card || x.card === "Unknown" || r.card === "Unknown")                          // bank debit + issuer/CRED confirmation
    }
    if (x.type !== r.type || Math.abs(x.amount - r.amount) > 0.01) return false
    // A card left "Unknown" (an SMS format we can't place) still matches the same payment on a known card
    if (x.card !== r.card && !((x.card === "Unknown" || r.card === "Unknown") && sameMerchant(x.merchant, r.merchant))) return false
    if (x.source !== r.source) return dt <= 30 * MIN ||                                           // email + SMS of one payment
      (r.amount >= 1000 && dt <= 18 * 60 * MIN && istDay(x.time) === istDay(r.time) && sameMerchant(x.merchant, r.merchant))   // a late bank email, same day
    return x.source === "SMS" && dt <= 10 * MIN                                                    // one SMS forwarded 2–3 times
  })
}

function sameMerchant(a, b) {
  const k = s => String(s || "").toLowerCase().replace(/[^a-z0-9]/g, "")
  const x = k(a), y = k(b), n = Math.min(x.length, y.length, 8)
  return n >= 4 && x.slice(0, n) === y.slice(0, n)
}

// ---------- parsing (plain JavaScript, no Google services; tested on real alerts) ----------

const MON3 = { jan: 0, feb: 1, mar: 2, apr: 3, may: 4, jun: 5, jul: 6, aug: 7, sep: 8, oct: 9, nov: 10, dec: 11 }
const num = s => Number(String(s).replace(/,/g, "").replace(/\.$/, ""))
const pad = n => String(n).padStart(2, "0")
const titleCase = s => s.toLowerCase().replace(/\b[a-z]/g, c => c.toUpperCase())

// IST wall-clock parts → epoch ms
function istTime(y, mo, d, h = 12, mi = 0, s = 0) { return Date.UTC(y, mo, d, h, mi, s) - 330 * 6e4 }
function istIso(ms) {
  const d = new Date(ms + 330 * 6e4)
  return `${d.getUTCFullYear()}-${pad(d.getUTCMonth() + 1)}-${pad(d.getUTCDate())}T${pad(d.getUTCHours())}:${pad(d.getUTCMinutes())}:00+05:30`
}
function hms(t) {
  const m = String(t || "").match(/(\d{1,2}):(\d{2})(?::(\d{2}))?\s*([AP]M)?/i)
  if (!m) return null
  let h = Number(m[1])
  if (m[4]) { const pm = /p/i.test(m[4]); if (h === 12) h = pm ? 12 : 0; else if (pm) h += 12 }
  return [h, Number(m[2]), Number(m[3] || 0)]
}
// Accepts "28 Sep, 2026", "02-10-2026", "02-10-26", "2026-07-23", "02/10/26", "28 September 2026".
function parseDay(s) {
  s = String(s || "").trim()
  let m = s.match(/^(\d{4})-(\d{2})-(\d{2})/)
  if (m) return [+m[1], +m[2] - 1, +m[3]]
  m = s.match(/^(\d{1,2})[-\/](\d{1,2})[-\/](\d{2,4})/)
  if (m) return [m[3].length === 2 ? 2000 + +m[3] : +m[3], +m[2] - 1, +m[1]]
  m = s.match(/^(\d{1,2})[\s-]+([A-Za-z]{3})[A-Za-z]*[,\s-]+(\d{2,4})/)
  if (m && MON3[m[2].toLowerCase()] != null) return [m[3].length === 2 ? 2000 + +m[3] : +m[3], MON3[m[2].toLowerCase()], +m[1]]
  return null
}
// Text date (+ optional time) → ms; date-only uses the email's own time when it's the same day.
function when(dayStr, timeStr, fallbackMs) {
  const d = parseDay(dayStr)
  if (!d) return fallbackMs
  const t = hms(timeStr)
  if (t) return istTime(d[0], d[1], d[2], t[0], t[1], t[2])
  const f = new Date(fallbackMs + 330 * 6e4)
  if (f.getUTCFullYear() === d[0] && f.getUTCMonth() === d[1] && f.getUTCDate() === d[2]) return fallbackMs
  return istTime(d[0], d[1], d[2], 12)
}

const hasDigits = (text, digits) => new RegExp(`(?:^|[^\\d])${digits}(?!\\d)`).test(text)
function cardOf(text, digitsHint) {
  if (digitsHint) {
    const byDigits = CARDS.find(c => c.digits.some(d => d.endsWith(digitsHint) || digitsHint.endsWith(d)))
    if (byDigits) return byDigits.name
  }
  for (const c of CARDS) if (c.words.test(text) && (c.digits.length === 0 || !CARDS.some(o => o !== c && o.digits.some(d => hasDigits(text, d))) || c.digits.some(d => hasDigits(text, d)))) return c.name
  for (const c of CARDS) if (c.digits.some(d => hasDigits(text, d))) return c.name
  return "Unknown"
}
function cardFromName(name) {
  if (/american express|\bamex\b|aebc/i.test(name)) return cardByBank("amex")
  for (const [test] of BANK_WORDS) if (test.test(name)) { const c = MY_CARDS.find(c => test.test(c.name) && !/rupay/i.test(c.name)); if (c) return c.name }
  return "Unknown"
}

function cleanMerchant(s) {
  return String(s || "")
    .replace(/\s+(What.s more\?.*|You can convert this.*|Not you\??|Want to.*|Need help.*|Avl\.?\s*(Bal|Limit).*|Available.*|Info:.*|Ref(?:\s*No)?\.?.*|UPI transaction.*|For more.*)$/i, "")
    .replace(/[\s.,:;-]+$/, "").replace(/\s+/g, " ").trim().slice(0, 80)
}
function stripBalances(s) {
  return String(s || "")
    .replace(/(avl\.?|available|avail\.?|total|current)?\s*(bal(ance)?|limit|credit limit|outstanding)\s*[:\-]?\s*(is\s*)?(Rs\.?|INR|₹)?\s*[\d,]+(\.\d+)?/gi, "[hidden]")
    .replace(/\b\d{4,8}\b(?=[^.]{0,30}\b(otp|one[- ]time))/gi, "[otp]")
}

function guessCategory(merchant) {
  if (FAMILY_NAMES.test(merchant)) return "Family transfer"
  if (/\blic\b|life insurance corp/i.test(merchant)) return "LIC"
  if (/policybazaar|star health|hdfc ergo|niva bupa|care health|icici lombard|aditya birla health|bajaj allianz/i.test(merchant)) return "Health insurance"
  if (/bajaj housing|bajaj fin|home loan|\bemi\b|loan|lic housing|pnb housing|tata capital|aditya birla fin|idfc first|hdfc ltd|hdfc limited|icici home|axis finance|kotak mahindra prime|l ?& ?t finance|cholamandalam|shriram fin|muthoot|mahindra fin|home credit|piramal|aavas|indiabulls|poonawalla|hero fincorp/i.test(merchant)) return "Loan EMI"
  for (const [cat, re] of CATEGORY_RULES) if (re.test(merchant)) return cat
  return /^[a-z]+(\s+[a-z]+){1,3}$/i.test(merchant.trim()) && !/\b(ltd|pvt|private|limited|store|stores|mart|services|enterprises|traders|trading|co)\b/i.test(merchant) ? "Personal" : "Other"
}

// Returns a transaction, or { skip: reason }.
function parseAlert(msg) {
  const subject = String(msg.subject || "")
  const from = String(msg.from || "")
  const isSms = /\[CC-SMS\]/i.test(subject)
  const body = String(msg.body || "").replace(/&amp;/g, "&").replace(/&#39;/g, "'").replace(/\|/g, " ").replace(/\s+/g, " ").trim()
  // Currency symbols → codes, so "$53.10" reads like "USD 53.10"
  const sym = s => s.replace(/(?:US\s?\$|\$)\s?(?=\d)/g, "USD ").replace(/€\s?(?=\d)/g, "EUR ").replace(/£\s?(?=\d)/g, "GBP ")
  const text = sym(isSms ? body : subject + " · " + body)
  const emailMs = msg.date instanceof Date ? msg.date.getTime() : Date.parse(msg.date)
  const base = { messageId: msg.id, source: isSms ? "SMS" : "Email", status: "Parsed", raw: "" }
  const out = (o) => {
    const r = Object.assign({}, base, o)
    r.amount = Math.round(num(r.amount) * 100) / 100
    if (!(r.amount > 0)) return { skip: "no amount" }
    r.merchant = cleanMerchant(r.merchant) || (r.type === "Bill payment" ? "Card bill payment" : "Unknown")
    r.time = r.time || emailMs
    r.iso = istIso(r.time)
    r.category = r.category || (r.type === "Bill payment" ? "Other" : guessCategory(r.merchant))
    if (r.status === "Needs review") r.raw = stripBalances(body).slice(0, 300)
    tagOwn(r, text)
    return r
  }

  if (isSms && !body) return { skip: "empty test SMS" }
  if (isSms && /^\s*test\b/i.test(body) && !/(Rs\.?|INR|₹)\s*\d/.test(body)) return { skip: "test SMS" }
  if (/\botp\b|one[- ]time password|verification code/i.test(text) && !/spent|debited/i.test(text)) return { skip: "OTP" }
  if (/statement/i.test(subject)) return { skip: "statement" }
  if (/declined|unsuccessful|could not be processed|has failed|(txn|transaction|payment) failed/i.test(text) && !/was successful/i.test(text)) return { skip: "declined or failed" }
  if (/bill is due|payment (is )?due|is due on|bill (has been )?generated|new bill|minimum amount due|upcoming|will be debited|reminder/i.test(text) && !/has been debited|is debited|spent/i.test(text)) return { skip: "reminder" }
  if (IGNORE_BANKS.test(from + " " + (isSms ? body.slice(0, 40) : ""))) return { skip: "not your bank" }
  let m

  // ----- credit card: spends, refunds, reversals -----
  if ((m = text.match(/(?:Rs\.?|INR)\s*([\d,]+(?:\.\d+)?) (?:has been|is) debited from your HDFC Bank Credit Card ending (\d{4}) towards (.+?) on (\d{1,2} \w{3},? \d{4})(?: at (\d{1,2}:\d{2}(?::\d{2})?))?/i)))
    return out({ amount: m[1], card: cardOf(text, m[2]), merchant: m[3], time: when(m[4], m[5], emailMs), type: "Spend", channel: "Card" })
  if ((m = text.match(/Spent (Rs\.?|INR|[A-Z]{3})\s*([\d,]+(?:\.\d+)?) On HDFC Bank Card (\d{4}) At (.+?) On (\d{4}-\d{2}-\d{2}):?(\d{2}:\d{2}(?::\d{2})?)?/i)))
    return out({ amount: m[2], card: cardOf(text, m[3]), merchant: m[4], time: when(m[5], m[6], emailMs), type: "Spend", channel: "Card",
      fx: /^(Rs\.?|INR)$/i.test(m[1]) ? null : { cur: m[1].toUpperCase(), amt: num(m[2]) } })
  if ((m = text.match(/transaction reversal of Rs\.?\s*([\d,]+(?:\.\d+)?) has been initiated to your HDFC Bank Credit Card ending (\d{4}) From Merchant:\s*(.+?)\s*Date ?Time:\s*(\d{1,2} \w{3},? \d{4})(?: at (\d{1,2}:\d{2}(?::\d{2})?))?/i)))
    return out({ amount: m[1], card: cardOf(text, m[2]), merchant: m[3], time: when(m[4], m[5], emailMs), type: "Reversal", channel: "Card" })
  if ((m = text.match(/Transaction Reversed!?\s*On HDFC Bank CREDIT Card x*(\d{4}) Amt:?\s*Rs\.?\s*([\d,]+(?:\.\d+)?) By (.+?)(?: On | Not you|\.\s|$)/i)))
    return out({ amount: m[2], card: cardOf(text, m[1]), merchant: m[3], type: "Reversal", channel: "Card" })
  if ((m = text.match(/Rs\.?\s*([\d,]+(?:\.\d+)?) (?:has been )?refunded by (.+?)(?: on .*?)? (?:and )?(?:has been )?adjusted against (?:your )?HDFC Bank Credit Card (?:ending )?(\d{4})/i)))
    return out({ amount: m[1], card: cardOf(text, m[3]), merchant: m[2], type: "Refund", channel: "Card" })
  if ((m = text.match(/PAYMENT OF Rs\.?\s*([\d,]+(?:\.\d+)?) (?:HAS BEEN )?RECEIVED TOWARDS YOUR (?:HDFC BANK )?CREDIT CARD ENDING (?:WITH )?(\d{4})/i)))
    return out({ amount: m[1], card: cardOf(text, m[2]), merchant: "HDFC card payment received", type: "Bill payment", channel: "Card bill payment" })

  if ((m = text.match(/Transaction Amount:?\s*INR\s*([\d,]+(?:\.\d+)?)\s*Merchant Name:?\s*(.+?)\s*Axis Bank Credit Card No\.?\s*X*(\d{4})\s*Date\s*&\s*Time:?\s*(\d{2}-\d{2}-\d{4}),?\s*(\d{2}:\d{2}(?::\d{2})?)?/i)))
    return out({ amount: m[1], card: cardOf(text, m[3]), merchant: m[2], time: when(m[4], m[5], emailMs), type: "Spend", channel: "Card" })
  if ((m = text.match(/Spent\s*(INR|[A-Z]{3})\s*([\d,]+(?:\.\d+)?)\s*Axis Bank Card no\.?\s*X*(\d{4})\s*(\d{2}-\d{2}-\d{2,4})\s*(\d{2}:\d{2}:\d{2})?\s*(?:IST)?\s*(.+?)\s*(?:Avl|Not you|$)/i)))
    return out({ amount: m[2], card: cardOf(text, m[3]), merchant: m[6], time: when(m[4], m[5], emailMs), type: "Spend", channel: "Card",
      fx: /^INR$/i.test(m[1]) ? null : { cur: m[1].toUpperCase(), amt: num(m[2]) } })

  if (/scapia|federal/i.test(text)) {
    if ((m = text.match(/received your payment of ₹\s*([\d,]+(?:\.\d+)?)/i)))
      return out({ amount: m[1], card: cardByBank("scapia"), merchant: "Scapia repayment", type: "Bill payment", channel: "Card bill payment" })
    const amt = text.match(/Amount\s*₹\s*([\d,]+(?:\.\d+)?)/i) || text.match(/(?:txn|transaction) (?:of|for) ₹\s*([\d,]+(?:\.\d+)?)/i)
    const merch = text.match(/Merchant\s*:?\s*(.+?)(?=\s+(?:Not you|Want to|Need help|You'?ve|Regards|Check)|$)/i) || text.match(/₹\s*[\d,.]+ at (.+?) on your Scapia/i)
    const dt = text.match(/on (\d{2}-\d{2}-\d{4}) at (\d{1,2}:\d{2}\s*[AP]M)/i)
    const card = /rupay/i.test(text) ? cardByBank("scapia rupay") : cardByBank("scapia")
    const channel = card === cardByBank("scapia rupay") && /upi|vpa/i.test(text) ? "UPI on credit card" : "Card"
    if (amt) {
      const type = /reversed/i.test(text) ? "Reversal" : /refund/i.test(text) ? "Refund" : /successful|success/i.test(text) ? "Spend" : null
      if (type) return out({ amount: amt[1], card, merchant: merch ? merch[1] : "", time: dt ? when(dt[1], dt[2], emailMs) : emailMs, type, channel, status: merch ? "Parsed" : "Needs review" })
    }
  }

  if ((m = text.match(/your credit card (?:bill )?payment was successful.*?(HDFC Bank|Axis Bank|Scapia|Federal Bank|American Express|Amex|[A-Z][A-Za-z]+ Bank)\s*•+\s*(\d{4,5}).*?amount paid\s*₹\s*([\d,]+(?:\.\d+)?)/i)))
    return out({ amount: m[3], card: cardOf(m[1], m[2]) !== "Unknown" ? cardOf(m[1], m[2]) : cardFromName(m[1]), merchant: "Bill payment via CRED", type: "Bill payment", channel: "Card bill payment" })
  if ((m = text.match(/American Express Card Bill Payment was successful.*?Payment Amount \(INR\)\s*([\d,]+(?:\.\d+)?)/i)))
    return out({ amount: m[1], card: cardByBank("amex"), merchant: "Bill payment via BillDesk", type: "Bill payment", channel: "Card bill payment" })
  if (/amex|american express/i.test(text)) {
    if ((m = text.match(/payment of (?:INR|Rs\.?|₹)\s*([\d,]+(?:\.\d+)?) (?:was|has been) (?:received|credited)/i)))
      return out({ amount: m[1], card: cardByBank("amex"), merchant: "Amex payment received", type: "Bill payment", channel: "Card bill payment" })
    if ((m = text.match(/spent (INR|Rs\.?|₹|[A-Z]{3})\s*([\d,]+(?:\.\d+)?) on your AMEX card \**\s*(\d{4,5})? ?at (.+?) on (\d{1,2} \w+ \d{4})(?: at (\d{1,2}:\d{2}(?:\s*[AP]M)?))?/i))) {
      const fx = /^(INR|Rs\.?|₹)$/i.test(m[1]) ? null : { cur: m[1].toUpperCase(), amt: num(m[2]) }
      return out({ amount: m[2], card: cardByBank("amex"), merchant: m[4], time: when(m[5], m[6], emailMs), type: "Spend", channel: "Card", fx })
    }
  }

  // ----- bank account: UPI, debit card, auto-debits, charges -----
  let bank = null
  if ((m = text.match(/(?:Rs\.?|INR)\s*([\d,]+(?:\.\d+)?) is debited from your account ending (\d{3,6}) towards VPA (\S+) \((.+?)\) on (\d{2}-\d{2}-\d{2,4})/i)))
    bank = { amount: m[1], merchant: m[4], vpa: m[3], time: when(m[5], null, emailMs), channel: "UPI from bank a/c" }
  else if ((m = text.match(/Sent Rs\.?\s*([\d,]+(?:\.\d+)?)\s*From HDFC Bank A\/?C \*?x*(\d{4})\s*To (.+?)\s*On (\d{2}\/\d{2}\/\d{2,4})/i)))
    bank = { amount: m[1], merchant: m[3], time: when(m[4], null, emailMs), channel: "UPI from bank a/c" }
  else if ((m = text.match(/(?:Rs\.?|INR)\s*([\d,]+(?:\.\d+)?) (?:is|has been) debited from your HDFC Bank Debit Card ending (\d{4}) at (.+?) on (\d{1,2} \w{3},? \d{4})(?: at (\d{2}:\d{2}(?::\d{2})?))?/i)))
    bank = { amount: m[1], merchant: m[3], time: when(m[4], m[5], emailMs), channel: /e-?mandate|netflix|apple|google|spotify|youtube|prime/i.test(text) ? "EMI / auto-debit" : "Other bank debit" }
  else if (/set up through E-?mandate|successfully paid using your HDFC Bank Debit Card/i.test(text))
    return { skip: "e-mandate notice (the debit card alert is recorded instead)" }
  else if (/(debited|debit|spent|sent|paid|withdrawn)/i.test(text) && /(a\/?c|account|acct|debit card)\b/i.test(text) && !/credit card/i.test(text)) {
    const amt = text.match(/(?:Rs\.?|INR|₹)\s*([\d,]+(?:\.\d+)?)/i)
    if (!amt) return { skip: "no amount" }
    if (/credited to your (a\/?c|account)|has been credited|received (Rs|INR|₹)/i.test(text) && !/debited/i.test(text)) return { skip: "money in" }
    const who = text.match(/UPI\/P2[AM]\/\d+\/([^\/\n]+?)(?:\/|\s{2}|\s+Not you|$)/i) || text.match(/credited to (?:VPA )?(?:\S+@\S+ )?\(?([A-Za-z][A-Za-z .&'-]{2,40}?)\)? (?:via|on|UPI|Ref)/i) ||
      text.match(/\bto (?:VPA )?(?:\S+@\S+\s*)?\(?([A-Z][A-Za-z .&'-]{2,40}?)\)?\s+(?:on|via|UPI|Ref|\d)/) || text.match(/towards (.+?) on \d/i) || text.match(/;\s*([A-Z][A-Za-z .&'-]{2,40}) credited/)
    const day = text.match(/(\d{2}[-\/]\d{2}[-\/]\d{2,4}|\d{1,2} \w{3},? \d{4})/)
    const tm = text.match(/\b(\d{2}:\d{2}:\d{2})\b/)
    const channel = /ACH|NACH|ECS|e-?mandate|mandate|standing instruction|auto ?pay|autopay|\bSI\b/i.test(text) ? "EMI / auto-debit"
      : /charges?|fee\b|penalty|AMB|min(imum)? bal/i.test(text) && !/UPI/i.test(text) ? "Bank charges"
      : /cheque|chq|clg/i.test(text) ? "Cheque"
      : /UPI|VPA/i.test(text) ? "UPI from bank a/c" : "Other bank debit"
    bank = { amount: amt[1], merchant: who ? who[1] : "", time: day ? when(day[1], tm && tm[1], emailMs) : emailMs, channel, status: who ? "Parsed" : "Needs review" }
  }

  if (bank) {
    const name = cleanMerchant(bank.merchant) + " " + (bank.vpa || "")
    if (SELF_NAMES.test(name)) return { skip: "own account transfer" }
    if (/zerodha|\biccl\b|indian clearing|nse clearing|\bbse\b|mutual fund|\bsip\b|groww|kuvera|coin by|\bnps\b|\bppf\b|digital gold|recurring deposit|fixed deposit/i.test(name + " " + text.slice(0, 200))) return { skip: "investment" }
    if (/credpay|cred.?swiggy/i.test(name)) return out(Object.assign(bank, { card: "Bank account", type: "Spend", merchant: "Swiggy (via CRED UPI)", category: "Food & Dining" }))
    if (/cred\.?club|cred club|dreamplug|american express|aebc\d|\bamex\b|scapia|card ?bill|credit card bill/i.test(name))
      return out(Object.assign(bank, { card: /cred|dreamplug/i.test(name) ? "Unknown" : cardFromName(name), type: "Bill payment", channel: "Card bill payment", merchant: /cred|dreamplug/i.test(name) ? "Bill payment via CRED (bank)" : "Bill payment via UPI" }))
    if (/^\s*cred\s*$/i.test(cleanMerchant(bank.merchant))) return out(Object.assign(bank, { card: "Bank account", type: "Spend", merchant: "CRED bill pay (UPI)", category: "Bills & Utilities" }))
    if (/\blic\b|life insurance corp/i.test(name)) return out(Object.assign(bank, { card: "Bank account", type: "Spend", merchant: "LIC premium", category: "LIC" }))
    const r = Object.assign(bank, { card: "Bank account", type: "Spend" })
    r.merchant = cleanMerchant(r.merchant)
    if (r.merchant && r.merchant === r.merchant.toUpperCase() && /\s/.test(r.merchant) && !/\b(LTD|PVT|INDIA|SERVICES)\b/.test(r.merchant)) r.merchant = titleCase(r.merchant)
    if (r.channel === "Bank charges") r.category = "Fees & Charges"
    if (r.channel === "EMI / auto-debit" && /bajaj housing|loan|emi/i.test(text)) r.category = "Loan EMI"
    return out(r)
  }

  // ----- anything else that looks like a card spend: keep it, but ask you to check -----
  if (/credit card|card (no|ending|xx)/i.test(text) && /spent|debited|charged|transaction of/i.test(text) && (m = text.match(/(?:Rs\.?|INR|₹)\s*([\d,]+(?:\.\d+)?)/i)))
    return out({ amount: m[1], card: cardOf(text), merchant: "", type: "Spend", channel: "Card", status: "Needs review" })

  return { skip: "not a transaction" }
}
// ---------- recognise your cards, accounts and name from bank emails ----------
// Reads the alert emails/SMS already in your Gmail and lists the credit cards (bank + last digits),
// bank accounts, and your name as banks write it. Nothing to type in.
const BANK_NAMES = { hdfc: "HDFC", axis: "Axis", icici: "ICICI", sbi: "SBI", kotak: "Kotak", idfc: "IDFC FIRST", yes: "YES Bank", indusind: "IndusInd",
  hsbc: "HSBC", rbl: "RBL", au: "AU", federal: "Federal", amex: "Amex", "american express": "Amex", scapia: "Scapia", onecard: "OneCard", citi: "Citi", sc: "Standard Chartered", "standard chartered": "Standard Chartered" }
const bankLabel = s => { const k = String(s || "").toLowerCase().replace(/\s*bank$/, "").trim(); return BANK_NAMES[k] || (k.length <= 4 ? k.toUpperCase() : k.replace(/\b\w/g, c => c.toUpperCase())) }
function detectAccounts(msgs) {
  const cards = {}, accounts = {}, names = {}
  const addCard = (bank, digits, rupay) => {
    const b = bankLabel(bank)
    const key = b + "|" + (rupay ? "rupay" : digits)
    // the same card written with 4 or 5 digits (Amex shows 4 in some alerts, 5 in others)
    const same = Object.keys(cards).find(k => k.startsWith(b + "|") && !rupay && k.split("|")[1] !== "rupay" && (k.split("|")[1].endsWith(digits) || digits.endsWith(k.split("|")[1])))
    const c = cards[same || key] || (cards[same || key] = { bank: b, digits: rupay ? "" : digits, rupay: !!rupay, count: 0 })
    if (digits && digits.length > c.digits.length && !rupay) c.digits = digits
    c.count++
  }
  for (const m of msgs) {
    const text = (String(m.subject || "") + " · " + String(m.body || "")).replace(/&amp;/g, "&").replace(/&#39;/g, "'").replace(/\|/g, " ").replace(/\s+/g, " ")
    if (/\botp\b|one[- ]time password/i.test(text) && !/spent|debited/i.test(text)) continue
    let x
    const re = (r, f) => { for (const mm of text.matchAll(r)) f(mm) }
    re(/\b([A-Z][A-Za-z]+) Bank Credit Card (?:no\.?|number|ending(?: in| with)?|xx+|x+|\*+)?\s*[x*]*\s*(\d{4})\b/gi, mm => addCard(mm[1], mm[2]))
    re(/\b(SBI|ICICI|Kotak|IDFC FIRST|IDFC|IndusInd|RBL|AU|HSBC|Citi|OneCard|YES|HDFC|Axis) (?:Bank )?Credit Card (?:no\.?|number|ending(?: in| with)?|xx+|x+|\*+)?\s*[x*]*\s*(\d{4})\b/gi, mm => addCard(mm[1], mm[2]))
    re(/\b(?:spent|debited)[^.]{0,60}\bon ([A-Z][A-Za-z]+) Bank Card (\d{4})\b/gi, mm => addCard(mm[1], mm[2]))                       // HDFC SMS
    re(/\b([A-Z][A-Za-z]+) Bank Card no\.?\s*x*(\d{4})\b/gi, mm => { if (/spent/i.test(text) && !/debit card/i.test(text)) addCard(mm[1], mm[2]) })   // Axis SMS
    re(/\b(?:AMEX|American Express)(?: Card)?(?: card)?\s*(?:ending(?: in)?|\*+|x+|no\.?)?\s*\**\s*(\d{4,5})\b/gi, mm => addCard("amex", mm[1]))
    re(/\bScapia Federal (?:Visa )?Credit Card ending in (\d{4})\b/gi, mm => addCard("scapia", mm[1]))
    if (/Scapia Federal RuPay credit card/i.test(text)) addCard("scapia", "", true)
    re(/\bSBI Card (?:ending(?: with)?|no\.?|xx+)\s*[x*]*(\d{4})\b/gi, mm => addCard("sbi", mm[1]))
    if (/cred\.club/i.test(m.from || "")) re(/\b(HDFC Bank|Axis Bank|ICICI Bank|SBI Card|Kotak(?: Mahindra)?(?: Bank)?|IDFC First(?: Bank)?|AMEX|American Express|Yes Bank|IndusInd(?: Bank)?|RBL(?: Bank)?|AU(?: Bank)?|HSBC|Federal Bank|OneCard)\s*•+\s*(\d{4,5})\b/gi,
      mm => addCard(/amex|american/i.test(mm[1]) ? "amex" : mm[1].replace(/\s*(Bank|Card|Mahindra)\b/gi, "").trim(), mm[2]))
    re(/\b(?:account|a\/c|acct)\s*(?:no\.?|number)?\s*(?:ending(?: in| with)?)?\s*[x*]*\s*(\d{4,6})\b/gi, mm => { accounts[mm[1]] = (accounts[mm[1]] || 0) + 1 })
    re(/\bDear (?:Mr\.?|Ms\.?|Mrs\.?|Dr\.?)?\s*([A-Z][a-z]+(?: [A-Z][a-z]+){1,3})\b/g, mm => { if (!/customer|cardmember|card member|sir|madam|valued/i.test(mm[1])) names[mm[1]] = (names[mm[1]] || 0) + 1 })
    re(/\b(?:Hey|Dear),?\s+([A-Z]{2,}(?: [A-Z]{2,}){1,3})\b/g, mm => { if (!/CUSTOMER|CARDMEMBER/.test(mm[1])) { const n = mm[1].toLowerCase().replace(/\b\w/g, c => c.toUpperCase()); names[n] = (names[n] || 0) + 1 } })
  }
  const cardList = Object.values(cards).sort((a, b) => b.count - a.count)
    .map(c => ({ name: c.rupay ? `${c.bank} RuPay` : `${c.bank} ••${c.digits}`, bank: c.bank, digits: c.digits ? [c.digits] : [], seen: c.count }))
  // Your name: the fullest form seen (first, middle and last over first and last)
  const nameList = Object.entries(names).sort((a, b) => b[1] - a[1]).map(([n]) => n)
  const name = nameList.sort((a, b) => b.split(" ").length - a.split(" ").length || names[b] - names[a])[0] || ""
  const cardDigits = new Set(cardList.flatMap(c => c.digits))
  return { cards: cardList, accounts: Object.keys(accounts).filter(d => !cardDigits.has(d)).sort((a, b) => accounts[b] - accounts[a]), name }
}
// Keeps a name you already gave a card in Notion (e.g. "HDFC Regalia ••1234") when the digits match.
function mergeCardNames(found, existing) {
  return found.map(c => {
    const keep = existing.find(n => c.digits.length ? c.digits.some(d => new RegExp(`(?:^|\\D)${d.slice(-4)}\\s*$`).test(n)) : /rupay/i.test(n) && n.toLowerCase().includes(c.bank.toLowerCase()))
    return keep ? Object.assign({}, c, { name: keep }) : c
  })
}

// Reads the last 4 months of bank emails, saves your cards/name, and adds the cards to Notion's Card columns.
function detectAndSave(token) {
  const props = PropertiesService.getScriptProperties()
  const msgs = []
  for (const t of GmailApp.search(GMAIL_QUERY.replace(`newer_than:${LOOKBACK}`, "newer_than:120d") + ' OR (from:cred.club "bill summary" newer_than:120d)', 0, 250))
    for (const m of t.getMessages().slice(-10)) msgs.push({ subject: m.getSubject(), from: m.getFrom(), body: m.getPlainBody().slice(0, 3000) })
  const found = detectAccounts(msgs)
  const existing = cardOptions(token, EXPENSES_DB).map(o => o.name).filter(n => !/^example card/i.test(n))
  const merged = mergeCardNames(found.cards, existing)
  // keep cards you added by hand in Notion too
  const saved = JSON.parse(props.getProperty("MY_CARDS") || "[]")
  const gone = removedCardDigits(token)
  const all = merged.concat(saved.filter(s => !merged.some(m => m.name === s.name))).filter(c => !(c.digits || []).some(d => gone.includes(crLast4(d))))
  props.setProperty("MY_CARDS", JSON.stringify(all.map(c => ({ name: c.name, digits: c.digits }))))
  if (found.name && !props.getProperty("MY_NAME")) props.setProperty("MY_NAME", found.name)
  props.setProperty("MY_ACCOUNTS", JSON.stringify(found.accounts))
  props.setProperty("DETECTED_AT", String(Date.now()))
  const names = all.map(c => c.name)
  for (const db of [EXPENSES_DB, ALERTS_DB, BILLS_DB]) if (db) {
    try { addCardOptions(token, db, names.concat(db === ALERTS_DB ? ["Bank account", "Unknown"] : [])) } catch (e) { Logger.log("Card options: " + e) }
  }
  return { cards: all, name: props.getProperty("MY_NAME") || "", accounts: found.accounts }
}
function cardOptions(token, db) {
  const d = notion(token, "get", `databases/${db}`)
  return (d.properties && d.properties.Card && d.properties.Card.select && d.properties.Card.select.options) || []
}
// Adds the cards to a table's Card column (keeps what's there, drops the template's example).
function addCardOptions(token, db, names) {
  const cur = cardOptions(token, db)
  const keep = cur.filter(o => !/^example card/i.test(o.name))
  const add = names.filter(n => !keep.some(o => o.name === n))
  if (!add.length && keep.length === cur.length) return
  notion(token, "patch", `databases/${db}`, { properties: { Card: { select: { options: keep.map(o => ({ id: o.id, name: o.name, color: o.color })).concat(add.map(name => ({ name }))) } } } })
}

// ---------- friends & family: file transfers to and from people in your Money Diary ----------
// A transfer whose payee or sender matches a name (or UPI ID) on a person in your People table becomes a diary entry.
// Domestic help (or anyone with "Counts as spending" ticked): the payment also becomes an expense, unless you logged it already.
// Everyone else: the bank alert is marked Ignored, so it isn't flagged as a payment you forgot to log.
let PEOPLE_DB = ""    // People: found by name
let DIARY_DB = ""     // Money Diary: found by name
const HOUSEHOLD = "Household"                           // Expenses category for domestic help
const nameKey = s => String(s || "").toLowerCase().replace(/[^a-z0-9@. ]+/g, " ").replace(/\s+/g, " ").trim()
function loadPeople(token) {
  return queryAll(token, PEOPLE_DB, {}).map(pg => { const p = pg.properties
    const name = plain(p.Name), keys = [name].concat(plain(p["Names in bank alerts"]).split(/[,;\n]/)).map(nameKey).filter(k => k.length >= 3 && (k.includes("@") || k.includes(" ")))
    return { id: pg.id, name, group: sel(p.Group), spend: !!(p["Counts as spending"] && p["Counts as spending"].checkbox), keys } })
    .filter(x => x.name)
}
// The person whose name or UPI ID appears in a bank alert's payee. Longest match wins.
function personFor(people, merchant) {
  const m = " " + nameKey(merchant) + " "
  let best = null, len = 0
  for (const p of people) for (const k of p.keys) {
    if ((m.includes(" " + k + " ") || (k.includes("@") && m.includes(k))) && k.length > len) { best = p; len = k.length }
  }
  return best
}
function syncPeople(token) {
  if (!PEOPLE_DB || !DIARY_DB) return { filed: 0 }
  const people = loadPeople(token)
  if (!people.length) return { filed: 0 }
  const since = shiftDay(istDay(Date.now()), -CHECK_DAYS)
  const filed = new Set()
  for (const pg of queryAll(token, DIARY_DB, { filter: { property: "Date", date: { on_or_after: shiftDay(since, -7) } } })) for (const id of rel(pg.properties.Alert)) filed.add(id.replace(/-/g, ""))
  const alerts = queryAll(token, ALERTS_DB, { filter: { and: [{ property: "Date", date: { on_or_after: since } }, { property: "Card", select: { equals: "Bank account" } },
    { or: [{ property: "Type", select: { equals: "Spend" } }, { property: "Type", select: { equals: "Received" } }] }] } })
  let n = 0
  for (const pg of alerts) {
    if (n >= 40) break
    const p = pg.properties, id = pg.id.replace(/-/g, "")
    if (filed.has(id) || sel(p.Status) === "Duplicate" || /EMI|charges|bill/i.test(sel(p.Channel))) continue
    const who = personFor(people, plain(p.Merchant))
    if (!who) continue
    const type = sel(p.Type) === "Received" ? "Received" : "Paid", amount = p.Amount.number || 0
    const start = p.Date.date ? p.Date.date.start : pg.created_time
    const day = start.length === 10 ? start : istDay(Date.parse(start))
    let expense = rel(p.Expenses)[0] || ""
    if (type === "Paid" && who.spend && !expense) {
      expense = notion(token, "post", "pages", { parent: { database_id: EXPENSES_DB }, properties: {
        "Category Text": { title: [{ text: { content: HOUSEHOLD } }] }, Category: { select: { name: HOUSEHOLD } }, Amount: { number: amount },
        Date: { date: { start: day } }, Merchant: { rich_text: [{ text: { content: who.name } }] }, "Payment Method": { select: { name: "UPI" } },
        Notes: { rich_text: [{ text: { content: "From Money Diary" } }] }, Alerts: { relation: [{ id: pg.id }] } } }).id
    }
    const props = {
      Entry: { title: [{ text: { content: `${type} · ${who.name}` } }] }, Person: { relation: [{ id: who.id }] }, Date: { date: { start: day } },
      Type: { select: { name: type } }, Kind: { select: { name: who.group === "Domestic help" && type === "Paid" ? "Pay" : "Other" } }, Amount: { number: amount },
      Source: { select: { name: "Bank alert" } }, Alert: { relation: [{ id: pg.id }] },
    }
    if (expense) props.Expense = { relation: [{ id: expense }] }
    notion(token, "post", "pages", { parent: { database_id: DIARY_DB }, properties: props })
    if (type === "Paid" && !who.spend && !(p.Ignored && p.Ignored.checkbox)) notion(token, "patch", `pages/${pg.id}`, { properties: { Ignored: { checkbox: true } } })
    filed.add(id); n++
  }
  return { filed: n }
}

const PM_MONTHS = { jan: 1, feb: 2, mar: 3, apr: 4, may: 5, jun: 6, jul: 7, aug: 8, sep: 9, sept: 9, oct: 10, nov: 11, dec: 12 }
function pmDay(s) {
  s = String(s || "").trim()
  let m = s.match(/^(\d{1,2})[-\/. ](\d{1,2})[-\/. ](\d{4})$/)
  if (m) return `${m[3]}-${m[2].padStart(2, "0")}-${m[1].padStart(2, "0")}`
  const mon = w => PM_MONTHS[String(w).slice(0, 3).toLowerCase()]
  m = s.match(/^(\d{1,2})(?:st|nd|rd|th)?[-\/ ]([A-Za-z]{3,9})[-\/ ,]+(\d{4})$/)
  if (m && mon(m[2])) return `${m[3]}-${String(mon(m[2])).padStart(2, "0")}-${m[1].padStart(2, "0")}`
  m = s.match(/^([A-Za-z]{3,9})\s+(\d{1,2}),?\s+(\d{4})$/)
  if (m && PM_MONTHS[m[1].slice(0, 3).toLowerCase()]) return `${m[3]}-${String(PM_MONTHS[m[1].slice(0, 3).toLowerCase()]).padStart(2, "0")}-${m[2].padStart(2, "0")}`
  return ""
}
const PM_DATE = "(\\d{1,2}[-\\/.]\\d{1,2}[-\\/.]\\d{4}|\\d{1,2}(?:st|nd|rd|th)?[-\\/ ][A-Za-z]{3,9}[-\\/ ,]+\\d{4}|[A-Za-z]{3,9}\\s+\\d{1,2},?\\s+\\d{4})"
const pmAmt = s => Number(String(s || "").replace(/,/g, "")) || 0
function policyTypeOf(text, kind) {
  if (/griha|home shield|home insurance|householder|home (and|&) contents|dwelling/i.test(text)) return "Home"
  if (/motor|private car|two[- ]wheeler|\bcar insurance|\bbike insurance/i.test(text)) return "Motor"
  if (/(?<!long\s)\bterm\s*(life|plan|insurance|assurance)\b|sampoorna raksha|iprotect|click 2 protect|smart shield/i.test(text)) return "Term life"
  if (/health|mediclaim|optima|medicare|hospital|reassure|activ (one|fit|health)|family floater|super top[- ]?up|top[- ]?up/i.test(text)) return "Health"
  if (kind === "health") return "Health"
  if (kind === "life") return "Life"
  return "Other"
}
// One email → what it says about one policy, or null
function parsePolicyMail(msg) {
  const body = String(msg.body || "").replace(/ /g, " ").replace(/[ \t]+/g, " ")
  const text = `${msg.subject}\n${body}`
  const ins = INSURERS.find(([re]) => re.test(msg.from)) || INSURERS.find(([re]) => re.test(text))
  if (!ins) return null
  // marketing (new fund offers, "buy a plan") carries no policy number: skipped
  const noM = text.match(/policy\s*(?:no\.?|number|#)\s*[:.\-]?\s*([A-Z]{0,4}[\/-]?\d[\dA-Z\/-]{5,30})/i) || text.match(/\bpolicy\s+([A-Z]{0,4}\d{6,20})\b/i)
  // LIC's receipts (via BillDesk) and bill reminders (CRED, banks) carry no policy number: the premium amount stands in for it
  const lic = ins[1] === "LIC" && /receipt amount|amount due|premium|payment/i.test(text)
  if (!noM && !lic) return null
  const number = noM ? noM[1].replace(/[^A-Z0-9]/gi, "").toUpperCase() : ""
  const out = { insurer: ins[1], broker: ins[2] === "broker" ? ((msg.from + " " + text).match(/policybazaar|insurancedekho|ditto|turtlemint|beshak/i) || [""])[0].replace(/^\w/, c => c.toUpperCase()) : "", last4: number.slice(-4), number, day: istDay(msg.date.getTime ? msg.date.getTime() : msg.date), type: policyTypeOf(text, ins[2]) }
  const name = (text.match(/policy name\s*[:\-]?\s*([A-Za-z][^\n|]{2,60}?)\s*(?:\n|\||policy|$)/i) || [])[1]
    || (text.match(/your\s+(?:HDFC ERGO\s+)*(?:HDFC ERGO\s*-\s*)?([A-Z][A-Z0-9 &+\-]{4,70}?)\s+policy\s+(?:with\s+)?(?:policy\s+)?no/i) || [])[1]
    || (text.match(/your\s+([A-Z][A-Za-z0-9 &+]{3,40}?(?:insurance|plan|policy))\b/) || [])[1]
  if (name) out.name = name.replace(/\s+/g, " ").replace(/\s*-\s*$/, "").replace(/\s+policy$/i, "").trim()
  const due = text.match(new RegExp(`(?:premium\\s+due\\s+date|next\\s+(?:premium\\s+)?due\\s+date|due\\s+date|due\\s+on|renewal\\s+(?:due\\s+)?date|expir(?:y|es|ing)\\s+(?:date|on)|valid\\s+(?:till|upto|until))\\s*(?:is|:|-)?\\s*${PM_DATE}`, "i"))
  if (due) out.due = pmDay(due[1])
  const fr = text.match(/(?:payment\s+frequency|premium\s+(?:payment\s+)?(?:frequency|mode)|mode\s+of\s+payment|payment\s+mode)\s*[:\-]?\s*(monthly|quarterly|half[- ]?yearly|semi[- ]annual|annual(?:ly)?|yearly|single)/i)
  if (fr) out.freq = /month/i.test(fr[1]) ? "Monthly" : /quarter/i.test(fr[1]) ? "Quarterly" : /half|semi/i.test(fr[1]) ? "Half-yearly" : /single/i.test(fr[1]) ? "Single" : "Yearly"
  const paid = text.match(new RegExp(`received\\s+(?:your\\s+|the\\s+)?(?:renewal\\s+)?premium\\s+(?:payment\\s+)?(?:of\\s+)?(?:₹|rs\\.?|inr)\\s*([\\d,]+(?:\\.\\d+)?)[^\\n]{0,80}?\\bon\\s+${PM_DATE}`, "i"))
    || text.match(new RegExp(`premium\\s+(?:of\\s+)?(?:₹|rs\\.?|inr)\\s*([\\d,]+(?:\\.\\d+)?)\\s+(?:has\\s+been\\s+|was\\s+)?(?:received|paid|debited)[^\\n]{0,60}?\\bon\\s+${PM_DATE}`, "i"))
  if (paid) { out.paidAmount = pmAmt(paid[1]); out.paidDay = pmDay(paid[2]) }
  const bd = text.match(/receipt amount\s*:?\s*(?:₹|rs\.?|inr)?\s*([\d,]+(?:\.\d+)?)/i), bdDay = text.match(/payment date\s*:?\s*(\d{1,2}[-\/]\d{1,2}[-\/]\d{4})/i)
  if (bd && !out.paidDay) { out.paidAmount = pmAmt(bd[1]); out.paidDay = bdDay ? pmDay(bdDay[1]) : out.day }
  const owed = text.match(/(?:total\s+)?amount\s+due\s*(?:is|:)?\s*(?:₹|rs\.?|inr)\s*([\d,]+(?:\.\d+)?)/i)
  if (owed && !out.premium) out.premium = pmAmt(owed[1])
  if (!number) { const amt = out.paidAmount || out.premium || 0; if (!amt) return null; out.last4 = ""; out.amountKey = Math.round(amt / 100) }
  const prem = text.match(/(?:renewal\s+premium|premium\s+(?:amount|payable|due)|total\s+premium|installment\s+premium)\s*(?:is|of|:|-)?\s*(?:₹|rs\.?|inr)\s*([\d,]+(?:\.\d+)?)/i)
  if (prem) out.premium = pmAmt(prem[1])
  const cover = text.match(/(?:sum\s+(?:insured|assured)|cover(?:age)?\s+amount|life\s+cover)\s*(?:of|:|-)?\s*(?:₹|rs\.?|inr)\s*([\d,]+)/i)
  if (cover && pmAmt(cover[1]) >= 10000) out.cover = pmAmt(cover[1])
  if (/standing instruction|auto[- ]?debit|auto[- ]?pay|e-?mandate|\bnach\b|\bsi\b registered|autopay/i.test(text)) out.autopay = true
  if (!out.due && !out.paidDay && !out.premium && !out.cover && !out.name) return null
  return out
}
// All emails about one policy, newest facts winning
function policyMailSummary(found) {
  const by = {}
  const insurerOf = {}   // a broker's email (Policybazaar) names no insurer: it joins the insurer's emails for the same policy number
  for (const f of found) if (f.insurer) insurerOf[f.last4] = f.insurer
  for (const f of found.sort((a, b) => a.day < b.day ? -1 : 1)) {
    const ins = f.insurer || insurerOf[f.last4]
    if (!ins) continue
    // no policy number (LIC): emails about the same premium (within 3%) are one policy
    let k = ins + "|" + f.last4
    if (!f.last4) { const amt = f.paidAmount || f.premium; k = Object.keys(by).find(x => x.startsWith(ins + "|amt") && Math.abs(by[x].amt - amt) <= amt * 0.03) || `${ins}|amt${f.amountKey}` }
    const s = by[k] = by[k] || { insurer: ins, last4: f.last4, type: f.type, amt: f.paidAmount || f.premium || 0 }
    for (const key of ["name", "freq", "premium", "cover"]) if (f[key]) s[key] = f[key]
    if (f.type && f.type !== "Other" && (s.type === "Other" || s.type === "Life")) s.type = f.type
    if (f.due && (!s.due || f.due > s.due)) s.due = f.due
    if (f.paidDay && (!s.paidDay || f.paidDay >= s.paidDay)) { s.paidDay = f.paidDay; s.paidAmount = f.paidAmount }
    if (f.autopay) s.autopay = true
    if (f.broker) s.broker = f.broker
  }
  return Object.values(by)
}
function addMonthsDay(iso, n) {
  const [y, m, d] = iso.split("-").map(Number), t = new Date(Date.UTC(y, m - 1 + n, 1))
  return `${t.getUTCFullYear()}-${String(t.getUTCMonth() + 1).padStart(2, "0")}-${String(Math.min(d, new Date(Date.UTC(t.getUTCFullYear(), t.getUTCMonth() + 1, 0)).getUTCDate())).padStart(2, "0")}`
}
// The Insurance table: found by name once, or made next to your Expenses table the first time a policy turns up.
function policiesDb(token, create) {
  const props = PropertiesService.getScriptProperties(), saved = props.getProperty("POLICIES_DB")
  if (saved) return saved
  const res = notion(token, "post", "search", { query: "Insurance", filter: { property: "object", value: "database" }, page_size: 20 })
  const hit = (res.results || []).find(d => (d.title || []).map(t => t.plain_text).join("").trim().toLowerCase() === "insurance")
  if (hit) { props.setProperty("POLICIES_DB", hit.id.replace(/-/g, "")); return hit.id.replace(/-/g, "") }
  if (!create) return ""
  // next to your Expenses table, or (if that sits at the top of your workspace, where the API can't add tables) next to Loans or People
  let page = ""
  for (const id of [EXPENSES_DB, typeof LOANS_DB !== "undefined" ? LOANS_DB : "", typeof PEOPLE_DB !== "undefined" ? PEOPLE_DB : ""].filter(Boolean)) {
    try { const d = notion(token, "get", `databases/${id}`); if (d.parent && d.parent.type === "page_id") { page = d.parent.page_id; break } } catch (e) {}
  }
  if (!page) { Logger.log("Insurance: couldn't find a page to add the Insurance table to. Make a table called Insurance in Notion and connect your integration."); return "" }
  const db = notion(token, "post", "databases", { parent: { type: "page_id", page_id: page }, title: [{ type: "text", text: { content: "Insurance" } }], properties: POLICY_COLUMNS() })
  props.setProperty("POLICIES_DB", db.id.replace(/-/g, ""))
  return db.id.replace(/-/g, "")
}

function syncPolicyMails(token) {
  const props = PropertiesService.getScriptProperties(), now = Date.now()
  if (now - Number(props.getProperty("POLICY_MAIL_AT2") || 0) < 20 * 36e5) return { updated: 0, added: 0 }
  const first = !props.getProperty("POLICY_MAIL_AT2")   // "2": looked back a year again once, to pick up LIC emails
  const found = []
  for (const t of GmailApp.search(POLICY_QUERY(first ? 400 : 4), 0, first ? 200 : 40))
    for (const m of t.getMessages()) {
      try { const r = parsePolicyMail({ from: m.getFrom(), subject: m.getSubject(), body: m.getPlainBody(), date: m.getDate() }); if (r) found.push(r) } catch (e) {}
    }
  const summary = policyMailSummary(found)
  let res = { updated: 0, added: 0 }
  if (summary.length) { const db = policiesDb(token, true); if (db) res = applyPolicyMails(token, db, summary, istDay(now)) }
  props.setProperty("POLICY_MAIL_AT2", String(now))
  return res
}
function applyPolicyMails(token, db, summary, today) {
  const date = p => (p && p.date ? p.date.start.slice(0, 10) : "")
  const rows = queryAll(token, db, {}).map(pg => { const p = pg.properties
    return { id: pg.id, insurer: plain(p.Insurer), last4: plain(p["Policy no"]).slice(-4), due: date(p["Next due"]), paid: date(p["Last paid"]), freq: (p.Frequency && p.Frequency.select || {}).name || "",
      premium: (p.Premium || {}).number || 0, cover: (p.Cover || {}).number || 0, autopay: !!(p["Auto-pay"] && p["Auto-pay"].checkbox) } })
  let updated = 0, added = 0
  for (const s of summary) {
    const sameIns = r => !r.insurer || r.insurer.toLowerCase().split(" ")[0] === s.insurer.toLowerCase().split(" ")[0]
    const row = s.last4 ? rows.find(r => r.last4 === s.last4 && sameIns(r))
      : rows.find(r => sameIns(r) && r.premium && Math.abs(r.premium - (s.premium || s.paidAmount || s.amt)) <= r.premium * 0.03)
    const freq = s.freq || (row && row.freq) || (!s.last4 ? "Yearly" : "")
    // the next due date: from a due notice, or a month/quarter/year after the last premium paid
    let due = s.due || ""
    if (s.paidDay && POLICY_FREQ[freq] && (!due || due <= s.paidDay)) {
      // the instalment just paid was due on the saved date (if it's close) or on the email's due date; the next one is a period later
      const paidFor = row && row.due && Math.abs(Date.parse(row.due) - Date.parse(s.paidDay)) <= 25 * 864e5 ? row.due : due && Math.abs(Date.parse(due) - Date.parse(s.paidDay)) <= 25 * 864e5 ? due : s.paidDay
      due = addMonthsDay(paidFor, POLICY_FREQ[freq])
    }
    const props = {}
    if (due && (!row || due > row.due)) props["Next due"] = { date: { start: due } }
    if (s.paidDay && (!row || s.paidDay > row.paid)) { props["Last paid"] = { date: { start: s.paidDay } }; props["Last paid amount"] = { number: s.paidAmount || null } }
    if (freq && (!row || !row.freq)) props.Frequency = { select: { name: freq } }
    const premium = s.premium || s.paidAmount
    if (premium && (!row || !row.premium)) props.Premium = { number: premium }
    if (s.cover && (!row || !row.cover)) props.Cover = { number: s.cover }
    if (s.autopay && (!row || !row.autopay)) props["Auto-pay"] = { checkbox: true }
    if (row) { if (Object.keys(props).length) { notion(token, "patch", `pages/${row.id}`, { properties: props }); updated++ } continue }
    if (s.name && s.name === s.name.toUpperCase()) s.name = s.name.toLowerCase().replace(/\b[a-z]/g, c => c.toUpperCase())
    if (!s.last4) { s.name = s.name || "policy"; props.Notes = { rich_text: [{ text: { content: `From ${s.insurer}'s payment emails, which don't name the policy. Yearly is a guess from one payment; add the policy number and fix it with Edit.` } }] } }
    const title = s.name ? `${s.insurer} · ${s.name.replace(new RegExp("^" + s.insurer.split(" ")[0] + "\\s*", "i"), "")}` : `${s.insurer} ${s.type === "Other" ? "policy" : s.type.toLowerCase()}`
    const shown = s.last4 ? title : `${s.insurer} policy · ₹${Math.round(s.premium || s.paidAmount || s.amt).toLocaleString("en-IN")}`
    Object.assign(props, { Policy: { title: [{ text: { content: shown.slice(0, 120) } }] }, Insurer: { rich_text: [{ text: { content: s.insurer } }] },
      Type: { select: { name: s.type } }, "Policy no": { rich_text: [{ text: { content: s.last4 } }] }, Match: { rich_text: [{ text: { content: [s.insurer === "LIC" ? "LIC, Life Insurance" : s.insurer.replace(/ Life$/, ""), s.broker].filter(Boolean).join(", ") } }] },
      "Grace days": { number: freq === "Monthly" ? 15 : 30 }, Suggested: { checkbox: true }, Active: { checkbox: false } })
    if (!props.Notes) props.Notes = { rich_text: [{ text: { content: `Found in your email (${today})` } }] }
    notion(token, "post", "pages", { parent: { database_id: db }, properties: props })
    added++
  }
  return { updated, added }
}
// ---------- reading a policy document (schedule, renewal notice, premium receipt) ----------
// Shared by the dashboard (PDFs read on the iPhone) and the Google script (scans and emailed PDFs).
// Picks out what's reliable across insurers (policy number, cover, premium, frequency, dates) and, best-effort,
// the benefits worth knowing: bonus, room rent, co-pay, waiting periods, check-ups, riders. Names, dates of birth
// and nominee details are never kept; for who's covered only the relationships are (Self, Spouse, Mother…).
function parsePolicyDoc(text, meta) {
  meta = meta || {}
  const raw = String(text || "").replace(/\u00a0/g, " ").replace(/\r/g, "")
  const lines = raw.split("\n").map(l => l.replace(/[ ]{2,}/g, " ").trim()).filter(Boolean)
  const flat = lines.join("\n")
  const MON = { jan: 1, feb: 2, mar: 3, apr: 4, may: 5, jun: 6, jul: 7, aug: 8, sep: 9, oct: 10, nov: 11, dec: 12 }
  const DATE = "(\\d{1,2}[-\\/. ]\\d{1,2}[-\\/. ]\\d{4}|\\d{1,2}(?:st|nd|rd|th)?[-\\/ ][A-Za-z]{3,9}[-\\/ ,]+\\d{4}|[A-Za-z]{3,9}\\s+\\d{1,2},?\\s+\\d{4}|\\d{4}-\\d{2}-\\d{2})"
  const day = s => {
    s = String(s || "").trim()
    let m = s.match(/^(\d{4})-(\d{2})-(\d{2})$/); if (m) return s
    m = s.match(/^(\d{1,2})[-\/. ](\d{1,2})[-\/. ](\d{4})$/); if (m && +m[2] <= 12) return `${m[3]}-${m[2].padStart(2, "0")}-${m[1].padStart(2, "0")}`
    m = s.match(/^(\d{1,2})(?:st|nd|rd|th)?[-\/ ]([A-Za-z]{3,9})[-\/ ,]+(\d{4})$/); if (m && MON[m[2].slice(0, 3).toLowerCase()]) return `${m[3]}-${String(MON[m[2].slice(0, 3).toLowerCase()]).padStart(2, "0")}-${m[1].padStart(2, "0")}`
    m = s.match(/^([A-Za-z]{3,9})\s+(\d{1,2}),?\s+(\d{4})$/); if (m && MON[m[1].slice(0, 3).toLowerCase()]) return `${m[3]}-${String(MON[m[1].slice(0, 3).toLowerCase()]).padStart(2, "0")}-${m[2].padStart(2, "0")}`
    return ""
  }
  const money = s => { const m = String(s || "").match(/(?:₹|rs\.?|inr)?\s*([\d,]+(?:\.\d{1,2})?)\s*(lakhs?|lacs?|crores?|cr|l)?\b/i); if (!m) return 0
    let n = Number(m[1].replace(/,/g, "")) || 0; const u = (m[2] || "").toLowerCase()
    if (/^cr/.test(u)) n *= 1e7; else if (/^(l|lakh|lac)/.test(u)) n *= 1e5
    return n }
  // the value next to a label: rest of the same line (after ":" or a tab), else the next line
  const after = re => {
    for (let i = 0; i < lines.length; i++) {
      const m = lines[i].match(re); if (!m) continue
      const rest = lines[i].slice(m.index + m[0].length).replace(/^[\s:\-–|\t.]+/, "").trim()
      if (rest) return rest
      if (lines[i + 1]) return lines[i + 1]
    }
    return ""
  }
  const amountAfter = (re, min) => { for (let i = 0; i < lines.length; i++) { if (!re.test(lines[i])) continue
      const seg = lines[i].replace(re, "\u0001").split("\u0001").slice(1).join(" ") + " " + (/^\s*(₹|rs\.?|inr)?\s*\d/i.test(lines[i + 1] || "") ? lines[i + 1] : "")
      const n = money(seg.replace(/\b(19|20)\d{2}\b(?![,\d])/g, " "))
      if (n >= (min || 1)) return n }
    return 0 }
  // labelled values: docKV() reads tables, labels on their own line and label/value blocks; the older patterns fill gaps
  const kv = docKV(raw), g = (labels, kind, o) => kv.get(labels, kind, o)
  const out = { kind: /premium (paid )?receipt|receipt for premium|we have received|(total )?amount received/i.test(flat) ? "receipt" : /renewal (notice|premium|reminder)|due for renewal/i.test(flat) ? "renewal" : "schedule" }
  // insurer
  const INS = [[/tata aia/i, "Tata AIA Life"], [/hdfc ergo/i, "HDFC ERGO"], [/hdfc life/i, "HDFC Life"], [/icici pru/i, "ICICI Pru Life"], [/icici lombard/i, "ICICI Lombard"], [/star health/i, "Star Health"],
    [/niva bupa|max bupa/i, "Niva Bupa"], [/care health|religare/i, "Care Health"], [/aditya birla (health|capital health)/i, "Aditya Birla Health"], [/life insurance corporation/i, "LIC"],
    [/axis max life|max life/i, "Axis Max Life"], [/sbi life/i, "SBI Life"], [/sbi general/i, "SBI General"], [/bajaj allianz life|bajaj life/i, "Bajaj Life"], [/bajaj allianz|bajaj general/i, "Bajaj Allianz"],
    [/go ?digit/i, "Digit"], [/acko/i, "Acko"], [/kotak (mahindra )?life/i, "Kotak Life"], [/manipal ?cigna/i, "ManipalCigna"], [/tata aig/i, "Tata AIG"], [/pnb met ?life/i, "PNB MetLife"], [/reliance general/i, "Reliance General"]]
  const ins = INS.find(([re]) => re.test(flat.slice(0, 4000))) || INS.find(([re]) => re.test(flat))
  if (ins) out.insurer = ins[1]
  // policy number: only the last 4 digits leave this function
  const no = flat.match(/policy\s*(?:no\.?|number|#)\s*[:.\-]?\s*([A-Z]{0,4}[\/-]?\d[\dA-Z\/-]{5,30})/i) || flat.match(/\bpolicy\s+([A-Z]{0,4}\d{6,20})\b/i) || String(meta.filename || "").match(/([A-Z]?\d{8,20})/)
  const noKV = g(["policy\\s*(?:no\\.?|number|#)", "certificate\\s*(?:no\\.?|number)"], "code", { test: v => v.replace(/\D/g, "").length >= 6 })
  if (noKV) out.last4 = noKV.replace(/[^0-9]/g, "").slice(-4)
  else if (no) out.last4 = no[1].replace(/[^0-9]/g, "").slice(-4)
  // plan name
  const plan = after(/\b(plan name|product name|name of (the )?(plan|product|policy)|product)\b(?!\s*(code|uin))/i).replace(/\s*\(?\buin\b[^)]*\)?/i, "").replace(/[|\t].*$/, "").trim()
  if (plan && plan.length <= 70 && !/^\d/.test(plan) && !/\b(uin|code)\b/i.test(plan)) out.name = plan
  // type
  const t = flat.slice(0, 6000)
  out.type = /griha|home shield|home insurance|householder|dwelling|building sum insured/i.test(t) ? "Home"
    : /motor|private car|two[- ]wheeler|\bidv\b|insured declared value/i.test(t) ? "Motor"
    : /(?<!long\s)\bterm\s*(life|plan|insurance|assurance)\b|sampoorna raksha|iprotect|click 2 protect|smart shield|death benefit/i.test(t) ? "Term life"
    : /health|mediclaim|hospitali[sz]ation|optima|floater|top[- ]?up|cashless/i.test(t) ? "Health"
    : /life insurance|sum assured|maturity/i.test(t) ? "Life" : "Other"
  // cover
  out.cover = g(["total\\s*sum\\s*(?:insured|assured)", "sum\\s*(?:insured|assured)\\s*on\\s*death", "(?:basic|base)\\s*sum\\s*(?:insured|assured)", "sum\\s*(?:insured|assured)(?!\\s*(?:per|for each))", "life\\s*cover", "cover(?:age)?\\s*amount", "insured\\s*declared\\s*value", "\\bIDV\\b"], "money",
    { min: 10000, not: /(building|contents?|rider|accidental[\w\s]*|critical[\w\s]*)\s*$/i }) || amountAfter(/\b(basic |base )?sum\s+(insured|assured)(\s+on\s+death)?\b(?!\s*(per|for each))/i, 10000) || amountAfter(/\b(life cover|cover(age)? amount|insured declared value|\bidv\b)/i, 10000) || undefined
  // premium: the total including GST if it's given
  out.premium = g((out.kind === "receipt" ? ["total\\s*amount\\s*received", "amount\\s*received", "premium\\s*received", "total\\s*amount(?:\\s*paid)?", "amount\\s*paid"] : [])
      .concat(["total\\s*premium\\s*(?:payable|amount)?\\s*(?:\\(?incl[^)\\n]*\\)?)?", "gross\\s*premium", "total\\s*(?:installment|instalment)\\s*premium", "(?:installment|instalment|modal)\\s*premium", "renewal\\s*premium", "premium\\s*payable", "premium\\s*amount", "premium(?!\\s*(?:paying|payment)?\\s*(?:term|frequency|mode|due))"]),
    "money", { min: 100, not: /(basic|base|net|rider|annuali[sz]ed|first\s*year)\s*$/i }) || amountAfter(/\b(total (installment |instalment )?premium( payable| amount)?( \(?incl\w*[^)]*\)?)?|gross premium|installment premium|instalment premium|modal premium|premium amount|renewal premium|premium payable)\b/i, 100)
    || amountAfter(/\bpremium\b(?!\s*(paying|payment) term)/i, 100) || undefined
  if (out.kind === "receipt") { const r = flat.match(new RegExp(`received[^\\n]{0,40}?(?:₹|rs\\.?|inr)\\s*([\\d,]+(?:\\.\\d+)?)[^\\n]{0,80}?\\bon\\s+${DATE}`, "i")); if (r) { out.premium = Number(r[1].replace(/,/g, "")); out.paidDay = day(r[2]) } }
  if (out.kind === "receipt" && !out.paidDay) out.paidDay = g(["receipt\\s*date", "date\\s*of\\s*receipt", "(?:premium\\s*)?pa(?:id|yment)\\s*(?:on|date)", "date\\s*of\\s*payment", "transaction\\s*date", "received\\s*on"], "date") || undefined
  // frequency
  const fr = after(/\b(premium (payment )?(frequency|mode)|mode of (premium )?payment|payment (frequency|mode)|frequency)\b/i)
  const FQ = /\b(monthly|quarterly|half[- ]?yearly|semi[- ]?annual|annual(ly)?|yearly|single|yly|hly|qly|mly)\b/i
  const frKV = g(["premium\\s*(?:payment|paying)?\\s*(?:frequency|mode)", "payment\\s*(?:frequency|mode)", "mode\\s*of\\s*(?:premium\\s*)?payment", "\\bmode\\b", "frequency"], "text", { test: v => FQ.test(v) })
  const fq = ((frKV || "").match(FQ) || [])[0] || (fr.match(/monthly|quarterly|half[- ]?yearly|semi[- ]?annual|annual(ly)?|yearly|single/i) || [])[0] || (flat.match(/\b(monthly|quarterly|half[- ]yearly|yearly|annual) (mode|premium)\b/i) || [])[1] || ""
  if (fq) out.freq = /month|^mly$/i.test(fq) ? "Monthly" : /quarter|^qly$/i.test(fq) ? "Quarterly" : /half|semi|^hly$/i.test(fq) ? "Half-yearly" : /single/i.test(fq) ? "Single" : "Yearly"
  else if (/single premium/i.test(flat)) out.freq = "Single"
  else if (out.type === "Health" || out.type === "Motor") out.freq = "Yearly"
  // dates
  const period = flat.match(new RegExp(`(?:policy period|period of (?:insurance|cover)|policy duration)[^\\n]{0,40}?(?:from)?\\s*:?\\s*${DATE}[^\\n]{0,30}?(?:to|till|-)\\s*(?:midnight of\\s*)?${DATE}`, "i"))
  if (period) { out.start = day(period[1]); out.ends = day(period[2]) }
  const dAfter = re => { const v = after(re); const m = v.match(new RegExp(DATE)); return m ? day(m[1]) : "" }
  out.start = out.start || g(["policy\\s*(?:start|commencement|inception)\\s*date", "date\\s*of\\s*(?:risk\\s*)?commencement(?:\\s*of\\s*(?:risk|policy|cover))?", "risk\\s*commencement\\s*date", "commencement\\s*date", "inception\\s*date", "policy\\s*issue\\s*date", "start\\s*date", "(?:period|cover)\\s*from"], "date") || dAfter(/\b(policy (start|commencement|inception) date|date of (risk )?commencement|risk commencement date|commencement date|start date|policy issue date|issue date)\b/i)
  const endD = g(["policy\\s*(?:end|expiry)\\s*date", "expiry\\s*date", "date\\s*of\\s*expiry", "end\\s*date", "valid\\s*(?:till|upto|until)", "maturity\\s*date", "date\\s*of\\s*maturity"], "date") || dAfter(/\b(policy (end|expiry) date|expiry date|date of expiry|maturity date|date of maturity|policy end date|end date|valid (till|upto|until))\b/i)
  if (endD) out.ends = endD
  out.due = g(["next\\s*(?:premium\\s*)?due\\s*date", "next\\s*due", "due\\s*date\\s*of\\s*next\\s*premium", "next\\s*renewal(?:\\s*date)?", "renewal\\s*(?:due\\s*)?date"], "date") || dAfter(/\b(next (premium )?due date|premium due date|due date of (next )?premium|renewal (due )?date|next renewal)\b/i)
  if (!out.due) {
    // a plain "Due Date": on a receipt it's the instalment just paid, so the next one is a period later
    let dd = g(["premium\\s*due\\s*date", "due\\s*date"], "date", { not: /(last|first|final)\s*(premium\s*)?$/i })
    const step = { Monthly: 1, Quarterly: 3, "Half-yearly": 6, Yearly: 12 }[out.freq]
    if (dd && out.kind === "receipt") { if (step) { const [y, m, d] = dd.split("-").map(Number), t = new Date(Date.UTC(y, m - 1 + step, 1)), last = new Date(Date.UTC(t.getUTCFullYear(), t.getUTCMonth() + 1, 0)).getUTCDate()
      dd = `${t.getUTCFullYear()}-${String(t.getUTCMonth() + 1).padStart(2, "0")}-${String(Math.min(d, last)).padStart(2, "0")}` } else dd = "" }
    if (dd) out.due = dd
  }
  if (!out.due && out.ends && (out.type === "Health" || out.type === "Motor" || out.type === "Home") && out.freq !== "Single") out.due = out.ends
  if (out.kind === "receipt") delete out.ends
  const monYear = iso => { if (!iso) return ""; const [y, m] = iso.split("-").map(Number); return new Date(Date.UTC(y, m - 1, 1)).toLocaleDateString("en-IN", { month: "short", year: "numeric", timeZone: "UTC" }) }
  const lastPrem = dAfter(/\b(due date of last premium|last premium due|premium paying term (ends|till|upto))\b/i)
  // who's covered: relationships only
  const rels = []
  const REL = ["Self", "Spouse", "Wife", "Husband", "Son", "Daughter", "Father", "Mother", "Father-in-law", "Mother-in-law", "Brother", "Sister"]
  const memStart = lines.findIndex(l => /insured (person|member|details)s?|details of (the )?insured|members? covered|persons? (insured|covered)|relationship (with|to) (the )?(proposer|policyholder)/i.test(l))
  if (memStart >= 0) for (const l of lines.slice(memStart, memStart + 25)) for (const r of REL) if (new RegExp(`\\b${r}\\b`, "i").test(l) && !rels.includes(r)) rels.push(r)
  const covers = rels.map(r => r === "Wife" || r === "Husband" ? "Spouse" : r).filter((r, i, a) => a.indexOf(r) === i)
  if (covers.length) out.covers = covers.join(", ")
  else if ((out.type === "Term life" || out.type === "Life") && out.kind === "schedule") out.covers = "Self"
  // benefits, best-effort
  const ben = []
  const add = s => { s = s.replace(/\s+/g, " ").trim(); if (s && s.length <= 90 && !ben.some(b => b.toLowerCase() === s.toLowerCase())) ben.push(s) }
  const inrS = n => n >= 1e7 ? `₹${+(n / 1e7).toFixed(2)} Cr` : n >= 1e5 ? `₹${+(n / 1e5).toFixed(1)} L` : "₹" + Math.round(n).toLocaleString("en-IN")
  const plusMonths = (iso, n) => { if (!iso) return ""; const [y, m] = iso.split("-").map(Number), d = new Date(Date.UTC(y, m - 1 + n, 1)); return d.toLocaleDateString("en-IN", { month: "short", year: "numeric", timeZone: "UTC" }) }
  const months = s => { const m = String(s).match(/(\d{1,3})\s*(months?|years?|yrs?|days?)/i); if (!m) return 0; return /day/i.test(m[2]) ? Math.round(+m[1] / 30) : /y/i.test(m[2]) ? +m[1] * 12 : +m[1] }
  const bonus = after(/\b(cumulative bonus|no[- ]claim bonus|ncb|loyalty bonus|bonus sum insured)\b/i)
  if (bonus) { const p = bonus.match(/(\d{1,3})\s*%/), a = money(bonus); add(p ? `No-claim bonus ${p[1]}%` : a >= 10000 ? `No-claim bonus ${inrS(a)}` : "") }
  const room = after(/\broom rent\b/i)
  if (room && !/^\d+$/.test(room)) add(`Room rent: ${room.replace(/[|\t].*$/, "").slice(0, 60)}`)
  const copay = after(/\bco-?pay(ment)?\b/i)
  if (copay) { const p = copay.match(/(\d{1,2})\s*%/); add(p && +p[1] > 0 ? `Co-pay ${p[1]}%` : /\b(nil|no|not applicable|0\s*%)\b/i.test(copay) ? "No co-pay" : "") }
  const ded = amountAfter(/\b(aggregate )?deductible\b/i, 10000)
  if (ded) add(`Deductible ${inrS(ded)} (top-up)`)
  const ped = after(/\bpre-?existing (disease|illness|condition)s?\b[^\n]{0,30}?(waiting)?/i), pedM = months(ped)
  if (pedM) add(`Pre-existing illnesses covered after ${pedM >= 12 && pedM % 12 === 0 ? pedM / 12 + " years" : pedM + " months"}${out.start ? ` (from ${plusMonths(out.start, pedM)})` : ""}`)
  const spec = after(/\bspecific (disease|illness|ailment|treatment)s?\b[^\n]{0,20}?(waiting)?/i), specM = months(spec)
  if (specM) add(`Specific illnesses waiting period ${specM >= 12 && specM % 12 === 0 ? specM / 12 + " years" : specM + " months"}${out.start ? ` (till ${plusMonths(out.start, specM)})` : ""}`)
  const pre = flat.match(/pre[- ]hospitali[sz]ation[^\n]{0,40}?(\d{2,3})\s*days/i), post = flat.match(/post[- ]hospitali[sz]ation[^\n]{0,40}?(\d{2,3})\s*days/i)
  if (pre || post) add(`Hospital expenses ${pre ? pre[1] + " days before" : ""}${pre && post ? " and " : ""}${post ? post[1] + " days after" : ""} a stay`)
  if (/(free|preventive|annual) health check[- ]?up/i.test(flat)) add("Free health check-up")
  if (/\b(restore|recharge|reinstatement|refill|reload) (benefit|of sum insured)|automatic restoration/i.test(flat)) add("Restore benefit: cover refills after a claim")
  if (/\bmaternity\b[^\n]{0,30}(covered|benefit|yes)/i.test(flat)) add("Maternity covered")
  if (/\bayush\b/i.test(flat) && out.type === "Health") add("AYUSH treatment covered")
  if (/terminal illness/i.test(flat) && /life/i.test(out.type)) add("Terminal illness benefit")
  if (/waiver of premium/i.test(flat)) add("Waiver of premium")
  for (const [re, label] of [[/accidental death( benefit)?( rider)?/i, "Accidental death rider"], [/critical illness( benefit)?( rider)?/i, "Critical illness rider"], [/accidental (total and permanent )?disability/i, "Accidental disability rider"]]) {
    const v = amountAfter(re, 10000); if (v) add(`${label} ${inrS(v)}`) }
  const pt = after(/\bpolicy term\b/i), ppt = after(/\bpremium pay(ing|ment) term\b/i)
  if (months(pt) >= 12 && /life/i.test(out.type)) add(`Policy term ${Math.round(months(pt) / 12)} years`)
  if (months(ppt) >= 12 && /life/i.test(out.type)) add(`Premiums for ${Math.round(months(ppt) / 12)} years${lastPrem ? ` (last one ${monYear(lastPrem)})` : ""}`)
  const ncbMotor = after(/\bno claim bonus\b|\bncb\b/i)
  if (out.type === "Motor" && /\d+\s*%/.test(ncbMotor)) add(`NCB ${ncbMotor.match(/\d+\s*%/)[0]}`)
  const bld = amountAfter(/\bbuilding (sum insured|si|value)\b/i, 10000), cnt = amountAfter(/\bcontents? (sum insured|si|value)\b/i, 10000)
  if (bld) add(`Building cover ${inrS(bld)}`); if (cnt) add(`Contents cover ${inrS(cnt)}`)
  if (ben.length) out.benefits = ben.slice(0, 12)
  for (const k of Object.keys(out)) if (out[k] === undefined || out[k] === "" || out[k] === 0) delete out[k]
  const useful = ["cover", "premium", "due", "ends", "benefits", "last4"].filter(k => out[k]).length
  return useful >= 2 || (out.kind === "receipt" && out.premium) ? out : null
}

// ---------- insurance documents: PDFs your insurers email you, and ones you send from the dashboard ----------
// Policy schedules, renewal notices and policy copies attached to insurers' emails (or sent to yourself tagged
// [POLICY-DOC]) are read here, a few per run. Their figures go on the matching policy in the Insurance table for you to
// confirm on the dashboard ("Use these"); a policy not in the table yet is added as a suggestion. Receipts are skipped.
function syncPolicyDocs(token) {
  const props = PropertiesService.getScriptProperties(), now = Date.now()
  if (props.getProperty("DOCS_OFF") === "1") return { read: 0, locked: 0 }
  const seen = new Set(JSON.parse(props.getProperty("POLICY_DOCS_SEEN") || "[]")), first = !props.getProperty("POLICY_DOCS_AT")
  const insurers = '(tataaia OR "tata aia" OR hdfcergo OR "hdfc ergo" OR hdfclife OR "hdfc life" OR icicilombard OR iciciprulife OR starhealth OR "star health" OR nivabupa OR "niva bupa" OR careinsurance OR "care health" OR licindia OR maxlife OR sbilife OR sbigeneral OR bajajallianz OR godigit OR acko OR policybazaar OR adityabirla OR manipalcigna OR tataaig)'
  const q = `has:attachment filename:pdf newer_than:${first ? 400 : 4}d (subject:"[POLICY-DOC" OR ((schedule OR "policy document" OR "policy copy" OR "policy kit" OR renewal OR "policy certificate" OR "policy issued") ${insurers})) -subject:receipt -category:promotions`
  let db = "", rows = null, read = 0, locked = 0, budget = 3
  for (const t of GmailApp.search(q, 0, first ? 40 : 15)) for (const m of t.getMessages()) {
    const subject = m.getSubject(), hint = (subject.match(/\[POLICY-DOC[^\]]*?(\d{4})\]/) || [])[1] || ""
    for (const att of m.getAttachments({ includeInlineImages: false })) {
      if (!/\.pdf$/i.test(att.getName()) && att.getContentType() !== "application/pdf") continue
      if (/receipt|invoice|tax certificate|claim ?form|proposal form|brochure|wording|terms/i.test(att.getName())) continue
      const key = m.getId() + "|" + att.getName()
      if (seen.has(key)) continue
      if (budget-- <= 0) break
      seen.add(key)
      let out; try { out = pdfToText(att.copyBlob()) } catch (e) { out = { error: String(e.message || e) } }
      const fig = out.text ? parsePolicyDoc(out.text, { filename: att.getName() }) : null
      if (fig && hint && !fig.last4) fig.last4 = hint
      if (!fig && !out.locked) continue
      if (!db) {
        db = policiesDb(token, true); if (!db) return { read, locked }
        if (props.getProperty("POLICY_DOC_COL") !== "1") { notion(token, "patch", `databases/${db}`, { properties: { "Doc figures": { rich_text: {} } } }); props.setProperty("POLICY_DOC_COL", "1") }
        rows = policyDocRows(token, db)
      }
      const entry = { file: att.getName().slice(0, 60), day: istDay(m.getDate().getTime()), link: `https://mail.google.com/mail/u/0/#all/${m.getThread().getId()}` }
      if (fig) { entry.fig = fig; read++ } else { entry.locked = true; entry.used = true; locked++ }
      attachPolicyDoc(token, db, rows, entry, fig ? fig.last4 : hint)
    }
  }
  props.setProperty("POLICY_DOCS_SEEN", JSON.stringify([...seen].slice(-300)))
  if (budget > 0) props.setProperty("POLICY_DOCS_AT", String(now))
  return { read, locked }
}
function policyDocRows(token, db) {
  return queryAll(token, db, {}).map(pg => { const p = pg.properties
    return { id: pg.id, insurer: plain(p.Insurer), last4: plain(p["Policy no"]).slice(-4), docs: (() => { try { return JSON.parse(plain(p["Doc figures"]) || "[]") } catch (e) { return [] } })() } })
}
function policyDocsText(docs) {
  let list = docs.slice(-6)
  for (let i = 0; JSON.stringify(list).length > 1990 && i < list.length; i++) if (list[i].used) delete list[i].fig
  while (JSON.stringify(list).length > 1990 && list.length > 1) list = list.slice(1)
  return JSON.stringify(list).slice(0, 1990)
}
function attachPolicyDoc(token, db, rows, entry, last4) {
  const f = entry.fig || {}
  let row = last4 && rows.find(r => r.last4 === last4 && (!f.insurer || !r.insurer || r.insurer.split(" ")[0].toLowerCase() === f.insurer.split(" ")[0].toLowerCase()))
  if (!row && f.last4 && f.insurer && (f.cover || f.premium)) {
    // a policy not in the table yet: added as a suggestion, already filled in from the document
    const rt = v => ({ rich_text: v ? [{ text: { content: String(v).slice(0, 1900) } }] : [] })
    const name = f.name && f.name === f.name.toUpperCase() ? f.name.toLowerCase().replace(/\b[a-z]/g, c => c.toUpperCase()) : f.name
    const props = { Policy: { title: [{ text: { content: `${f.insurer}${name ? " · " + name.replace(new RegExp("^" + f.insurer.split(" ")[0] + "\\s*", "i"), "") : ""}`.slice(0, 120) } }] },
      Insurer: rt(f.insurer), Type: { select: { name: f.type || "Other" } }, "Policy no": rt(f.last4), Match: rt(f.insurer.replace(/ Life$/, "")), Covers: rt(f.covers),
      Benefits: rt((f.benefits || []).join("\n")), Suggested: { checkbox: true }, Active: { checkbox: false }, "Grace days": { number: f.freq === "Monthly" ? 15 : 30 },
      Notes: rt(`Found in a PDF from your email (${entry.day})`), "Doc figures": rt(policyDocsText([Object.assign({}, entry, { fig: undefined, used: true })])) }
    if (f.cover) props.Cover = { number: f.cover }
    if (f.premium && f.kind !== "receipt") props.Premium = { number: f.premium }
    if (f.freq) props.Frequency = { select: { name: f.freq } }
    if (f.due) props["Next due"] = { date: { start: f.due } }
    if (f.ends) props.Ends = { date: { start: f.ends } }
    if (f.start) props.Start = { date: { start: f.start } }
    const pg = notion(token, "post", "pages", { parent: { database_id: db }, properties: props })
    rows.push({ id: pg.id, insurer: f.insurer, last4: f.last4, docs: [] })
    return
  }
  if (!row) return
  row.docs = row.docs.filter(d => !(d.file === entry.file && d.day === entry.day)).concat([entry])
  notion(token, "patch", `pages/${row.id}`, { properties: { "Doc figures": { rich_text: [{ text: { content: policyDocsText(row.docs) } }] } } })
}

// ---------- cards and bank accounts you add, rename or remove on the dashboard ----------
// The dashboard (Settings › Cards & bank accounts) keeps them in Notion: cards as the Card options of Expenses
// ("HDFC Regalia ••1234", "HDFC Debit ••1234"), bank accounts as the Account options of CC Transactions
// ("HDFC ••8760"), and removed cards on the Card Bills description ("Removed cards: ••3943"). Read each run.
const CR_WORDS = [
  [/scapia|federal/i, /scapia|federal bank/i], [/hdfc|diners|regalia|infinia|millennia|tata neu|swiggy/i, /hdfc bank|diners/i],
  [/axis/i, /axis bank/i], [/amex|american express/i, /\bamex\b|american express/i], [/icici/i, /icici bank/i], [/\bsbi\b/i, /\bsbi\b/i],
  [/kotak/i, /kotak/i], [/idfc/i, /idfc first/i], [/yes bank/i, /yes bank/i], [/indusind/i, /indusind/i], [/hsbc/i, /hsbc/i],
  [/onecard|one card/i, /onecard|one card/i], [/\bau\b/i, /\bau (small finance )?bank\b/i], [/rbl/i, /\brbl\b/i],
  [/bank of baroda|\bbob\b/i, /bank of baroda|\bbob\b/i], [/\bpnb\b|punjab national/i, /punjab national|\bpnb\b/i], [/canara/i, /canara/i], [/union bank/i, /union bank/i],
]
let ACCOUNT_NAMES = []
const crLast4 = d => String(d || "").slice(-4)
const crDigitsOf = n => (String(n).match(/(\d{3,6})\s*$/) || [])[1] || ""
function removedCardDigits(token) {
  try {
    if (!BILLS_DB) return []
    const d = notion(token, "get", `databases/${BILLS_DB}`), desc = (d.description || []).map(t => t.plain_text).join("")
    const m = desc.match(/Removed cards:\s*([^\n]*)/)
    return m ? (m[1].match(/\d{3,6}/g) || []).map(crLast4) : []
  } catch (e) { return [] }
}
function loadCardRegistry(token) {
  try {
    const gone = removedCardDigits(token)
    const db = notion(token, "get", `databases/${EXPENSES_DB}`)
    const named = ((((db.properties || {}).Card || {}).select || {}).options || []).map(o => o.name)
      .filter(n => !/^(example card|bank account|unknown)/i.test(n)).map(n => ({ name: n, d: crDigitsOf(n) })).filter(x => x.d)
    const same = (c, x) => c.digits.some(d => crLast4(d) === crLast4(x.d))
    // renamed: follow the new name (same last digits)
    for (const c of CARDS) {
      const hit = c.digits.length && named.find(x => same(c, x))
      if (hit && hit.name !== c.name) {
        if (typeof CREDIT_CARDS !== "undefined") { const i = CREDIT_CARDS.indexOf(c.name); if (i >= 0) CREDIT_CARDS[i] = hit.name }
        c.name = hit.name
      }
    }
    // added: recognised by its last digits (debit cards only from alerts that say "debit card")
    for (const x of named) if (!gone.includes(crLast4(x.d)) && !CARDS.some(c => same(c, x))) {
      const debit = /\bdebit\b/i.test(x.name), w = CR_WORDS.find(([t]) => t.test(x.name))
      const words = debit ? /$^/ : w ? w[1] : new RegExp(x.name.split(/\s+/)[0].replace(/[^A-Za-z]/g, "") || "$^", "i")
      CARDS.push({ name: x.name, digits: [x.d], words, debit })
      if (!debit && typeof CREDIT_CARDS !== "undefined" && !CREDIT_CARDS.includes(x.name)) CREDIT_CARDS.push(x.name)
    }
    for (const c of CARDS) if (/\bdebit\b/i.test(c.name)) { c.debit = true; c.words = /$^/ }
    // removed: no longer tagged
    for (let i = CARDS.length - 1; i >= 0; i--) if (CARDS[i].digits.some(d => gone.includes(crLast4(d)))) CARDS.splice(i, 1)
  } catch (e) { Logger.log("cards from Notion: " + e) }
  try { const a = notion(token, "get", `databases/${ALERTS_DB}`); ACCOUNT_NAMES = ((((a.properties || {}).Account || {}).select || {}).options || []).map(o => o.name) } catch (e) { ACCOUNT_NAMES = [] }
}
// A bank alert that names one of your debit cards is tagged to it; one from an account you added gets its name.
function tagOwn(r, text) {
  text = String(text || "")
  if ((r.card === "Bank account" || r.card === "Unknown") && /debit\s*card/i.test(text)) {
    const c = CARDS.find(c => c.debit && c.digits.some(d => hasDigits(text, d) || hasDigits(text, crLast4(d))))
    if (c) { r.card = c.name; r.channel = "Card" }
  }
  if (ACCOUNT_NAMES.length && r.card === "Bank account") {
    const m = text.match(/(?:a\/c|acct|account)(?:\s*(?:no\.?|number))?\s*(?:ending(?:\s*(?:in|with))?)?\s*[:\-]?\s*[x*•.\s]*(\d{3,6})\b/i)
    const acc = m && ACCOUNT_NAMES.find(n => crLast4(crDigitsOf(n)) === crLast4(m[1]))
    if (acc) r.account = acc
  }
}
